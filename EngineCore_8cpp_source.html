96/**
97 * @copydoc EngineCore::tick()
98 */
99void asge::EngineCore::tick()
100{
101 if (!_services._running.load())
102 return;
103
104 using clock = std::chrono::high_resolution_clock;
105 auto t_start = clock::now();
106
107 uint64_t local_decoded = 0;
108 uint64_t local_popped = 0;
109 auto t_dec0 = clock::now();
110
111 if (_services._network && _services._gameRunning.load()) {
112 std::deque<asge::network::PacketEnvelope> local_queue;
113 {
114 std::lock_guard<std::mutex> lk(_net_queue_mutex);
115 if (!_net_queue.empty()) {
116 local_popped = _net_queue.size();
117 local_queue.swap(_net_queue);
118 }
119 }
120
121 for (auto &env : local_queue) {
122 try {
123 std::size_t off = 0;
124 _services.protocol().decode(env.payload.data(), env.payload.size(), off);
125 ++local_decoded;
126 }
127 catch (const std::exception &e) {
128 if (_services._debug)
129 _services.debug().log(asge::debug::Level::Error,
130 std::string("[EngineCore] decode failed: ") + e.what());
131 }
132 catch (...) {
133 if (_services._debug)
134 _services.debug().log(asge::debug::Level::Error,
135 "[EngineCore] decode failed: unknown");
136 }
137 }
138 }
139 auto t_dec1 = clock::now();
140
141 if (_services._timer) _services.timer().tick();
142 const double dt = (_services._timer) ? _services.timer().deltaTime()
143 : _services.config().fixedDeltaSeconds;
144
145 if (_services._input) {
146 try {
147 _services.input().update();
148 if (_services._world)
149 _services.input().updatePlayerInputs(_services);
150 }
151 catch (...) {
152 if (_services._debug)
153 _services.debug().log(asge::debug::Level::Error, "[EngineCore] input.update() threw");
154 }
155 }
156
157 if (_services._scene) {
158 _services.scene().flushTransitions();
159 _services.scene().update(dt);
160 }
161
162 if (_services._scripts) {
163 try {
164 if (_services.config().runMode == asge::EngineRunMode::Server)
165 _services.scripts().callHook("on_update", dt);
166 auto actions = _services.scripts().drainQueuedActions();
167 if (!actions.empty()) {
168 script::ActionHandler::processScriptActions(_services, actions);
169 }
170 }
171 catch (const std::exception &e) {
172 if (_services._debug)
173 _services.debug().log(asge::debug::Level::Error, std::string("[EngineCore] scripts.on_update threw: ") + e.what());
174 }
175 }
176
177 if (_services._events)
178 _services.events().flush();
179
180 if (_services._world && _movement_system)
181 _movement_system->update(_services, static_cast<float>(dt));
182 if (_services._world && _services.config().runMode == asge::EngineRunMode::Server
183 && _services._gameRunning.load()) {
184 try {
185 if (_player_system)
186 _player_system->changeDirection(_services);
187 if (_player_system)
188 _player_system->replaceInsideScreen(_services);
189 if (_collision_system)
190 _collision_system->update(_services);
191 if (_player_system)
192 _player_system->fire(_services, static_cast<float>(dt));
193
194 if (_level_spawner && _movement_system) {
195 auto *pos = _services.world().getComponent<components::position_attr_t>(0);
196 auto *spd = _services.world().getComponent<components::speed_attr_t>(0);
197 auto *dir = _services.world().getComponent<components::direction_attr_t>(0);
198 if (pos && spd && dir) {
199 _accumulated_distance +=
200 static_cast<std::uint64_t>(std::abs(dir->dx) * (spd->speed / 100.0f) * dt * 100.0f);
201 _level_spawner->update(_accumulated_distance);
202 }
203 }
204 }
205 catch (const std::exception &e) {
206 if (_services._debug) {
207 _services.debug().log(asge::debug::Level::Error,
208 std::string("[EngineCore] world update failed: ") + e.what());
209 }
210 }
211 catch (...) {
212 if (_services._debug) {
213 _services.debug().log(asge::debug::Level::Error,
214 "[EngineCore] world update failed: unknown");
215 }
216 }
217 }
218
219 auto t_r0 = clock::now();
220 if (_services._renderer) {
221 if (_services.renderer().shouldClose()) {
222 _services._running = false;
223 _services._gameRunning.store(false);
224 return;
225 }
226
227 try {
228 _services.renderer().beginFrame();
229 _services.renderer().clear(0.0f, 0.0f, 0.0f, 255.0f);
230
231 if (_render_system)
232 _render_system->render(_services);
233
234 if (_services._scene)
235 _services.scene().draw();
236
237 _services.renderer().submitFrame();
238 }
239 catch (const std::exception &e) {
240 if (_services._debug)
241 _services.debug().log(asge::debug::Level::Error, std::string("[EngineCore] render threw: ") + e.what());
242 }
243 }
244 auto t_r1 = clock::now();
245
246 if (_hooks.onUpdateFixed) {
247 try {
248 _hooks.onUpdateFixed(_services, _services.config().fixedDeltaSeconds);
249 }
250 catch (...) {}
251 }
252 if (_hooks.onUpdateVar) {
253 try {
254 _hooks.onUpdateVar(_services, dt);
255 }
256 catch (...) {}
257 }
258
259 auto t_end = clock::now();
260
261 g_tickstats.frames++;
262 g_tickstats.pkts_decoded += local_decoded;
263 g_tickstats.pkts_queue_popped += local_popped;
264 g_tickstats.ms_decode += std::chrono::duration<double, std::milli>(t_dec1 - t_dec0).count();
265 g_tickstats.ms_render += std::chrono::duration<double, std::milli>(t_r1 - t_r0).count();
266 g_tickstats.ms_total += std::chrono::duration<double, std::milli>(t_end - t_start).count();
267}

270/**
271 * @copydoc EngineCore::runLoop()
272 */
273void asge::EngineCore::runLoop()
274{
275 if (_services.config().runMode == asge::EngineRunMode::Server) {
276 auto templ = _services.scripts().getTemplate("player");
277 if (!templ) {
278 _services.debug().log(debug::Level::Error, "No player template found. Aborting.");
279 return;
280 }
281
282 std::vector<uint16_t> comp_ids { net::GameProtocol::AvailableComponents::POSITION };
283 for (int i = 0; i < _player_cpt; ++i) {
284 uint32_t entity_id = asge::script::EntityFactory::createFromTemplate(_services, *templ);
285 asge::network::EngineProtocol::sendEntityCreate(_services, entity_id);
286
287 auto pos = _services.world().getComponent<components::position_attr_t>(i);
288 pos->x = 300.0;
289 pos->y = 100.0 * (i + 1);
290
291 asge::network::EngineProtocol::sendEntityComponents(_services, entity_id, comp_ids);
292 }
293
294 _services._gameRunning.store(true);
295 }
296
297 if (_services._timer) {
298 try {
299 _services.timer().reset();
300 }
301 catch (...) {}
302 }
303
304 while (_services._running.load()) {
305 auto frame_start = std::chrono::steady_clock::now();
306 tick();
307
308 if (_services.config().runMode == asge::EngineRunMode::Server) {
309 const auto target = std::chrono::duration<double>(_services.config().fixedDeltaSeconds);
310 auto frame_end = std::chrono::steady_clock::now();
311 auto elapsed = frame_end - frame_start;
312
313 if (elapsed < target) {
314 std::this_thread::sleep_for(
315 std::chrono::duration_cast<std::chrono::milliseconds>(target - elapsed));
316 }
317 }
318 }
319}

359/**
360 * @copydoc EngineCore::connectToServer(const std::string &, unsigned short)
361 */
362bool asge::EngineCore::connectToServer(const std::string& addr, unsigned short port)
363{
364 if (!_services._network || !_services.config().enableNetwork)
365 return false;
366
367 auto *netmgr = dynamic_cast<asge::network::NetworkManager*>(_services._network.get());
368 if (!netmgr) {
369 if (_services._debug)
370 _services.debug().log(asge::debug::Level::Error,
371 "[EngineCore::connectToServer] NetworkManager cast failed");
372 return false;
373 }
374
375 if (addr.empty() || port == 0) {
376 if (_services._debug)
377 _services.debug().log(asge::debug::Level::Warning,
378 "[EngineCore::connectToServer] invalid endpoint");
379 return false;
380 }
381
382 std::optional<std::uint32_t> maybe_id;
383 try {
384 maybe_id = netmgr->getConnId(addr, port);
385 }
386 catch (...) {
387 maybe_id = std::nullopt;
388 }
389
390 if (!maybe_id.has_value()) {
391 try {
392 netmgr->connectTo(addr, port);
393 }
394 catch (const std::exception &e) {
395 if (_services._debug)
396 _services.debug().log(asge::debug::Level::Warning,
397 std::string("[EngineCore::connectToServer] connectTo threw: ") + e.what());
398 }
399 catch (...) {
400 if (_services._debug)
401 _services.debug().log(asge::debug::Level::Warning,
402 "[EngineCore::connectToServer] connectTo threw unknown");
403 }
404 }
405
406 std::uint32_t connection_id = 0;
407 try {
408 connection_id = netmgr->getOrCreateId(addr, port);
409 }
410 catch (const std::exception &e) {
411 if (_services._debug)
412 _services.debug().log(asge::debug::Level::Error,
413 std::string("[EngineCore::connectToServer] getOrCreateId failed: ") + e.what());
414 return false;
415 }
416
417 std::vector<std::uint8_t> req = { asge::network::MSG_CONNECT_REQUEST };
418
419 try {
420 netmgr->sendReliable(connection_id, req, 0);
421 }
422 catch (const std::exception &e) {
423 if (_services._debug)
424 _services.debug().log(asge::debug::Level::Warning,
425 std::string("[EngineCore::connectToServer] sendReliable failed: ") + e.what());
426 return false;
427 }
428 catch (...) {
429 if (_services._debug)
430 _services.debug().log(asge::debug::Level::Warning,
431 "[EngineCore::connectToServer] sendReliable failed: unknown");
432 return false;
433 }
434
435 const auto start = std::chrono::steady_clock::now();
436 const auto timeout = std::chrono::seconds(3);
437 const auto deadline = start + timeout;
438 bool accepted = false;
439 std::uint32_t assigned_id = 0;
440
441 while (!accepted && std::chrono::steady_clock::now() < deadline) {
442 {
443 std::lock_guard<std::mutex> lk(_net_queue_mutex);
444
445 if (!_net_queue.empty()) {
446 for (auto it = _net_queue.begin(); it != _net_queue.end(); ) {
447 auto &env = *it;
448
449 if (env.channel == 0 && !env.payload.empty()) {
450 if (env.payload[0] == asge::network::MSG_CONNECT_ACCEPT &&
451 env.payload.size() >= 1 + 4) {
452 std::uint32_t id = 0;
453 id |= static_cast<std::uint32_t>(env.payload[1]) << 0;
454 id |= static_cast<std::uint32_t>(env.payload[2]) << 8;
455 id |= static_cast<std::uint32_t>(env.payload[3]) << 16;
456 id |= static_cast<std::uint32_t>(env.payload[4]) << 24;
457 assigned_id = id;
458 accepted = true;
459
460 it = _net_queue.erase(it);
461 break;
462 }
463 }
464 ++it;
465 }
466 }
467 }
468
469 if (accepted)
470 break;
471
472 try {
473 netmgr->poll([&](const std::vector<std::uint8_t> & /*payload*/) {});
474 }
475 catch (...) {}
476
477 std::this_thread::sleep_for(std::chrono::milliseconds(5));
478 }
479
480 if (!accepted) {
481 if (_services._debug)
482 _services.debug().log(asge::debug::Level::Warning,
483 "[EngineCore::connectToServer] handshake timed out (no CONNECT_ACCEPT)");
484 return false;
485 }
486
487 netmgr->getOrCreateId(addr, port);
488
489 if (_services._debug)
490 _services.debug().log(asge::debug::Level::Info,
491 "[EngineCore::connectToServer] handshake complete, assigned id=" +
492 std::to_string(assigned_id));
493
494 _services.player_id = assigned_id;
495 return true;
496}

498void asge::EngineCore::pollPlayer(std::optional<std::chrono::milliseconds> timeout)
499{
500 if (!_services._network) {
501 if (_services._debug)
502 _services.debug().log(asge::debug::Level::Error,
503 "[EngineCore::pollPlayer] network service absent");
504 return;
505 }
506
507 auto *netmgr = dynamic_cast<asge::network::NetworkManager*>(_services._network.get());
508 if (!netmgr) {
509 if (_services._debug)
510 _services.debug().log(asge::debug::Level::Error,
511 "[EngineCore::pollPlayer] NetworkManager cast failed");
512 return;
513 }
514
515 const auto start = std::chrono::steady_clock::now();
516 const auto deadline = timeout ? (start + *timeout) :
517 std::chrono::steady_clock::time_point::max();
518 bool accepted = false;
519
520 while (!accepted && std::chrono::steady_clock::now() < deadline) {
521 {
522 std::deque<asge::network::PacketEnvelope> local_queue;
523
524 {
525 std::lock_guard<std::mutex> lk(_net_queue_mutex);
526
527 if (!_net_queue.empty())
528 local_queue.swap(_net_queue);
529 }
530
531 for (auto &env : local_queue) {
532 if (env.channel != 0 || env.payload.empty())
533 continue;
534
535 const std::uint8_t message_type = env.payload[0];
536 if (message_type != asge::network::MSG_CONNECT_REQUEST)
537 continue;
538
539 const std::uint32_t connection_id = netmgr->getOrCreateId(env.fromAddr, env.fromPort);
540
541 std::vector<std::uint8_t> reply;
542 reply.reserve(1 + 4);
543 reply.push_back(asge::network::MSG_CONNECT_ACCEPT);
544 reply.push_back(static_cast<std::uint8_t>((connection_id >> 0) & 0xFF));
545 reply.push_back(static_cast<std::uint8_t>((connection_id >> 8) & 0xFF));
546 reply.push_back(static_cast<std::uint8_t>((connection_id >> 16) & 0xFF));
547 reply.push_back(static_cast<std::uint8_t>((connection_id >> 24) & 0xFF));
548
549 try {
550 netmgr->sendReliable(connection_id, reply, 0);
551
552 _player_ids_conns.push_back(std::make_pair(_player_cpt, connection_id));
553 ++_player_cpt;
554
555 if (_services._debug) {
556 _services.debug().log(asge::debug::Level::Info,
557 "[EngineCore::pollPlayer] sent CONNECT_ACCEPT to " +
558 env.fromAddr + ":" + std::to_string(env.fromPort) +
559 " id=" + std::to_string(connection_id));
560 }
561 }
562 catch (const std::exception &e) {
563 if (_services._debug) {
564 _services.debug().log(asge::debug::Level::Warning,
565 std::string("[EngineCore::pollPlayer] sendReliable failed: ") + e.what());
566 }
567 continue;
568 }
569 catch (...) {
570 if (_services._debug) {
571 _services.debug().log(asge::debug::Level::Warning,
572 "[EngineCore::pollPlayer] sendReliable failed: unknown");
573 }
574 continue;
575 }
576
577 accepted = true;
578 break;
579 }
580 }
581
582 if (accepted)
583 break;
584
585 try {
586 netmgr->poll([&](const std::vector<std::uint8_t> &/*payload*/) {});
587 }
588 catch (const std::exception &e) {
589 if (_services._debug)
590 _services.debug().log(asge::debug::Level::Error,
591 std::string("[EngineCore::pollPlayer] poll exception: ") + e.what());
592 }
593 catch (...) {
594 if (_services._debug)
595 _services.debug().log(asge::debug::Level::Error,
596 "[EngineCore::pollPlayer] poll unknown exception");
597 }
598
599 std::this_thread::sleep_for(std::chrono::milliseconds(50));
600 }
601
602 if (!accepted && _services._debug) {
603 _services.debug().log(asge::debug::Level::Info,
604 "[EngineCore::pollPlayer] no player accepted (timeout or cancelled)");
605 }
606 return;
607}

760/**
761 * @brief Bootstraps the scripting subsystem and loads initial packages.
762 */
763void asge::EngineCore::initScripts()
764{
765 if (_services._scripts)
766 return;
767 if (!_services._debug)
768 initDebug();
769 if (!_services._world)
770 initWorld();
771 if (!_services._storage)
772 initStorage();
773
774 _services._scripts = std::make_unique<script::ScriptManager>(_services,
775 std::make_unique<script::JSONRuntime>(_services.debug()));
776
777 bool res = _services.scripts().loadPackage(_services.config().gameId, _services.config().assetsRoot);
778 if (!res) {
779 if (_services._debug)
780 _services.debug().log(asge::debug::Level::Error,

837/**
838 * @brief Launches the networking layer and worker thread.
839 */
840void asge::EngineCore::initNetwork()
841{
842 if (!_services.config().enableNetwork) {
843 _services._network.reset();
844 return;
845 }
846
847 std::unique_ptr<asge::network::NetworkChannel> chan;
848 switch (_services.config().networkTransport) {
849 case NetworkTransportKind::UDP:
850 case NetworkTransportKind::Auto:
851 default:
852 chan = std::make_unique<asge::network::UDPChannel>(_services._debug != nullptr);
853 break;
854 case NetworkTransportKind::Loopback:
855 chan = std::make_unique<asge::network::UDPChannel>(_services._debug != nullptr);
856 break;
857 case NetworkTransportKind::Null:
858 _services._network.reset();
859 return;
860 }
861
862 _services._network = std::make_unique<asge::network::NetworkManager>(
863 std::move(chan),
864 _services._debug != nullptr
865 );
866
867 auto* netmgr = dynamic_cast<asge::network::NetworkManager*>(_services._network.get());
868 if (!netmgr) {
869 if (_services._debug)
870 _services.debug().log(asge::debug::Level::Error,
871 "[EngineCore] initNetwork(): NetworkManager instance missing ou invalide");
872 return;
873 }
874
875 netmgr->setHandler(0, [this](const asge::network::PacketEnvelope& env) {
876 std::lock_guard<std::mutex> lk(_net_queue_mutex);
877 _net_queue.push_back(env); // copie du PacketEnvelope
878 });
879
880 netmgr->setHandler(1, [this](const asge::network::PacketEnvelope& env) {
881 std::lock_guard<std::mutex> lk(_net_queue_mutex);
882 _net_queue.push_back(env); // copie du PacketEnvelope
883 });
884
885 if (_services._debug)
886 _services.debug().log(asge::debug::Level::Info, "[EngineCore] NetworkManager initialized");
887
888 if (!_services._game_protocol) {
889 _services._game_protocol = std::make_unique<asge::net::GameProtocol>(
890 _services, 1);
891 }
892
893 if (_services._network) {
894 _networkThreadRunning.store(true);
895 _networkThread = std::thread([this]() {
896 auto fallback = [this](const std::vector<std::uint8_t> &payload) {
897 asge::network::PacketEnvelope env{ payload, std::string{}, 0, 0 };
898
899 {
900 std::lock_guard<std::mutex> lk(_net_queue_mutex);
901 _net_queue.push_back(std::move(env));
902 }
903 };
904
905 while (_networkThreadRunning.load()) {
906 try {
907 _services.network().poll(fallback);
908 }
909 catch (const std::exception& e) {
910 if (_services._debug) {
911 _services.debug().log(asge::debug::Level::Error,
912 std::string("[EngineCore::netThread] poll exception: ") + e.what());
913 }
914 }
915 catch (...) {
916 if (_services._debug) {
917 _services.debug().log(asge::debug::Level::Error,
918 "[EngineCore::netThread] poll exception (unknown)");
919 }
920 }
921
922 std::this_thread::sleep_for(_services._running.load()
923 ? std::chrono::milliseconds(5)
924 : std::chrono::milliseconds(10));
925 }
926
927 if (_services._debug)
928 _services.debug().log(asge::debug::Level::Info, "[EngineCore::netThread] exiting");
929 });
930
931 if (_services._debug)
932 _services.debug().log(asge::debug::Level::Info, "[EngineCore] Network thread started");
933 }
934}
