29/**
30 * @brief Packs two 32-bit signed integers into a single 64-bit key.
31 *
32 * @param x The X coordinate.
33 * @param y The Y coordinate.
34 * @return A 64-bit unsigned integer encoding both coordinates.
35 *
36 * @note The function casts both coordinates to unsigned 32-bit integers
37 * before combining them with bit shifting and OR operations.
38 */
39uint64_t makeKey(int x, int y) noexcept
40{
41 uint64_t ux = static_cast<uint32_t>(x);
42 uint64_t uy = static_cast<uint32_t>(y);
43
44 return (ux << 32) | uy;
45}

49/**
50 * @brief Decodes a 64-bit key back into its original (x, y) coordinates.
51 *
52 * @param key The packed 64-bit value.
53 * @return A pair of signed integers (x, y) extracted from the key.
54 *
55 * @note The higher 32 bits contain X, and the lower 32 bits contain Y.
56 */
57std::pair<int, int> decodeKey(uint64_t key) noexcept
58{
59 int32_t x = static_cast<int32_t>(key >> 32);
60 int32_t y = static_cast<int32_t>(key & 0xffffffffu);
61
62 return std::make_pair(x, y);
63}
