26void asge::world::systems::PlayerSystem::changeDirection(Services &services)
27{
28 World &game_world = services.world();
29
30 for (const uint32_t entity_id : _filtered_entities) {
31 components::input_attr_t *input_component =
32 game_world.getComponent<components::input_attr_t>(entity_id);
33 components::direction_attr_t *direction_component =
34 game_world.getComponent<components::direction_attr_t>(entity_id);
35
36 if (!direction_component || !input_component) {
37 continue;
38 }
39
40 const int8_t player_pad = input_component->pad;
41 if (player_pad == components::PLAYER_PAD::NO_PAD) {
42 continue;
43 }
44
45 if (player_pad == components::PLAYER_PAD::UP ||
46 player_pad == components::PLAYER_PAD::UP_LEFT ||
47 player_pad == components::PLAYER_PAD::UP_RIGHT) {
48 direction_component->dy = -1.0;
49
50 if (player_pad == components::PLAYER_PAD::UP_LEFT) {
51 direction_component->dx = -1.0;
52 }
53 else if (player_pad == components::PLAYER_PAD::UP_RIGHT) {
54 direction_component->dx = 1.0;
55 }
56 else {
57 direction_component->dx = 0.0;
58 }
59 }
60 else if (player_pad == components::PLAYER_PAD::DOWN ||
61 player_pad == components::PLAYER_PAD::DOWN_LEFT ||
62 player_pad == components::PLAYER_PAD::DOWN_RIGHT) {
63 direction_component->dy = 1.0;
64
65 if (player_pad == components::PLAYER_PAD::DOWN_LEFT) {
66 direction_component->dx = -1.0;
67 }
68 else if (player_pad == components::PLAYER_PAD::DOWN_RIGHT) {
69 direction_component->dx = 1.0;
70 }
71 else {
72 direction_component->dx = 0.0;
73 }
74 }
75 else if (player_pad == components::PLAYER_PAD::LEFT) {
76 direction_component->dx = -1.0;
77 direction_component->dy = 0.0;
78 }
79 else if (player_pad == components::PLAYER_PAD::RIGHT) {
80 direction_component->dx = 1.0;
81 direction_component->dy = 0.0;
82 }
83
84 std::vector<uint16_t> comp_ids { net::GameProtocol::AvailableComponents::DIRECTION };
85 network::EngineProtocol::sendEntityComponents(services, entity_id, comp_ids);
86 }
87}

92void asge::world::systems::PlayerSystem::replaceInsideScreen(Services &services)
93{
94 world::World &game_world = services.world();
95
96 for (const uint32_t entity_id : _filtered_entities) {
97 components::position_attr_t *position_component =
98 game_world.getComponent<components::position_attr_t>(entity_id);
99 components::hitbox_attr_t *hitbox_component =
100 game_world.getComponent<components::hitbox_attr_t>(entity_id);
101
102 if (!position_component || !hitbox_component) {
103 continue;
104 }
105
106 const float position_x = position_component->x;
107 const float position_y = position_component->y;
108 const float half_hitbox_height = static_cast<float>(hitbox_component->height) * 0.5f;
109 const float half_hitbox_width = static_cast<float>(hitbox_component->width) * 0.5f;
110 float new_position_x = position_x;
111 float new_position_y = position_y;
112
113 if (position_y < half_hitbox_height) {
114 new_position_y = half_hitbox_height;
115 }
116 else if (position_y > VIRTUAL_SCREEN_HEIGHT - half_hitbox_height) {
117 new_position_y = static_cast<float>(VIRTUAL_SCREEN_HEIGHT) - half_hitbox_height;
118 }
119
120 if (position_x < half_hitbox_width) {
121 new_position_x = half_hitbox_width;
122 }
123 else if (position_x > VIRTUAL_SCREEN_WIDTH - half_hitbox_width) {
124 new_position_x = static_cast<float>(VIRTUAL_SCREEN_WIDTH) - half_hitbox_width;
125 }
126
127 if (position_x != new_position_x || position_y != new_position_y) {
128 position_component->x = new_position_x;
129 position_component->y = new_position_y;
130
131 std::vector<uint16_t> comp_ids { net::GameProtocol::AvailableComponents::POSITION };
132 network::EngineProtocol::sendEntityComponents(services, entity_id, comp_ids);
133 }
134 }

140void asge::world::systems::PlayerSystem::fire(Services &services, float tick_count)
141{
142 static_cast<void>(tick_count);
143 world::World &game_world = services.world();
144
145 for (const uint32_t entity_id : _filtered_entities) {
146 components::position_attr_t *position_component =
147 game_world.getComponent<components::position_attr_t>(entity_id);
148 components::hitbox_attr_t *hitbox_component =
149 game_world.getComponent<components::hitbox_attr_t>(entity_id);
150 components::input_attr_t *input_component =
151 game_world.getComponent<components::input_attr_t>(entity_id);
152
153 if (!position_component || !hitbox_component || !input_component) {
154 continue;
155 }
156 if (!input_component->fire) {
157 continue;
158 }
159
160 auto templ = services.scripts().getTemplate("bullet");
161 if (!templ)
162 continue;
163
164 const float bullet_origin_x = static_cast<float>(position_component->x + hitbox_component->width);
165 const float bullet_origin_y = static_cast<float>(position_component->y);
166
167 uint32_t bullet_id = script::EntityFactory::createFromTemplate(services, *templ,
168 bullet_origin_x, bullet_origin_y);
169 network::EngineProtocol::sendEntityCreate(services, bullet_id);
170
171 components::master_attr_t master { entity_id };
172 services.world().addComponent<components::master_attr_t>(bullet_id, master);
173
174 std::vector<uint16_t> comp_ids = {
175 net::GameProtocol::AvailableComponents::POSITION,
176 net::GameProtocol::AvailableComponents::MASTER
177 };
178 network::EngineProtocol::sendEntityComponents(services, bullet_id, comp_ids);
179 }
180}
