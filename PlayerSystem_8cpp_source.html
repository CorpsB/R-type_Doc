31void PlayerSystem::move(Coordinator &coord, float ticks, protocol::game::player_input_t input)
32{
33 for (const uint32_t id : _filtered_entities) {
34 auto *pos = coord.getComponent<position_attr_t>(id);
35 auto *dir = coord.getComponent<direction_attr_t>(id);
36 auto *spd = coord.getComponent<speed_attr_t>(id);
37
38 if (!pos || !dir || !spd)
39 continue;
40
41 float spModifier = 1.0f;
42
43 switch (input) {
44 case protocol::game::UP:
45 dir->dx = 0;
46 dir->dy = -1;
47 break;
48 case protocol::game::DOWN:
49 dir->dx = 0;
50 dir->dy = 1;
51 break;
52 case protocol::game::LEFT:
53 dir->dx = -1;
54 dir->dy = 0;
55 break;
56 case protocol::game::RIGHT:
57 dir->dx = 1;
58 dir->dy = 0;
59 break;
60 case protocol::game::UP_LEFT:
61 dir->dx = -1;
62 dir->dy = -1;
63 spModifier = 0.71f;
64 break;
65 case protocol::game::UP_RIGHT:
66 dir->dx = 1;
67 dir->dy = -1;
68 spModifier = 0.71f;
69 break;
70 case protocol::game::DOWN_LEFT:
71 dir->dx = -1;
72 dir->dy = 1;
73 spModifier = 0.71f;
74 break;
75 case protocol::game::DOWN_RIGHT:
76 dir->dx = 1;
77 dir->dy = 1;
78 spModifier = 0.71f;
79 break;
80 default:
81 dir->dx = 0;
82 dir->dy = 0;
83 break;
84 }
85
86 const float sp = (static_cast<float>(spd->speed) / 100.0f) * spModifier;
87 const float dx = dir->dx * sp * ticks * 100.0f;
88 const float dy = dir->dy * sp * ticks * 100.0f;
89
90 pos->x += static_cast<int>(std::round(dx));
91 pos->y += static_cast<int>(std::round(dy));
92 }
93}

95void PlayerSystem::updateScore(Coordinator &coord, int scoreToAdd)
96{
97 for (const uint32_t id : _filtered_entities) {
98 auto *scr = coord.getComponent<score_attr_t>(id);
99
100 if (!scr)
101 continue;
102
103 if (static_cast<long int>(scr->score) + scoreToAdd <= 0) {
104 scr->score = 0;
105 continue;
106 }
107 scr->score += scoreToAdd;
108 if (scr->score >= scr->max_score)
109 scr->max_score = scr->score;
110 }
111}
