22rtype::ecs::EntityManager::EntityManager()
23{
24 _free_ids.reserve(MAX_ENTITIES);
25 _entities.resize(MAX_ENTITIES);
26 for (int i = 0; i < MAX_ENTITIES; i++) {
27 _free_ids.push_back(MAX_ENTITIES - 1 - i);
28 _entities[i].generation = 0;
29 _entities[i].active = false;
30 _entities[i].signature.reset();
31 }
32}

34uint32_t rtype::ecs::EntityManager::createEntity()
35{
36 uint32_t id;
37
38 if (!_free_ids.empty()) {

40 _free_ids.pop_back();

50bool rtype::ecs::EntityManager::destroyEntity(uint32_t id)
51{
52 if (id >= _entities.size())
53 return false;
54
55 EntityData &data = _entities[id];
