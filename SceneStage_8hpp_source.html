10/**
11 * @file SceneStage.hpp
12 * @brief Defines the abstract base class for application-level scenes.
13 * @ingroup scene
14 *
15 * @details
16 * `SceneStage` represents a single active or backgrounded scene (menu, level, etc.)
17 * managed by the `SceneManager`. Unlike engine systems, scenes define both their
18 * lifecycle (init, enter, update, draw…) and their policies for background behavior.
19 *
20 * Typical lifecycle:
21 * ```
22 * init() → onEnter() → [update()/draw()/handleEvent()] → onExit()
23 * ```
24 *
25 * @note All callbacks are executed on the main/game thread.
26 * @note SceneManager is responsible for calling these functions in the correct order.
27 */

42/**
43 * @class SceneStage
44 * @brief Abstract interface implemented by user-defined scenes.
45 *
46 * @details
47 * This interface defines the full lifecycle and behavior of a game scene.
48 * SceneStage is *not* an `IScene`; rather, it’s a higher-level construct
49 * managed by the `SceneManager`.
50 *
51 * - Each stage can represent a level, a menu, or a state of the application.
52 * - It defines how it reacts to lifecycle events and per-frame updates.
53 * - Background behavior is customizable through `wants*WhenBackground()` flags.
54 */
55class SceneStage {
56 public:
57 /// @brief Virtual destructor for safe polymorphic destruction.
58 virtual ~SceneStage() = default;
59
60 // ──────────────── Lifecycle ────────────────────────────────────────────────
61
62 virtual void init(EngineCore &core) = 0;
63
64 /// @brief Called when the scene becomes the active one (entered top of stack).
65 virtual void onEnter() = 0;
66
67 /// @brief Called when the scene is leaving the active stack (about to be destroyed or paused).
68 virtual void onExit() = 0;
69
70 /// @brief Called when the scene is paused (another scene pushed on top).
71 virtual void pause() = 0;
72
73 /// @brief Called when the scene resumes after being paused.
74 virtual void resume() = 0;
75
76 /**
77 * @brief Update the scene logic.
78 * @param dt Delta time (in seconds) since last frame.
79 */
80 virtual void update(Services &services, double dt) = 0;
81
82 /// @brief Render the scene contents to the screen.
83 virtual void draw(Services &Services) const = 0;
84
85 /**
86 * @brief Handle an input or system event.
87 *
88 * @param type Event type identifier (e.g., "Input/KeyDown", "UI/Click").
89 * @param payload Optional serialized event payload.
90 * @return `true` if the event is consumed and should not propagate further.
91 */
92 virtual bool handleEvent(const std::string &type, const std::string &payload) = 0;
93
94 /**
95 * @brief Whether this scene continues to draw while in background.
96 * @return false by default.
97 */
98 virtual bool wantsDrawWhenBackground() const { return false; }
99
100 /**
101 * @brief Whether this scene continues to update while in background.
102 * @return false by default.
103 */
104 virtual bool wantsUpdateWhenBackground() const { return false; }
105
106 /**
107 * @brief Whether this scene should still receive input events when not active.
108 *
109 * @details
110 * If true, the SceneManager may call `handleEvent()` even while the scene
111 * is covered by others — provided that upper scenes did not consume the event.
112 *
113 * @return false by default.
114 */
115 virtual bool wantsInputWhenBackground() const { return false; }
116};
