8/**
9 * @file UDPChannel.hpp
10 * @brief Non-blocking UDP implementation of NetworkChannel.
11 * @since 0.1
12 * @ingroup network
13 *
14 * @details
15 * Implements a datagram-oriented channel built on Boost.Asio’s UDP sockets.
16 * This class provides non-blocking send/receive behavior and optional debug
17 * tracing. It forms the backbone of the runtime networking stack.
18 *
19 * Key features:
20 * - Fully non-blocking I/O.
21 * - Supports both client and server modes.
22 * - Configurable MTU (maximum transmission unit).
23 * - Thread-safe via internal mutex.
24 * - Optional debug output for packet events.
25 */

36/**
37 * @class UDPChannel
38 * @brief Concrete NetworkChannel based on UDP sockets using Boost.Asio.
39 * @ingroup network
40 *
41 * @details
42 * Provides low-level send/receive functions with optional batching and socket
43 * configuration. Designed for real-time game networking where minimal latency
44 * and fine-grained control are required.
45 *
46 * Usage:
47 * - @ref openServer(port) to bind for incoming packets.
48 * - @ref openClient(address, port) to prepare for outbound traffic.
49 * - @ref poll(callback) to process pending datagrams.
50 * - @ref send(bytes) or @ref sendTo(bytes, addr, port) for data transmission.
51 *
52 * @note Thread safety: internal mutex `_mx` guards socket state.
53 * @warning UDP is connectionless; packet delivery is not guaranteed.
54 */
55class UDPChannel : public ANetworkChannel {
56public:
57 /**
58 * @brief Construct a new UDPChannel.
59 * @param debug Enable verbose debug output for send/receive operations.
60 */
61 explicit UDPChannel(bool debug = false);
62
63 /**
64 * @brief Destructor — closes socket and releases context.

66 ~UDPChannel() override;

68 /**
69 * @brief Bind the socket as a UDP server.
70 * @param port Port number to listen on.
71 * @return True if binding succeeded.
72 * @throw asge::error::Error On invalid port or socket binding failure.
73 *
74 * @details
75 * Opens and binds the socket in non-blocking mode, ready to receive datagrams
76 * from any address.
77 */
78 bool openServer(std::uint16_t port) override;
79
80 /**
81 * @brief Open as a UDP client.
82 * @param address Remote address (IPv4/IPv6).
83 * @param port Remote port number.
84 * @return True if the socket was created successfully.
85 * @throw asge::error::Error On invalid endpoint parameters.
86 *
87 * @details
88 * Prepares a socket for sending datagrams. The provided address and port are

91 bool openClient(const std::string& address, std::uint16_t port) override;
92
93 /**

95 * @param nb True to enable non-blocking mode, false for blocking.
96 * @return True on success.

101 bool setNonBlocking(bool nb) override;
102
103 /**

105 * @param mtu Maximum transmission unit in bytes.
106 * @throw asge::error::Error If the value is too small to fit headers.

108 * @details
109 * Controls the maximum packet size allowed for transmission.
110 */
111 void setMtu(std::size_t mtu) override;
112

117 std::size_t mtu() const override;
118
119 /**

121 * @param bytes Payload to transmit.
122 * @return True if queued successfully.

126
127 /**
128 * @brief Send a datagram to a specific address and port.
129 * @param bytes Data buffer to send.
130 * @param address Destination IPv4/IPv6 address.
131 * @param port Destination port.
132 * @return True if the datagram was accepted for transmission.
133 * @throw asge::error::Error On invalid address, port, or MTU overflow.
134 */
135 bool sendTo(const std::vector<std::uint8_t>& bytes, const std::string& address, std::uint16_t port) override;
136
137 /**
138 * @brief Poll the socket for incoming datagrams.
139 * @param onPacket Callback invoked for each received packet.
140 * @throw asge::error::Error On socket I/O failure.
141 *
142 * @details
143 * Reads all pending packets from the socket and invokes the callback with:
144 * - The received data buffer.
145 * - Source address string.
146 * - Source port.
147 *
148 * Non-blocking behavior ensures the call returns immediately if no data
149 * is available.
150 */
151 void poll(std::function<void(const std::vector<std::uint8_t>&, const std::string&, std::uint16_t)> onPacket) override;
152
153 /**
154 * @brief Close the socket and release any resources.
155 *
156 * @details
157 * Shuts down the UDP socket and clears the default remote endpoint.
158 * Safe to call multiple times.
159 */
160 void close() override;
