18/**
19 * @file PluginHandle.hpp
20 * @brief Lightweight handle representing a loaded plugin instance.
21 * @since 0.1
22 * @ingroup plugin
23 * @see IPlugin, PluginLoader, PluginManager
24 *
25 * @details
26 * The PluginHandle class acts as a managed reference to a single loaded plugin.
27 * It encapsulates an identifier, the plugin instance, its source path, and
28 * loading state. This abstraction allows the @ref PluginManager to track and
29 * control all loaded plugins safely.
30 *
31 * A handle does not perform dynamic loading itself—it simply provides access
32 * to a plugin that was already created by the @ref PluginLoader.
33 *
34 * @note The handle ensures resource cleanup when @ref unload() is called.
35 * @warning Do not copy PluginHandle instances; they are intended to be managed
36 * by shared pointers inside the PluginManager.
37 */
38

41/**
42 * @class PluginHandle
43 * @brief Represents a managed instance of a loaded plugin.
44 *
45 * @details
46 * Each PluginHandle is associated with a unique identifier (plugin ID),
47 * the path of the loaded shared library, and a shared pointer to the
48 * @ref IPlugin implementation. It exposes basic accessors and an @ref unload()
49 * method to safely release the plugin’s resources.
50 */
51class PluginHandle {
52 public:
53 /**
54 * @brief Constructs a PluginHandle for a given plugin instance.
55 * @param id Unique identifier of the plugin.
56 * @param instance Shared pointer to the loaded plugin implementation.
57 * @param path Filesystem path from which the plugin was loaded.
58 *
59 * @details
60 * The handle is considered "loaded" upon creation. The plugin can later
61 * be released via @ref unload().
62 */
63 explicit PluginHandle(std::string id, std::shared_ptr<IPlugin> instance, const std::string &path)
64 : _plugin_identifier(std::move(id)),
65 _plugin_instance(std::move(instance)),
66 _plugin_path(path),
67 _is_loaded(true)
68 {
69 }
70
71 /**
72 * @brief Retrieves the plugin identifier.
73 * @return Constant reference to the plugin’s ID string.
74 */
75 const std::string &id() const
76 {
77 return _plugin_identifier;
78 }
79
80 /**
81 * @brief Retrieves the path from which the plugin was loaded.
82 * @return Constant reference to the plugin’s filesystem path.
83 */
84 const std::string &path() const
85 {
86 return _plugin_path;
87 }
88
89 /**
90 * @brief Checks if the plugin is currently loaded.
91 * @return True if the plugin is marked as loaded, false otherwise.
92 */
93 bool isLoaded() const
94 {
95 return _is_loaded;
96 }
97
98 /**
99 * @brief Provides access to the plugin instance.
100 * @return Shared pointer to the underlying @ref IPlugin.
101 */
102 std::shared_ptr<IPlugin> instance() const
103 {
104 return _plugin_instance;
105 }
106
107 /**
108 * @brief Unloads the plugin safely.
109 * @details
110 * Invokes the plugin’s @ref IPlugin::unloadAll() method and updates
111 * the internal loaded state.
112 * @throw error::Error If the plugin instance is invalid or unloading fails.
113 */
114 void unload();
115
116 private:
117 std::string _plugin_identifier; /**< Unique identifier of the plugin. */
118 std::shared_ptr<IPlugin> _plugin_instance; /**< Shared pointer to the plugin instance. */
119 std::string _plugin_path; /**< Filesystem path of the plugin. */
120 bool _is_loaded; /**< Indicates whether the plugin is loaded. */
121};
