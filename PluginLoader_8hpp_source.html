23/**
24 * @file PluginLoader.hpp
25 * @brief Platform-aware dynamic loader that instantiates IPlugin implementations.
26 * @since 0.1
27 * @ingroup plugin
28 * @see IPlugin, PluginHandle, PluginManager
29 *
30 * @details
31 * The PluginLoader is responsible for loading shared libraries (DLL on Windows,
32 * .so on Linux, .dylib on macOS), resolving the standard factory symbol
33 * (commonly named <tt>CreatePlugin</tt>), and constructing a concrete
34 * @ref IPlugin instance.
35 *
36 * Successfully opened native library handles are tracked internally so they can
37 * be released later via @ref unloadAll(). The loader does not manage logical
38 * plugin metadata (IDs, dependencies, versions); that orchestration is handled
39 * by @ref PluginManager.
40 *
41 * @note This header declares the interface and data members only. Platform-
42 * specific details are implemented in the corresponding source file.
43 * @warning The caller is responsible for holding the returned @c std::shared_ptr<IPlugin>.
44 * Releasing it does not automatically close the native library; call
45 * @ref unloadAll() to free all native handles owned by the loader.
46 */
47

50/**
51 * @class PluginLoader
52 * @brief Loads shared libraries and creates @ref IPlugin instances.
53 *
54 * @details
55 * Typical usage:
56 * - Call @ref load() with a filesystem path to a shared library.
57 * - Use the returned plugin instance in higher-level systems.
58 * - On shutdown, call @ref unloadAll() to release all native handles.
59 */
60class PluginLoader {
61 public:
62 /**

65 * @note The debugger pointer is stored (not owned) and is expected to
66 * outlive the loader.
67 */

75 ~PluginLoader() = default;

77 /**
78 * @brief Loads a shared library and instantiates an @ref IPlugin.
79 * @param path Filesystem path to the plugin library.
80 * @return A @c std::shared_ptr<IPlugin> constructed from the library’s
81 * factory entry point (e.g., @c CreatePlugin).
82 *
83 * @details
84 * This function:
85 * 1. Opens the native library (platform-specific).
86 * 2. Resolves the factory symbol (e.g., @c CreatePlugin).
87 * 3. Constructs and returns the plugin instance.
88 * 4. Stores the native handle internally for later release.
89 *
90 * @throw error::Error If the path is empty, the library cannot be opened,
91 * the factory symbol cannot be resolved, or the factory returns
92 * a null instance.
93 */
94 virtual std::shared_ptr<IPlugin> load(const std::string &path);
95
96 /**
97 * @brief Releases all native library handles previously opened by @ref load().
98 * @details Safe to call multiple times; subsequent calls after the first
99 * will have no effect.
100 */
101 virtual void unloadAll();
102
103 protected:
104 /**
105 * @brief Optional debug sink used for logging loader activity.
106 * @note Non-owning pointer; must remain valid during the loader’s lifetime.
107 */
108 debug::IDebug *_debugger;
109
110 private:
111 /**
112 * @brief Collection of platform-native library handles to be closed on shutdown.
113 * @details Each successful @ref load() pushes one handle to this list.
114 */
