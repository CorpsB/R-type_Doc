10/**
11 * @file ErrorTypes.hpp
12 * @brief Enumerations describing error categories and severities.
13 *
14 * @details
15 * ErrorType enumerates logical subsystems that may produce errors. ErrorSeverity
16 * classifies how serious an error is and indicates potential runtime response.
17 * These types are intentionally coarse-grained to keep error handling simple and
18 * portable across engine subsystems.
19 */
20
21namespace asge::error {
22
23/**
24 * @enum ErrorType
25 * @brief Broad categories of errors produced by the engine.
26 */
27enum class ErrorType {
28 Generic, ///< Unknown or uncategorized error.
29 Audio, ///< Errors originating from the audio subsystem.
30 Debug, ///< Diagnostic/logging related errors.
31 Event, ///< Event system related errors.
32 Input, ///< Input handling failures.
33 Job, ///< Job system / task scheduling issues.
34 Network, ///< Networking problems.
35 Plugin, ///< Plugin subsystem errors.
36 Renderer, ///< Rendering and GPU related errors.
37 Script, ///< Scripting / template subsystem failures.
38 Storage, ///< File I/O / asset loading errors.
39 Timer, ///< Timing / scheduling issues.
40 World, ///< Game world and simulation errors.
41 Unknown ///< Explicit unknown category, use when category cannot be determined.
42};
43
44/**
45 * @enum ErrorSeverity
46 * @brief Severity levels that indicate the impact of an error.
47 */
48enum class ErrorSeverity {
49 Info, ///< Informational; no action required.
50 Warning, ///< Non-fatal but notable; should be logged.
51 Recoverable, ///< Recoverable with retry or fallback behaviour.
52 Fatal ///< Requires supervisor action (shutdown/restart).
53};

55}
