66/**
67 * @brief Program entry point.
68 * @details
69 * Steps:
70 * 1. Parse flags once to capture an optional -path for the config file.
71 * 2. Load configuration key/values from the file if available.
72 * 3. Re-apply flags so that CLI values override file values.
73 * 4. Normalize and validate the final configuration.
74 * 5. If requested, print help and exit; otherwise construct and run the server.
75 * 6. Report any runtime error and exit with a failure code.
76 *
77 * @param ac Argument count.
78 * @param av Argument vector.
79 * @return 0 on success; 84 on invalid configuration or runtime error.
80 * @see parser_flag, parser_cfg, check_config, print_help
81 */
82int main(int ac, char** av)
83{
84 parser_args_t args;
85 parser_flag(args, ac, av);
86 parser_cfg(args, ac, av);
87 parser_flag(args, ac, av);
88 check_config(args, ac, av);
89
90 std::string why;
91 if (!check_config(args, why)) {
92 std::cerr << why << std::endl <<"[INFO] - You can use -h to see help message." <<std::endl;
93 return 84;
94 }
95
96 if (args.need_help) {
97 return print_help();
98 }
99 try {
100 rtype::server server(av);
101
102 server.run();
103 return 0;
104 } catch (const rtype::exception::Error &e) {
105 std::cerr << "[ERROR] - " << e.getWhere() << std::endl << e.getWhat() << std::endl;
106 return 84;
107 }
108}
