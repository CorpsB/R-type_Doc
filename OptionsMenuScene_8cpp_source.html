33OptionsMenuScene::OptionsMenuScene()
34{
35 auto &settings = GameSettings::getInstance();
36
37 _menus.reserve(3);
38
39 _menus.emplace_back(0, 0, _menuWidth, _menuHeight, "004_OPT_RESOLUTION");
40 _menus.back()
41 .addOption("004_OPT_RESOLUTION_720", [&]() { settings.setScreenSize({ 1280, 720 }); })
42 .addOption("004_OPT_RESOLUTION_900", [&]() { settings.setScreenSize({ 1600, 900 }); })
43 .addOption("004_OPT_RESOLUTION_1080", [&]() { settings.setScreenSize({ 1920, 1080 }); });
44
45 _menus.emplace_back(0, 0, _menuWidth, _menuHeight, "004_OPT_DISPLAY");
46 _menus.back()
47 .addOption("004_OPT_DISPLAY_W", []() { if (IsWindowFullscreen()) ToggleFullscreen(); })
48 .addOption("004_OPT_DISPLAY_F", []() { if (!IsWindowFullscreen()) ToggleFullscreen(); });
49
50 _menus.emplace_back(0, 0, _menuWidth, _menuHeight, "004_OPT_LANG");
51 _menus.back()
52 .addOption("004_OPT_LANG_FR", [&]() { settings.setLangTag("FR"); })
53 .addOption("004_OPT_LANG_EN", [&]() { settings.setLangTag("EN"); })
54 .addOption("004_OPT_LANG_ZU", [&]() { settings.setLangTag("ZU"); });
55}

57void OptionsMenuScene::update(SceneManager &manager, float dt)
58{
59 auto &settings = GameSettings::getInstance();
60
61 settings.syncWithWindow();
62
63 float screenWidth = settings.baseScreen.x;
64 float screenHeight = settings.baseScreen.y;
65
66 float totalHeight = _menus.size() * _menuHeight + (_menus.size() - 1) * _spacing;
67 float startY = (screenHeight - totalHeight) / 2.0f;
68 float centerX = (screenWidth - _menuWidth) / 2.0f;
69
70 for (int i = 0; i < _menus.size(); i++) {
71 float y = startY + i * (_menuHeight + _spacing);
72
73 _menus[i].getMainButton().setPosition(centerX, y);
74 _menus[i].update();
75 }
76}
