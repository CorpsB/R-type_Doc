14 * @details
15 * The `NetworkChannel` class defines the fundamental transport-level interface used by
16 * the networking layer of the R-Type engine.
17 * It represents an abstraction over different transport mechanisms (e.g. UDP, loopback, etc.)
18 * and exposes a minimal API for sending, receiving, and managing packets.
19 *
20 * Each implementation must ensure proper thread-safety and respect non-blocking semantics.
21 * This interface is extended by:
22 * - @ref UDPChannel
23 * - @ref LoopbackChannel
24 * - Any other transport-specific implementation.
25 */

36/**
37 * @class NetworkChannel
38 * @brief Abstract base class for low-level communication channels.
39 * @ingroup network
40 *
41 * @details
42 * Provides the minimal set of operations to manage a bidirectional packet-based transport channel.
43 * Implementations can differ in protocol and behavior, but all must adhere to the same lifecycle:
44 *
45 * 1. Open (server/client)
46 * 2. Configure (non-blocking, MTU)
47 * 3. Send / Poll loop
48 * 4. Close
49 *
50 * Example usage:
51 * @code
52 * std::unique_ptr<NetworkChannel> chan = std::make_unique<UDPChannel>();
53 * chan->openClient("127.0.0.1", 4200);
54 * chan->send({0x10, 0x20, 0x30});
55 * chan->poll([](auto data, auto addr, auto port) { handle packet });
56 * chan->close();
57 * @endcode
58 */
59class NetworkChannel {
60public:
61 /** @brief Virtual destructor for polymorphic cleanup. */
62 virtual ~NetworkChannel() = default;
63
64 /**
65 * @brief Open a channel as a server.
66 * @param port Port number to listen on.
67 * @return True if successfully opened.
68 * @throw asge::error::Error On invalid or unavailable port.
69 */
70 virtual bool openServer(std::uint16_t port) = 0;
71
72 /**
73 * @brief Open a channel as a client connecting to a remote endpoint.
74 * @param address Remote IP address or hostname.
75 * @param port Remote port number.
76 * @return True if the client channel was established successfully.
77 * @throw asge::error::Error On invalid parameters or connection failure.
78 */
79 virtual bool openClient(const std::string& address, std::uint16_t port) = 0;
80
81 /**
82 * @brief Enable or disable non-blocking I/O.
83 * @param nb True for non-blocking mode, false otherwise.
84 * @return True if the mode was successfully applied.
85 */
86 virtual bool setNonBlocking(bool non_blocking) = 0;
87
88 /**
89 * @brief Set the maximum transmission unit (MTU).
90 * @param mtu Maximum payload size in bytes.
91 * @throw asge::error::Error If the value is outside implementation limits.
92 */
93 virtual void setMtu(std::size_t mtu) = 0;
94
95 /**
96 * @brief Get the current MTU value.
97 * @return Configured MTU size in bytes.
98 */
99 virtual std::size_t mtu() const = 0;
100
101 /**
102 * @brief Send a data buffer to the default or connected endpoint.
103 * @param bytes Data to send.
104 * @return True if the send operation succeeded.
105 * @throw asge::error::Error If MTU exceeded or channel closed.
106 */
107 virtual bool send(const std::vector<std::uint8_t>& bytes) = 0;
108
109 /**
110 * @brief Send a data buffer to a specific destination.
111 * @param bytes Data payload.
112 * @param address Destination address.
113 * @param port Destination port.
114 * @return True if the packet was sent successfully.
115 * @throw asge::error::Error On invalid address or socket error.
116 */
117 virtual bool sendTo(const std::vector<std::uint8_t>& bytes, const std::string& address, std::uint16_t port) = 0;
118
119 /**
120 * @brief Poll for incoming packets.
121 * @param onPacket Callback executed for each received packet.
122 *
123 * @details
124 * The callback receives the packet data, sender address, and sender port.
125 * The method should be non-blocking in compliant implementations.
126 */
127 virtual void poll(std::function<void(const std::vector<std::uint8_t>&, const std::string&, std::uint16_t)> onPacket) = 0;
128
129 /**
130 * @brief Close the network channel and release resources.
131 *
132 * @details
133 * Safe to call multiple times.
134 * After closing, all send/poll operations must be considered invalid.
135 */
136 virtual void close() = 0;
137};
