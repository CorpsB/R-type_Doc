25void asge::world::systems::MovementSystem::update(Services &services, float tick_count)
26{
27 world::World &game_world = services.world();
28
29 for (const uint32_t entity_id : _filtered_entities) {
30 components::position_attr_t *position_component =
31 game_world.getComponent<components::position_attr_t>(entity_id);
32 components::direction_attr_t *direction_component =
33 game_world.getComponent<components::direction_attr_t>(entity_id);
34 components::speed_attr_t *speed_component =
35 game_world.getComponent<components::speed_attr_t>(entity_id);
36 components::name_attr_t *name_component =
37 game_world.getComponent<components::name_attr_t>(entity_id);
38
39 if (position_component == nullptr || direction_component == nullptr ||
40 speed_component == nullptr || name_component == nullptr) {
41 continue;
42 }
43
44 const float speed_scaling = static_cast<float>(speed_component->speed) / 100.0f;
45 const float offset_x = direction_component->dx * speed_scaling * tick_count * 100.0f;
46 const float offset_y = direction_component->dy * speed_scaling * tick_count * 100.0f;
47
48 position_component->x += static_cast<int>(std::round(offset_x));
49 position_component->y += static_cast<int>(std::round(offset_y));
50
51 std::vector<uint16_t> comp_ids { net::GameProtocol::AvailableComponents::POSITION };
52 network::EngineProtocol::sendEntityComponents(services, entity_id, comp_ids);
53
54 try {
55 services.scripts().notifyEntityPosition(entity_id, name_component->name,
56 static_cast<double>(position_component->x), static_cast<double>(position_component->y));
57 }
58 catch (...) {
59 if (services._debug) {
60 services.debug().log(
61 debug::Level::Warning,
62 "Fail to exec position hook on entity " + std::to_string(entity_id));
63 }
64 }
65 }
66}
