24void Text::draw(asge::Services &services) const
25{
26 asge::renderer::IRenderer &renderer = services.renderer();
27
28 switch (_alignment) {
29 case Alignment::LEFT: {
30 asge::renderer::Vec2f textSize = renderer.measureText(_label, _font_size);
31 renderer.drawText(_label, _pos.x, _pos.y - textSize.y / 2.f, _font_size, _color);
32 } break;
33 case Alignment::RIGHT: {
34 asge::renderer::Vec2f textSize = renderer.measureText(_label, _font_size);
35 renderer.drawText(_label, _pos.x - textSize.x, _pos.y - textSize.y / 2.f, _font_size, _color);
36 } break;
37 case Alignment::CENTER: {
38 asge::renderer::Vec2f textSize = renderer.measureText(_label, _font_size);
39 renderer.drawText(_label, _pos.x - textSize.x / 2.f, _pos.y - textSize.y / 2.f, _font_size, _color);
40 } break;
41 }
42}
