15 * @details
16 * Implements the logic behind global scaling, resolution management, and configuration
17 * synchronization for the R-Type client. The GameSettings class ensures visual consistency
18 * and proportional scaling across all rendering contexts.
19 *
20 * ### Key Responsibilities:
21 * - Maintain proportional scaling according to a 16:9 reference.
22 * - Handle fullscreen/windowed transitions.
23 * - Manage and apply temporary language and display settings.
24 * - Cache render width and height to avoid redundant scaling computations.
25 *
26 * @note Scaling updates are only recalculated when the window size changes.
27 * @warning All Raylib interactions must occur on the main rendering thread.
28 */

40void GameSettings::setScreenSize(Vector2 size)
41{
42 float ratio = size.x / size.y;
43 const float targetRatio = 16.0f / 9.0f;
44
45 if (std::fabs(ratio - targetRatio) > 0.01f)
46 size.y = size.x / targetRatio;
47 SetWindowSize(static_cast<int>(size.x), static_cast<int>(size.y));
48 syncWithWindow();
49}

51void GameSettings::applyTMP()
52{
53 if (_fullscreenTMP == "F" && !IsWindowFullscreen())

55 if (_fullscreenTMP == "W" && IsWindowFullscreen())
56 ToggleFullscreen();

64void GameSettings::syncWithWindow()
65{
66 const int rw = GetRenderWidth();
67 const int rh = GetRenderHeight();
68
69 if (rw == _renderW && rh == _renderH)
70 return;
71 _renderW = rw;
72 _renderH = rh;
73
74 const float sx = static_cast<float>(rw) / baseScreen.x;
75 const float sy = static_cast<float>(rh) / baseScreen.y;
76
77 _scale = (sx < sy) ? sx : sy;
78 _viewportOffset.x = (rw - baseScreen.x * _scale) * 0.5f;
79 _viewportOffset.y = (rh - baseScreen.y * _scale) * 0.5f;
80}
