35void ResourceManager::loadTexture(const std::string &key, const std::string &path)
36{
37 if (_textures.find(key) != _textures.end())
38 return;
39 Texture2D tex = LoadTexture(path.c_str());
40
41 if (tex.id == 0)
42 throw rtype::exception::Error("ResourceManager", "Failed to load texture: " + path);
43 _textures.emplace(key, tex);
44}

55void ResourceManager::loadFont(const std::string &key, const std::string &path, int size)
56{
57 if (_fonts.find(key) != _fonts.end())
58 return;
59 std::vector<int> codepoints;
60
61 for (int i = 0x0020; i <= 0x024F; i++) codepoints.push_back(i);
62 for (int i = 0x0400; i <= 0x04FF; i++) codepoints.push_back(i);
63 for (int i = 0x0370; i <= 0x03FF; i++) codepoints.push_back(i);
64 for (int i = 0x3040; i <= 0x309F; i++) codepoints.push_back(i);
65 for (int i = 0x30A0; i <= 0x30FF; i++) codepoints.push_back(i);
66 for (int i = 0xAC00; i <= 0xD7AF; i++) codepoints.push_back(i);
67 for (int i = 0x4E00; i <= 0x9FFF; i += 5) codepoints.push_back(i);
68 Font f = LoadFontEx(path.c_str(), size, codepoints.data(), codepoints.size());
69
70 if (f.baseSize == 0)
71 throw rtype::exception::Error("ResourceManager", "Failed to load font: " + path);
72 _fonts.emplace(key, f);
73}

104void ResourceManager::unloadAll()
105{
106 for (auto &kv : _textures)
107 UnloadTexture(kv.second);
108 for (auto &kv : _fonts)
109 UnloadFont(kv.second);
110 for (auto &kv : _sounds)
111 UnloadSound(kv.second);
112 _textures.clear();
113 _fonts.clear();
114 _sounds.clear();
