24/**
25 * @brief Serialize a CREATE_ENTITY payload into a raw byte buffer.
26 *
27 * Wire layout (bytes appended in order):
28 * - entity_id: 4 bytes (u32, network order)
29 * - entity_type: 1 byte (u8)
30 * - x: 4 bytes (position_t stored as 32-bit signed integer; transmitted as u32)
31 * - y: 4 bytes (position_t stored as 32-bit signed integer; transmitted as u32)
32 *
33 * @param p Strongly-typed payload to serialize.
34 * @return A protocol::byte_buffer_t (vector<uint8_t>) containing the serialized payload.
35 */
36protocol::byte_buffer_t serialize_create_entity(
37 const protocol::game::payload_create_entity_t &p)
38{
39 std::vector<uint8_t> buff;
40
41 protocol::utils::append_u32(buff, static_cast<uint32_t>(p.entity_id));
42 protocol::utils::append_u8(buff, p.entity_type);
43 protocol::utils::append_u32(buff, static_cast<uint32_t>(p.x));
44 protocol::utils::append_u32(buff, static_cast<uint32_t>(p.y));
45 return buff;
46}

48/**
49 * @brief Attempt to deserialize a CREATE_ENTITY payload from a byte buffer.
50 *
51 * @param buff Input buffer containing payload bytes.
52 * @param out Output reference filled on success with a payload_create_entity_t.
53 * @return true if deserialization succeeded and buffer had at least the required bytes (13 bytes),
54 * false otherwise.
55 *
56 * @details
57 * Validates buffer length before attempting reads. Field extraction order matches
58 * serialize_create_entity. The function performs integer conversion via utils::read_u32/read_u8.
59 */
60bool try_deserialize_create_entity(const protocol::byte_buffer_t &buff,
61 protocol::game::payload_create_entity_t &out)
62{
63 if (buff.size() < 13u)
64 return false;
65
66 std::size_t offset = 0;
67 out.entity_id = static_cast<protocol::game::entity_id_t>(protocol::utils::read_u32(buff, offset));
68 offset += 4;
69 out.entity_type = static_cast<uint8_t>(protocol::utils::read_u8(buff, offset));
70 offset += 1;
71 out.x = static_cast<protocol::game::position_t>(protocol::utils::read_u32(buff, offset));
72 offset += 4;
73 out.y = static_cast<protocol::game::position_t>(protocol::utils::read_u32(buff, offset));
74 offset += 4;
75 return true;
76}

205std::error_code protocol::game::GameProtocol::initialize(const std::string &local_address,
206 uint16_t local_port, const std::optional<std::string> &server_address,
207 const std::optional<uint16_t> &server_port) noexcept
208{
209 Protocol &p = Protocol::instance();
210 std::error_code err = p.init(local_address, local_port);
211
212 if (err)
213 return err;
214
215 if (server_address.has_value() && server_port.has_value()) {
216 boost::system::error_code boost_err;
217 boost::asio::ip::address addr = boost::asio::ip::make_address(
218 server_address.value(), boost_err);
219
220 if (!boost_err) {
221 p.storeEndpoint(udp::endpoint(addr, server_port.value()));
222 } else {
223 p.clearEndpoint();
224 }
225 } else {
226 p.clearEndpoint();
227 }
228
229 return {};
230}

284std::error_code protocol::game::GameProtocol::send_raw(uint16_t message_type,
285 const byte_buffer_t &payload, flag_t reliability,
286 const std::optional<udp::endpoint> &remote) noexcept
287{
288 Protocol &p = Protocol::instance();
289 auto endpoints = p.getStoredEndpoints();
290
291 if (remote.has_value())
292 return p.send_packet(remote.value(), message_type, reliability, payload);
293
294 std::error_code last_err{};
295 for (const auto &ep : endpoints) {
296 std::error_code err_code = p.send_packet(ep, message_type, reliability, payload);
297 if (err_code)
298 last_err = err_code;
299 }
300 return last_err;
301}

313/**
314 * @brief Convert a core packet event (protocol::packet_t inside core_event_t) into a game-level event.
315 *
316 * @param out_event Output game_event_t filled with the converted event.
317 * @param core Core event carrying a protocol::packet_t payload (PACKET_RECEIVED expected).
318 * @return true always (conversion attempted). On malformed or unknown message_type, the function
319 * will set out_event.type to RAW_PACKET and place the original packet into the payload.
320 *
321 * @details
322 * - Examines the packet.header.message_type and tries to interpret payload bytes as a known
323 * game message (CREATE_ENTITY, MOVE_ENTITY, DESTROY_ENTITY, PLAYER_INPUT).
324 * - For recognized types, it calls the corresponding try_deserialize_* helper. If deserialization
325 * succeeds, the function sets out_event.type to the corresponding game_event_type_t and fills
326 * out_event.payload with the typed payload.
327 * - If deserialization fails or the message_type is unknown, out_event.type is set to RAW_PACKET
328 * and the original protocol::packet_t is moved into the payload for the consumer to examine.
329 *
330 * @note The function does NOT perform protocol-level validation beyond buffer-size checks done by
331 * the try_deserialize_* helpers. The caller should ensure core.payload holds a packet_t.
332 */
333bool retrieve_package(protocol::game::game_event_t &out_event,
334 protocol::core_event_t &core)
335{
336 if (!std::holds_alternative<protocol::packet_t>(core.payload)) {
337 out_event.type = protocol::game::game_event_type_t::RAW_PACKET;
338 out_event.payload = protocol::packet_t{};
339 return true;
340 }
341
342 protocol::packet_t packet = std::get<protocol::packet_t>(core.payload);
343 uint16_t message_type = static_cast<uint16_t>(packet.header.message_type);
344
345 switch (static_cast<protocol::game::game_message_type_t>(message_type))
346 {
347 case protocol::game::game_message_type_t::CREATE_ENTITY: {
348 protocol::game::payload_create_entity_t pe;
349
350 if (try_deserialize_create_entity(packet.payload, pe)) {
351 out_event.type = protocol::game::game_event_type_t::CREATE_ENTITY;
352 out_event.payload = pe;
353 } else {
354 out_event.type = protocol::game::game_event_type_t::RAW_PACKET;
355 out_event.payload = std::move(packet);
356 }
357 break;
358 }
359
360 case protocol::game::game_message_type_t::MOVE_ENTITY: {
361 protocol::game::payload_move_entity_t pm;
362
363 if (try_deserialize_move_entity(packet.payload, pm)) {
364 out_event.type = protocol::game::game_event_type_t::MOVE_ENTITY;
365 out_event.payload = pm;
366 } else {
367 out_event.type = protocol::game::game_event_type_t::RAW_PACKET;
368 out_event.payload = std::move(packet);
369 }
370 break;
371 }
372
373 case protocol::game::game_message_type_t::DESTROY_ENTITY: {
374 protocol::game::payload_destroy_entity_t pd;
375
376 if (try_deserialize_destroy_entity(packet.payload, pd)) {
377 out_event.type = protocol::game::game_event_type_t::DESTROY_ENTITY;
378 out_event.payload = pd;
379 } else {
380 out_event.type = protocol::game::game_event_type_t::RAW_PACKET;
381 out_event.payload = std::move(packet);
382 }
383 break;
384 }
385
386 case protocol::game::game_message_type_t::PLAYER_INPUT: {
387 protocol::game::payload_player_input_t pi;
388
389 if (try_deserialize_player_input(packet.payload, pi)) {
390 out_event.type = protocol::game::game_event_type_t::PLAYER_INPUT;
391 out_event.payload = pi;
392 } else {
393 out_event.type = protocol::game::game_event_type_t::RAW_PACKET;
394 out_event.payload = std::move(packet);
395 }
396 break;
397 }
398
399 default:
400 out_event.type = protocol::game::game_event_type_t::RAW_PACKET;
401 out_event.payload = std::move(packet);
402 break;
403 }
404 return true;
405}

407bool protocol::game::GameProtocol::get_event(game_event_t &out_event) noexcept
408{
409 core_event_t core;
410
411 if (!Protocol::instance().get_event(core))
412 return false;
413
414 out_event.type = game_event_type_t::NONE;
415 out_event.payload = std::monostate{};
416
417 switch (core.type) {
418 case event_type_t::PACKET_RECEIVED: {
419 return retrieve_package(out_event, core);
420 }
421
422 case event_type_t::TRANSMIT_FAILED: {
423 if (std::holds_alternative<transmit_failed_t>(core.payload)) {
424 transmit_failed_t failure = std::get<transmit_failed_t>(core.payload);
425
426 out_event.type = game_event_type_t::TRANSMIT_FAILED;
427 out_event.payload = failure;
428 } else {
429 out_event.type = game_event_type_t::TRANSMIT_FAILED;
430 out_event.payload = transmit_failed_t{};
431 }
432 break;
433 }
434
435 case event_type_t::CONNECTION_ESTABLISHED: {
436 udp::endpoint ep;
437
438 if (std::holds_alternative<packet_t>(core.payload)) {
439 ep = std::get<packet_t>(core.payload).remote;
440 }
441 out_event.type = game_event_type_t::CONNECTED;
442 out_event.payload = ep;
443 break;
444 }
445
446 case event_type_t::CONNECTION_LOST: {
447 udp::endpoint ep;
448
449 if (std::holds_alternative<packet_t>(core.payload)) {
450 ep = std::get<packet_t>(core.payload).remote;
451 }
452 out_event.type = game_event_type_t::DISCONNECTED;
453 out_event.payload = ep;
454 break;
455 }
456
457 default:
458 out_event.type = game_event_type_t::NONE;
459 out_event.payload = std::monostate{};
460 break;
461 }
462
463 return true;
464}
