40/**
41 * @class CollisionWorld
42 * @brief Implements a spatial-hash broad-phase and SAT-based narrow-phase collision engine.
43 *
44 * @details
45 * CollisionWorld maps world-space to discrete cells (cellKey from makeKey(cx,cy)).
46 * For each entity it:
47 * - computes oriented rectangle corners (fillBoxCorners),
48 * - converts corners to an AABB (cornersToAABB),
49 * - inserts the entity into all intersected cells,
50 * Then checkCollisions iterates cells and:
51 * - checks pairs within the same cell,
52 * - checks pairs across neighbor cells while avoiding duplicates using key ordering,
53 * - uses projection tests (projectOnAxis/isAxisSeparation) to determine overlap.
54 *
55 * @note
56 * - Positions are interpreted using the project's fixed-point / scaling convention.
57 * - The collisionEffect method is a hook for application-specific response handling.
58 */
59class CollisionWorld {
60 public:

64 /**
65 * @brief Run narrow-phase collision checks across the grid and call collisionEffect on hits.
66 *
67 * @param coord Coordinator to access positional and hitbox components.
68 *
69 * @details
70 * Iterates cells in sorted key order to ensure deterministic checking and avoids
71 * duplicate pair checks by only checking neighbor cells whose key is greater than the current.
72 */
73 void checkCollisions(Coordinator &coord);

75 /**
76 * @brief Build the spatial-hash grid from a set of entity ids.
77 *
78 * @param coord Coordinator to fetch components for each entity.
79 * @param entities Set of entity ids to insert into the grid (typically system->_filtered_entities).
80 *
81 * @details
82 * For each entity this function:
83 * - reads position_attr_t and hitbox_attr_t (skips entity if missing),
84 * - optionally reads direction_attr_t to orient the hitbox,
85 * - computes the four rotated box corners (fillBoxCorners),
86 * - converts corners to an AABB (cornersToAABB),
87 * - translates the AABB to cell coordinates via getCellsCount,
88 * - inserts the entity into each cell that intersects the AABB (insertEntityToCell).
89 *
90 * The grid is a map from packed cell key to vector of entity ids.
91 */
92 void buildGrid(Coordinator &coord, const std::set<uint32_t> &entities);

138 /**
139 * @brief Determine whether two entities collide (instant T) using oriented-rectangle SAT checks.
140 *
141 * @param coord Coordinator to access components.
142 * @param e1 First entity id.
143 * @param e2 Second entity id.
144 * @return true if entities overlap; false otherwise.
145 *
146 * @details
147 * - Guards against missing components and identical entity ids.
148 * - Computes each entity's four corners with fillBoxCorners (respecting optional direction).
149 * - Calls areAllAxisSeparated which performs axis projections and separation checks.
150 * - Returns true when no separating axis is found (i.e. rectangles overlap).
151 */
152 bool checkPairCollision(Coordinator &coord, uint32_t e1, uint32_t e2);

154 /**
155 * @brief SAT helper: Check all candidate axes between two 4-point rectangles.
156 *
157 * @param hb1 4 corners of first rectangle (pairs of x,y floats).
158 * @param hb2 4 corners of second rectangle.
159 * @return true when rectangles are NOT separated on any tested axis (i.e. they overlap).
160 *
161 * @details
162 * The function computes two edge axes per rectangle and tests each axis for projection separation.
163 * If any axis separates the intervals, rectangles do not overlap. The function returns false
164 * when a separating axis is found; otherwise true (meaning overlap).
165 */
166 bool areAllAxisSeparated(const std::array<std::pair<float, float>, 4> &hb1,
167 const std::array<std::pair<float, float>, 4> &hb2);
168
169 /**
170 * @brief Test separation on a single axis defined by (ax, ay).
171 *
172 * @param hb1 Corners of first rectangle.
173 * @param hb2 Corners of second rectangle.
174 * @param ax X component of normalization axis.
175 * @param ay Y component of normalization axis.
176 * @return true if the projections are separated on this axis (no overlap).
177 *
178 * @details
179 * Projects both rectangles onto the axis and compares the min/max intervals.
180 * Returns true if (maxA < minB) || (maxB < minA).
181 */
182 bool isAxisSeparation(const std::array<std::pair<float, float>, 4> &hb1,
183 const std::array<std::pair<float, float>, 4> &hb2, float ax, float ay);

187 /**
188 * @brief Compute the four corner points of a (possibly oriented) rectangle.
189 *
190 * @param posInt Position component (fixed-point integer units; implementation divides by 100).
191 * @param hbInt Hitbox component (width/height integers).
192 * @param dir Optional direction vector (if nullptr the rectangle is axis-aligned).
193 * @param out Output array of four (x,y) float corner coordinates.
194 *
195 * @details
196 * - Interprets posInt.x/posInt.y as fixed-point (divides by 100.0 to get float world units).
197 * - Computes a local forward vector from dir (or uses unit X if dir missing), normalizes it,
198 * computes perpendicular vector, then constructs corners using half-width/half-height.
199 * - Order of corners is stable and used by projection routines.
200 */
201 void fillBoxCorners(const position_attr_t &posInt,
202 const hitbox_attr_t &hbInt, const direction_attr_t *dir,
203 std::array<std::pair<float, float>, 4> &out) const;

205 /**
206 * @brief Convert 4 corner points to an integer AABB extent.
207 *
208 * @param corners Array of 4 corner (x,y) floats.
209 * @param minX Output min X (floor).
210 * @param minY Output min Y (floor).
211 * @param maxX Output max X (ceil).
212 * @param maxY Output max Y (ceil).
213 *
214 * @details
215 * Produces a conservative integer AABB that fully contains the oriented rectangle.
216 * Useful for mapping the oriented box to grid cells (broad-phase insertion).
217 */
218 void cornersToAABB(const std::array<std::pair<float, float>, 4> &corners,
219 int &minX, int &minY, int &maxX, int &maxY) const;

221 /**

223 *
224 * @param corners Rectangle corners.
225 * @param ax X component of axis (should be normalized).
226 * @param ay Y component of axis (should be normalized).
227 * @param outMin Output minimum projection scalar.
228 * @param outMax Output maximum projection scalar.
229 *
230 * @details
231 * Scalar projection = x*ax + y*ay. Used by SAT separation checks.
232 */
233 void projectOnAxis(const std::array<std::pair<float, float>, 4> &corners,
234 float ax, float ay, float &outMin, float &outMax);
