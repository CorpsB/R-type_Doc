29void rtype::ecs::CollisionWorld::checkCollisions(Coordinator &coord)
30{
31 std::vector<cellKey> keys;
32
33 keys.reserve(_grid.size());
34 for (auto &kv : _grid)
35 keys.push_back(kv.first);
36 std::sort(keys.begin(), keys.end());
37
38 for (cellKey k : keys) {
39 auto &vec = _grid[k];
40 const size_t n = vec.size();
41
42 // check pairs inside same cell
43 for (size_t i = 0; i < n; ++i) {
44 uint32_t e1 = vec[i];
45 for (size_t j = i + 1; j < n; ++j) {
46 uint32_t e2 = vec[j];
47 if (checkPairCollision(coord, e1, e2))
48 collisionEffect(e1, e2);
49 }
50 }
51
52 // check the cell against neighbor cells with larger key
53 auto [cx, cy] = decodeKey(k);
54 for (int dy = -1; dy <= 1; ++dy) {
55 for (int dx = -1; dx <= 1; ++dx) {
56 int neighbor_cx = cx + dx;
57 int neighbor_cy = cy + dy;
58 cellKey neighbor_k = makeKey(neighbor_cx, neighbor_cy);
59
60 if (neighbor_k <= k)
61 continue;
62
63 auto it = _grid.find(neighbor_k);
64 if (it == _grid.end())
65 continue;
66
67 auto &other = it->second;
68 for (uint32_t e1 : vec) {
69 for (uint32_t e2 : other) {
70 if (checkPairCollision(coord, e1, e2))
71 collisionEffect(e1, e2);
72 }
73 }
74 }
75 }
76 }
77}

79void rtype::ecs::CollisionWorld::buildGrid(Coordinator &coord,
80 const std::set<uint32_t> &entities)
81{
82 for (uint32_t e : entities) {
83 auto pos = coord.getComponent<position_attr_t>(e);
84 auto hb = coord.getComponent<hitbox_attr_t>(e);
85
86 if (!pos || !hb)
87 continue;
88
89 auto dir = coord.getComponent<direction_attr_t>(e);
90
91 std::array<std::pair<float, float>, 4> corners;
92 fillBoxCorners(*pos, *hb, dir, corners);
93
94 int minX, minY, maxX, maxY;
95 cornersToAABB(corners, minX, minY, maxX, maxY);
96
97 auto [minCx, minCy] = getCellsCount(static_cast<float>(minX),
98 static_cast<float>(minY));
99 auto [maxCx, maxCy] = getCellsCount(static_cast<float>(maxX),
100 static_cast<float>(maxY));
101
102 for (int cy = minCy; cy <= maxCy; ++cy)
103 for (int cx = minCx; cx <= maxCx; ++cx)
104 insertEntityToCell(cx, cy, e);
105 }
106}

125bool rtype::ecs::CollisionWorld::checkPairCollision(Coordinator &coord,
126 uint32_t e1, uint32_t e2)
127{
128 if (e1 == e2)
129 return false;
130
131 position_attr_t *pos_e1 = coord.getComponent<position_attr_t>(e1);
132 position_attr_t *pos_e2 = coord.getComponent<position_attr_t>(e2);
133
134 if (!pos_e1 || !pos_e2)
135 return false;
136
137 hitbox_attr_t *hb_e1 = coord.getComponent<hitbox_attr_t>(e1);
138 hitbox_attr_t *hb_e2 = coord.getComponent<hitbox_attr_t>(e2);
139
140 if (!hb_e1 || !hb_e2)
141 return false;
142
143 direction_attr_t *dir_e1 = coord.getComponent<direction_attr_t>(e1);
144 direction_attr_t *dir_e2 = coord.getComponent<direction_attr_t>(e2);
145
146 std::array<std::pair<float, float>, 4> e1_box;
147 std::array<std::pair<float, float>, 4> e2_box;
148
149 fillBoxCorners(*pos_e1, *hb_e1, dir_e1, e1_box);
150 fillBoxCorners(*pos_e2, *hb_e2, dir_e2, e2_box);
151 return areAllAxisSeparated(e1_box, e2_box);
152}

154bool rtype::ecs::CollisionWorld::areAllAxisSeparated(
155 const std::array<std::pair<float, float>, 4> &hb1,
156 const std::array<std::pair<float, float>, 4> &hb2)
157{
158 float ax0x = hb1[1].first - hb1[0].first;
159 float ax0y = hb1[1].second - hb1[0].second;
160 float ax1x = hb1[2].first - hb1[0].first;
161 float ax1y = hb1[2].second - hb1[0].second;
162
163 float bx0x = hb2[1].first - hb2[0].first;
164 float bx0y = hb2[1].second - hb2[0].second;
165 float bx1x = hb2[2].first - hb2[0].first;
166 float bx1y = hb2[2].second - hb2[0].second;
167
168 normalize(ax0x, ax0y);
169 normalize(ax1x, ax1y);
170 normalize(bx0x, bx0y);
171 normalize(bx1x, bx1y);
172
173 if (isAxisSeparation(hb1, hb2, ax0x, ax0y) ||
174 isAxisSeparation(hb1, hb2, ax1x, ax1y) ||
175 isAxisSeparation(hb1, hb2, bx0x, bx0y) ||
176 isAxisSeparation(hb1, hb2, bx1x, bx1y))
177 return false;
178 return true;
179}

181bool rtype::ecs::CollisionWorld::isAxisSeparation(
182 const std::array<std::pair<float, float>, 4> &hb1,
183 const std::array<std::pair<float, float>, 4> &hb2, float ax, float ay)
184{
185 float minA, maxA, minB, maxB;
186
187 projectOnAxis(hb1, ax, ay, minA, maxA);
188 projectOnAxis(hb2, ax, ay, minB, maxB);
189 return (maxA < minB) || (maxB < minA);
190}

192void rtype::ecs::CollisionWorld::fillBoxCorners(const position_attr_t &posInt,
193 const hitbox_attr_t &hbInt, const direction_attr_t *dir,
194 std::array<std::pair<float, float>, 4> &out) const
195{
196 const float cx = static_cast<float>(posInt.x) / 100.0;
197 const float cy = static_cast<float>(posInt.y) / 100.0;
198 const float halfW = (static_cast<float>(hbInt.horizontal) * 0.5f) / 100.0;
199 const float halfH = (static_cast<float>(hbInt.vertical) * 0.5f) / 100.0;
200
201 float ux = 1.0f;
202 float uy = 0.0f;
203
204 if (dir) {
205 ux = dir->dx;
206 uy = dir->dy;
207 normalize(ux, uy);
208 }
209
210 float vx = -uy;
211 float vy = ux;
212
213 out[0] = { cx + ux * halfW + vx * halfH, cy + uy * halfW + vy * halfH };
214 out[1] = { cx + ux * halfW - vx * halfH, cy + uy * halfW - vy * halfH };
215 out[2] = { cx - ux * halfW + vx * halfH, cy - uy * halfW + vy * halfH };
216 out[3] = { cx - ux * halfW - vx * halfH, cy - uy * halfW - vy * halfH };
217}

219void rtype::ecs::CollisionWorld::cornersToAABB(
220 const std::array<std::pair<float, float>, 4> &corners,
221 int &minX, int &minY, int &maxX, int &maxY) const
222{
223 float min_fx = corners[0].first;
224 float max_fx = corners[0].first;
225 float min_fy = corners[0].second;
226 float max_fy = corners[0].second;
227
228 for (int i = 1; i < 4; ++i) {
229 min_fx = std::min(min_fx, corners[i].first);
230 max_fx = std::max(max_fx, corners[i].first);
231 min_fy = std::min(min_fy, corners[i].second);
232 max_fy = std::max(max_fy, corners[i].second);
233 }
234
235 minX = static_cast<int>(std::floor(min_fx));
236 minY = static_cast<int>(std::floor(min_fy));
237 maxX = static_cast<int>(std::ceil(max_fx));
238 maxY = static_cast<int>(std::ceil(max_fy));
239}

241void rtype::ecs::CollisionWorld::projectOnAxis(
242 const std::array<std::pair<float, float>, 4> &corners, float ax, float ay,
243 float &outMin, float &outMax)
244{
245 outMin = corners[0].first * ax + corners[0].second * ay;
246 outMax = outMin;
247
248 for (int i = 1; i < 4; ++i) {
249 const float tmp = corners[i].first * ax + corners[i].second * ay;
250
251 if (tmp < outMin)
252 outMin = tmp;
253 if (tmp > outMax)
254 outMax = tmp;
255 }
256}
