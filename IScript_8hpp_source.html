26namespace asge::script {
27
28/**
29 * @class IScript
30 * @brief Abstract public interface for the scripting subsystem.
31 *
32 * @details
33 * IScript provides the minimal API used by the engine to interact with the
34 * scripting subsystem: managing packages, invoking hooks, registering host
35 * APIs and handling events. Implementations must be thread-safe where noted.
36 */
37class IScript {
38 public:
39 /**
40 * @brief Host function signature used by script-hosted APIs.
41 *
42 * @param positional Positional arguments provided as ScriptValue entries.
43 * @param kwargs Keyword arguments provided as a string->ScriptValue map.
44 * @returns A ScriptValue representing the function result (may be null).
45 *
46 * @details
47 * HostFn implementations should be exception safe — exceptions are caught
48 * by callers and logged by the ScriptManager.
49 */
50 using HostFn = std::function<ScriptValue(
51 const std::vector<ScriptValue> &positional,
52 const std::unordered_map<std::string, ScriptValue> &kwargs)>;
53
54 virtual ~IScript() = default;
55
56 /**
57 * @brief Load a script package from storage.
58 * @param gameId Identifier of the game/package to load.
59 * @param assetsRoot Root path containing package assets.
60 * @returns true if at least one template was successfully loaded.
61 *
62 * @details
63 * Loading clears prior state and populates internal template maps.
64 */
65 virtual bool loadPackage(const std::string &gameId,
66 const std::string &assetsRoot) = 0;
67
68 /**
69 * @brief Reload currently loaded package.
70 * @returns true on success, false on failure.
71 *
72 * @details
73 * Reload attempts to re-discover and re-parse templates and assets without
74 * changing configured runtime instance.
75 */
76 virtual bool reload() = 0;
77
78 /**
79 * @brief Unload the current package and clear internal caches.
80 *
81 * @details
82 * After unload the IScript instance should hold no templates, queued
83 * actions or host API registrations.
84 */
85 virtual void unload() = 0;
86
87 /**
88 * @brief Invoke a lifecycle hook across loaded templates.
89 * @param hook Name of the hook (e.g. "on_create", "on_update").
90 * @param dt Optional delta time (seconds) for update hooks.
91 *
92 * @details
93 * The manager will collect any produced actions and enqueue them for later
94 * processing; this call is non-blocking with respect to action execution.
95 */
96 virtual void callHook(const std::string &hook, double dt = 0.0) = 0;
97
98 /**
99 * @brief Register a host API function callable from scripts.
100 * @param name API name exposed to scripts.
101 * @param fn Host function to be invoked.
102 *
103 * @details
104 * If a function with the same name already exists it will be replaced.
105 */
106 virtual void registerHostAPI(const std::string &name, HostFn fn) = 0;
107
108 /**
109 * @brief Notify the script subsystem of an external event.
110 * @param type Event type string.
111 * @param payload Event payload (raw string).
112 *
113 * @details
114 * Events may be dispatched to a registered host API or turned into queued
115 * actions (e.g. ScriptCommand).
116 */
117 virtual void onEvent(const std::string &type, const std::string &payload) = 0;
118
119 /**
120 * @brief Inform the script system that a new entity was created.
121 * @param entityId Identifier of the newly created entity.
122 * @param templateId Template id the entity was created from.
123 *
124 * @details
125 * May produce queued actions derived from the template's on_create behavior.
126 */
127 virtual void notifyEntityCreated(std::uint32_t entityId,
128 const std::string &templateId) = 0;
129
130 /**
131 * @brief Notify the script system that an entity was destroyed.
132 * @param entityId Identifier of the entity.
133 * @param templateId Template reference for bookkeeping.
134 */
135 virtual void notifyEntityDestroyed(std::uint32_t entityId,
136 const std::string &templateId) = 0;
137
138 /**
139 * @brief Inform scripts about an entity collision.
140 * @param entityId Entity involved in the collision.
141 * @param templateId Template id associated with the entity.
142 */
143 virtual void notifyEntityCollisionned(std::uint32_t entityId,
144 const std::string &templateId) = 0;
145
146 /**
147 * @brief Share the entity world position with scripts.
148 * @param entityId Entity identifier.
149 * @param templateId Template id associated with the entity.
150 * @param x X coordinate in world space.
151 * @param y Y coordinate in world space.
152 */
153 virtual void notifyEntityPosition(std::uint32_t entityId,
154 const std::string &templateId, double x, double y) = 0;
155
156 /**
157 * @brief Drain and return the current queued actions.
158 * @returns A vector of QueuedAction previously produced by hooks/events.
159 *
160 * @details
161 * Ownership of the returned vector transfers to the caller; the internal
162 * action queue is cleared.
163 */
164 virtual std::vector<QueuedAction> drainQueuedActions() = 0;
165
166 /**
167 * @brief Retrieve a template handle by id.
168 * @param templateId Template identifier to look up.
169 * @returns Pointer to a ScriptHandle on success, nullptr if not found.
170 *
171 * @details
172 * The returned pointer references internal storage and is valid until
173 * the template is removed or the manager is unloaded/reloaded.
174 */
175 virtual const ScriptHandle *getTemplate(const std::string &templateId) const = 0;
176
177 /**
178 * @brief Access all loaded templates map.
179 * @return const reference to internal templates map (read-only).
180 *
181 * @details
182 * The map is owned by the script implementation; the caller must not
183 * modify it. The reference is valid until the manager is reloaded/unloaded.
184 */
185 virtual const std::unordered_map<std::string, ScriptHandle> &templates() const = 0;
186
187 /**
188 * @brief Get parsed sprite asset by id (nullptr if not found).
189 */
190 virtual const SpriteAsset *getSprite(const std::string &spriteId) const = 0;
191
192 /**
193 * @brief Get parsed audio asset by id (nullptr if not found).
194 */
195 virtual const AudioAsset *getAudio(const std::string &audioId) const = 0;
196
197 /**
198 * @brief Get parsed map asset by id (nullptr if not found).
199 */
200 virtual const MapAsset *getMap(const std::string &mapId) const = 0;
201
202 /**
203 * @brief Returns the absolute root path of the currently loaded assets.
204 */
205 virtual std::string getAssetsRoot() const = 0;
206};
207
208}
