27/**
28 * @class IOption
29 * @brief Interface for selectable menu options.
30 *
31 * @details
32 * The IOption class represents a single selectable element (box) in the game’s menus.
33 * Each option can respond visually to selection or editing states.
34 * Implementing classes should define how the option behaves when highlighted or modified.
35 */
36class IOption {
37 public:
38 /**
39 * @brief Virtual destructor for proper cleanup in derived classes.

42
43 /**
44 * @brief Updates the state of the option.
45 * @param selected True if the option is currently highlighted.
46 * @param editing True if the option is currently being modified.
47 *
48 * @details
49 * Called once per frame to update the internal state of the option
50 * based on user navigation or editing actions.
51 */
52 virtual void update(bool selected) = 0;
53
54 /**
55 * @brief Draws the option on the screen.
56 * @param selected True if the option is currently highlighted.
57 * @param editing True if the option is currently being modified.
58 *
59 * @details
60 * Responsible for rendering the visual representation of the menu box,
61 * including selection or editing feedback.
62 */
63 virtual void draw(bool selected) const = 0;
64
65 /**
