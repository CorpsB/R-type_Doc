45void rtype::server::run()
46{
47 _coo.init();
48
49 rtype::ecs::factory::register_game_components(_coo);
50 auto collision = _coo.registerSystem<ecs::CollisionSystem>();
51 auto movement =_coo.registerSystem<ecs::MovementSystem>();
52 auto spawner = _coo.registerSystem<ecs::SpawnerSystem>();
53 auto player_sys = _coo.registerSystem<ecs::PlayerSystem>();
54
55 auto ec = protocol::game::GameProtocol::initialize(_args.ip, static_cast<uint16_t>(_args.port));
56 if (ec) {
57 std::cerr << "[ERROR] GameProtocol::initialize failed: " << ec.message() << "\n";
58 return;
59 }
60 protocol::game::GameProtocol::start_io();
61 _clock.start();
62 if (_args.debug >= 1)
63 std::cout << "[INFO] Server started, listening on " << _args.ip << ":" << _args.port << "\n";
64
65 protocol::game::game_event_t event;
66
67 while (!_isEnd) {
68 _clock.waitNextTick();
69
70 protocol::game::GameProtocol::process_pending(std::chrono::steady_clock::now());
71 while (protocol::game::GameProtocol::is_event())
72 {
73 std::cout << "Got something" << std::endl;
74 if (!protocol::game::GameProtocol::get_event(event))
75 break;
76
77 std::cout << "Got something " << (int)event.type << std::endl;
78 switch (event.type) {
79 case protocol::game::game_event_type_t::CONNECTED : {
80 auto ep = std::get<protocol::udp::endpoint>(event.payload);
81 std::string key = protocol::utils::endpoint_to_string(ep);
82 auto &proto = protocol::Protocol::instance();
83
84 if (_args.debug >= 2)
85 std::cout << "[DEBUG] - New player CONNECTED from " << key << std::endl;
86
87 proto.storeEndpoint(ep);
88
89 if (_client_ids.find(key) == _client_ids.end()) {
90 uint32_t assigned = _next_player_id++;
91 _client_ids.emplace(key, assigned);
92
93 protocol::game::payload_create_entity_t entity;
94 entity.entity_id = assigned;
95 entity.entity_type = protocol::game::entity_type_t::PLAYER;
96 entity.x = 100;
97 entity.y = 100;
98
99 protocol::game::GameProtocol::send_create_entity(entity, protocol::flag_t::RELIABLE_ORDERED);
100
101 if (_args.debug >= 2)
102 std::cout << "[DEBUG] - Assigned player id " << assigned << " to " << key << std::endl;
103 } else {
104 if (_args.debug >= 2)
105 std::cout << "[DEBUG] - CONNECTED from existing client " << key << std::endl;
106 }
107 break;
108 }
109
110 case protocol::game::game_event_type_t::DISCONNECTED : {
111 auto ep = std::get<protocol::udp::endpoint>(event.payload);
112 std::string key = protocol::utils::endpoint_to_string(ep);
113 auto &proto = protocol::Protocol::instance();
114
115 if (_args.debug >= 2)
116 std::cout << "[DEBUG] - DISCONNECTED from " << key << std::endl;
117
118 if (auto it = _client_ids.find(key); it != _client_ids.end()) {
119 uint32_t id = it->second;
120 _client_ids.erase(it);
121
122 protocol::game::payload_destroy_entity_t del;
123 del.entity_id = id;
124
125 protocol::game::GameProtocol::send_destroy_entity(del, protocol::flag_t::RELIABLE_ORDERED);
126
127 if (_args.debug >= 2)
128 std::cout << "[DEBUG] - Removed client " << key << " (id=" << id << ")" << std::endl;
129 } else {
130 if (_args.debug >= 2)
131 std::cout << "[DEBUG] - DISCONNECT from unknown client " << key << std::endl;
132 }
133 break;
134 }
135
136 case protocol::game::game_event_type_t::PLAYER_INPUT: {
137 auto target = std::get<protocol::game::payload_player_input_t>(event.payload);
138
139 if (_args.debug >= 2)
140 std::cout << "[DEBUG] - Player send :"<< std::endl;
141 player_sys->move(_coo, 1, static_cast<protocol::game::player_input_t>(target.buttons));
142 break;
143 }
144
145 case protocol::game::game_event_type_t::TRANSMIT_FAILED: {
146 auto const &tf = std::get<protocol::transmit_failed_t>(event.payload);
147
148 std::cout << "[DEBUG] - [TX FAIL] seq=" << tf.sequence << " msg=" << tf.message_type << "\n";
149 break;
150 }
151
152 default:
153 break;
154 }
155 }
156
157 collision->update(_coo);
158 movement->update(_coo);
159 spawner->update(_coo, _args.seed, _clock.getTick());
160 protocol::game::GameProtocol::process_pending(std::chrono::steady_clock::now());
161 }
162
163 protocol::game::GameProtocol::stop_io();
164 protocol::game::GameProtocol::shutdown();
165}
