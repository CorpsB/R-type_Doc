89bool LoopbackChannel::send(const std::vector<std::uint8_t>& bytes)
90{
91 if (_selfPort == 0) {
92 throw asge::error::Error(
93 asge::error::ErrorType::Network,
94 "LoopbackChannel::send",
95 "Channel not opened (call openServer/openClient first)",
96 asge::error::ErrorSeverity::Recoverable
97 );
98 }
99 if (bytes.size() > _mtu) {
100 throw asge::error::Error(
101 asge::error::ErrorType::Network,
102 "LoopbackChannel::send",
103 "Packet size exceeds MTU",
104 asge::error::ErrorSeverity::Recoverable
105 );
106 }
107
108 std::lock_guard<std::mutex> queue_lock(_mutex);
109 if (_queue.size() >= QUEUE_MAX) {
110 throw asge::error::Error(
111 asge::error::ErrorType::Network,
112 "LoopbackChannel::send",
113 "Loopback internal queue is full",
114 asge::error::ErrorSeverity::Recoverable
115 );
116 }
117
118 _queue.push_back(bytes);
119 return true;
120}

140void LoopbackChannel::poll(
141 std::function<void(const std::vector<std::uint8_t>&, const std::string&, std::uint16_t)> onPacket)
142{
143 if (_selfPort == 0) {
144 throw asge::error::Error(
145 asge::error::ErrorType::Network,
146 "LoopbackChannel::poll",
147 "Channel not opened (call openServer/openClient first)",
148 asge::error::ErrorSeverity::Recoverable
149 );
150 }
151 if (!onPacket) {
152 throw asge::error::Error(
153 asge::error::ErrorType::Network,
154 "LoopbackChannel::poll",
155 "Invalid callback (nullptr)",
156 asge::error::ErrorSeverity::Recoverable
157 );
158 }
159
160 std::deque<std::vector<std::uint8_t>> local;
161 {
162 std::lock_guard<std::mutex> queue_lock(_mutex);
163 local.swap(_queue);
164 }
165 for (auto& packet : local) {
166 onPacket(packet, _selfAddress, _selfPort);
167 }
168}
