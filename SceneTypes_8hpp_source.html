10/**
11 * @file SceneTypes.hpp
12 * @brief Fundamental POD types and enums used by the scene subsystem.
13 *
14 * @details
15 * This header declares lightweight, copyable types and enums used across the scene
16 * manager and scene implementations:
17 * - SceneId: numeric unique identifier for a pushed scene instance.
18 * - SceneParams: string-string parameter bag forwarded at scene creation.
19 * - SceneFlag: bitflags describing scene behaviour (Overlay, Modal, Persistent).
20 * - SceneHandle: a compact descriptor representing a pushed scene (id, type, flags, params).
21 * - SceneState: lifecycle states tracked by scenes.
22 *
23 * These types are intentionally small and header-only so they can be used in headers
24 * across the engine without requiring heavy includes.
25 */

41/**
42 * @enum SceneFlag
43 * @brief Flags describing a scene instance behaviour.
44 *
45 * @details
46 * The flags are used by SceneManager and SceneStage to control how a scene behaves
47 * in a stack. Typical usage:
48 * - Overlay: scene renders on top of the previous scene (previous remains visible).
49 * - Modal: blocks input to scenes underneath.
50 * - Persistent: the scene should not be automatically destroyed across some transitions.
51 */
52enum class SceneFlag : std::uint32_t {
53 None = 0u, ///< No special behaviour.
54 Overlay = 1u << 0, ///< Renders above previous scene (previous remains visible).
55 Modal = 1u << 1, ///< Blocks input to scenes below.
56 Persistent = 1u << 2 ///< Not automatically destroyed by some transitions.
57};

61 * @param a Left operand.
62 * @param b Right operand.
63 * @return Combined flags.

65inline SceneFlag operator|(SceneFlag a, SceneFlag b) {
66 return static_cast<SceneFlag>(static_cast<std::uint32_t>(a) | static_cast<std::uint32_t>(b));
67}

70 * @brief Bitwise AND operator for SceneFlag.

79/**
80 * @struct SceneHandle
81 * @brief Lightweight descriptor for a pushed scene instance.
82 *
83 * @details
84 * SceneHandle represents metadata for a scene instance pushed onto the SceneManager stack.
85 * It is intentionally copyable and small so the manager can queue pending operations using it.
86 *
87 * Members:
88 * - id: numeric id assigned by SceneManager (0 = invalid/unassigned).
89 * - type: textual type identifier used for diagnostics and factory lookups.
90 * - flags: SceneFlag bitset describing overlay/modal/persistence behaviour.
91 * - params: optional key/value parameters forwarded to scene init.
92 */
93struct SceneHandle {
94 SceneId id = 0; ///< Numeric instance identifier (0 = invalid).
95 std::string type; ///< Scene type name (e.g. "GameStage" or "MainMenu").
96 SceneFlag flags = SceneFlag::None; ///< Behaviour flags for the scene.
97 SceneParams params; ///< Optional startup parameters forwarded to the scene.
98};

100/**
101 * @enum SceneState
102 * @brief Lifecycle state machine tracked by IScene/SceneStage/SceneManager.
103 *
104 * @details
105 * Scenes transition through these states as they are created, initialized, entered,
106 * paused/resumed and finally destroyed. SceneManager uses this enum to decide which
107 * callbacks (onEnter/onExit) to invoke and to prevent double-init/double-destroy.
108 */
109enum class SceneState : std::uint8_t {
110 Created, ///< Scene object constructed but not initialized.
111 Initialized, ///< init() completed.
112 Entered, ///< onEnter() was invoked and scene is active.
113 Running, ///< Normal running state (after enter).
114 Paused, ///< Scene is paused (another scene overlays it).
115 Exited, ///< onExit() was called (pending destruction or reuse).
116 Destroyed ///< Final destructor state (object should be released).
117};
