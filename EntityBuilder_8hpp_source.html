12/**
13 * @file EntityBuilder.hpp
14 * @brief Fluent builder for composing ECS entities from script-provided templates.
15 * @since 1.0
16 * @ingroup scripting
17 *
18 * @details
19 * `EntityBuilder` offers a chainable API to assemble an entity using optional
20 * component data parsed from script templates. Only the components explicitly
21 * provided are attached to the entity. The builder is single-use: calling
22 * `build()` finalizes the instance and subsequent calls return the cached ID.
23 *
24 * Typical usage:
25 * @code
26 * EntityBuilder b(services);
27 * const auto eid = b.withTemplateId("enemy_basic")
28 * .withPosition(pos)
29 * .withSprite(sprite)
30 * .build(std::nullopt);
31 * @endcode
32 */
33

43/**
44 * @class EntityBuilder
45 * @brief Helper that assembles an entity based on optional component payloads.
46 *
47 * @details
48 * The builder defers entity creation until `build()`. If an existing entity
49 * identifier is supplied to `build()`, components are added onto that entity
50 * instead of creating a new one. The builder logs warnings through the debug
51 * service when component insertion fails but does not throw.
52 */
53class EntityBuilder
54{
55 public:
56 /**
57 * @brief Constructs a builder with access to engine services.
58 * @param services Reference to the engine service container.
59 * @note The builder does not take ownership of \p services and assumes
60 * the referenced services remain valid for the builder's lifetime.
61 */
62 explicit EntityBuilder(Services &services) noexcept;

69 * `IScript::notifyEntityCreated` after a successful build.
70 */

143 /**
144 * @brief Builds or completes the entity and attaches configured components.
145 *
146 * @param entity_id Optional existing entity identifier to reuse.
147 * If provided, components are added onto this entity.
148 * If not provided, a new entity is created.
149 * @return The created or reused entity identifier. Returns 0 (INVALID_ID)
150 * if world services are missing or if creation fails.
151 *
152 * @post On success, the builder becomes immutable; repeated calls return
153 * the same cached identifier without re-adding components.
154 *
155 * @warning Component insertion failures are logged as warnings and do
156 * not abort the build; partial entities may be produced.
157 */
158 uint32_t build(const std::optional<uint32_t> &entity_id);
