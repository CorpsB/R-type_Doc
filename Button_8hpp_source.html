18 * dynamic rendering, and callback invocation for user interactions.
19 *
20 * Responsibilities:
21 * - Manage button geometry, style, and label text.
22 * - Detect hover and click states through per-frame updates.
23 * - Execute user-defined callbacks upon interaction.
24 * - Render the button with custom colors, fonts, and radius settings.
25 *
26 * The button is suitable for lightweight, immediate-mode GUIs such as
27 * those used in Raylib or similar frameworks.
28 *
29 * @note Label localization can be handled through @ref LanguageManager.
30 * @warning The callback set via @ref onClickAction must remain valid throughout the button’s lifetime.

41/**
42 * @class Button
43 * @brief Interactive rectangular UI element for handling clicks and hover effects.
44 *
45 * @details
46 * The **Button** class provides a minimal yet flexible implementation of an interactive
47 * button rendered using Raylib. It supports hover detection, pressed states,
48 * callbacks on click, and customizable styles via @ref Button::Style.
49 *
50 * ### Example usage:
51 * @code{.cpp}
52 * using namespace rtype::client::ui;
53 *
54 * Button btn(100, 100, 200, 60, "Play");
55 * btn.onClickAction([]() { startGame(); });
56 *
57 * while (!WindowShouldClose()) {
58 * btn.update();
59 * BeginDrawing();
60 * ClearBackground(RAYWHITE);
61 * btn.draw();
62 * EndDrawing();
63 * }
64 * @endcode
65 *
66 * @see GameSettings, LanguageManager
67 */
68class Button {
69 public:
70 /**
71 * @struct Style
72 * @brief Defines visual and typographic properties for a button.
73 *
74 * @details
75 * Encapsulates color states, font settings, and border radius for rendering.
76 * Each button instance has its own @ref Style, allowing for flexible customization.
77 *
78 * @see Button::setStyle
79 */
80 struct Style {
81 Color normalColor = DARKGRAY; ///< Default fill color when idle.
82 Color hoverColor = GRAY; ///< Color displayed when hovered by the mouse.
83 Color pressedColor = LIGHTGRAY; ///< Color displayed when pressed.
84 Color textColor = RAYWHITE; ///< Text color for the button label.
85 Font font = GetFontDefault(); ///< Font used for the text.
86 float fontSize = 40.0f; ///< Base font size (scaled automatically).
87 float borderRadius = 6.0f; ///< Radius for rounded corners.
88 };
89
90 /**
91 * @brief Constructs a new Button instance.
92 *
93 * @details
94 * Initializes a rectangular button with given position, size, and label.
95 * The button becomes active once @ref update() is called within the game loop.
96 *
97 * @param x Horizontal position (top-left corner).
98 * @param y Vertical position (top-left corner).
99 * @param width Button width in pixels.
100 * @param height Button height in pixels.
101 * @param text Label text displayed on the button.
102 */
103 Button(float x, float y, float width, float height, const std::string &text) :
104 _bounds{ x, y, width, height }, _label(text) {};
105
106 /**

111 Button &setStyle(const Style &style) { _style = style; return *this; };

121 /**

127 Button &setSize(float w, float h) { _bounds.width = w; _bounds.height = h; return *this; };

129 /**
130 * @brief Updates the label text of the button.
131 * @param label New label text.
132 * @return Reference to the current @ref Button instance.
133 */
134 Button &setText(const std::string &label) { _label = label; return *this; };
135
136 /**
137 * @brief Registers a callback executed on click.
138 *
139 * @details
140 * Binds a user-defined function to be executed when the button is clicked.
141 * The callback is triggered upon mouse release while hovered.
142 *
143 * @param callBack Function to invoke on click.
144 * @return Reference to the current @ref Button instance.
145 *
146 * @warning The callback must remain valid during the button’s lifetime.
147 */
148 Button &onClickAction(const std::function<void()> &callBack) { _onClick = callBack; return *this; };
149
150 /**
151 * @brief Returns the button’s bounding rectangle.
152 * @return Reference to the internal Raylib @c Rectangle structure.
153 */
154 Rectangle &getBounds() { return _bounds; };
155
156 /**
157 * @brief Updates the button state based on user input.
158 *
159 * @details
160 * Detects mouse hover and click states, invoking the callback when necessary.

162 *
163 * @see draw
164 */
165 void update();
166
167 /**
168 * @brief Draws the button using Raylib primitives.
169 *
170 * @details
171 * Renders a rounded rectangle based on the current button state and style.
172 * Displays the text label centered within the bounds.
173 *
174 * @see LanguageManager::getTranslation
175 */
176 void draw() const;
177 private:
178 Rectangle _bounds; ///< Position and dimensions of the button.
179 Style _style; ///< Visual style configuration.
180 std::string _label; ///< Text label displayed on the button.
181
182 bool _isHovered = false; ///< True when the mouse is over the button.
183 bool _isPressed = false; ///< True when the mouse button is held down.
184
185 std::function<void()> _onClick; ///< Function executed when the button is clicked.
186};
