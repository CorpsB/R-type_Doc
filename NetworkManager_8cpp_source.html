68void writeU32(std::uint8_t *destination, std::uint32_t value)
69{
70 destination[0] = static_cast<std::uint8_t>(value & 0xFF);
71 destination[1] = static_cast<std::uint8_t>((value >> 8) & 0xFF);
72 destination[2] = static_cast<std::uint8_t>((value >> 16) & 0xFF);
73 destination[3] = static_cast<std::uint8_t>((value >> 24) & 0xFF);
74}

76std::uint32_t readU32(const std::uint8_t *source)
77{
78 return static_cast<std::uint32_t>(source[0])
79 | (static_cast<std::uint32_t>(source[1]) << 8)
80 | (static_cast<std::uint32_t>(source[2]) << 16)
81 | (static_cast<std::uint32_t>(source[3]) << 24);
82}

86void NetworkManager::writeHeader(std::uint8_t *destination, const NetHeader &header)
87{
88 writeU32(destination + 0, header.seq);
89 writeU32(destination + 4, header.ack);
90 destination[8] = header.flags;
91 destination[9] = header.channel;
92 destination[10] = static_cast<std::uint8_t>(header.reserved & 0xFF);
93 destination[11] = static_cast<std::uint8_t>((header.reserved >> 8) & 0xFF);
94}

96NetHeader NetworkManager::readHeader(const std::uint8_t *source, std::size_t size, bool &success)
97{
98 NetHeader header{};
99 success = false;
100 if (size < NET_HEADER_SIZE) {
101 return header;
102 }
103 header.seq = readU32(source + 0);
104 header.ack = readU32(source + 4);
105 header.flags = source[8];
106 header.channel = source[9];
107 header.reserved = static_cast<std::uint16_t>(source[10])
108 | (static_cast<std::uint16_t>(source[11]) << 8);
109 success = true;
110 return header;
111}

192 throw asge::error::Error(
193 asge::error::ErrorType::Network,

195 "No channel bound to NetworkManager",
196 asge::error::ErrorSeverity::Fatal
197 );
198 }
199 if (!_running) {
200 throw asge::error::Error(
201 asge::error::ErrorType::Network,

204 asge::error::ErrorSeverity::Recoverable
205 );
206 }
207 if (channel < 0 || channel > std::numeric_limits<std::uint8_t>::max()) {
208 throw asge::error::Error(
209 asge::error::ErrorType::Network,
210 "NetworkManager::send",
211 "Channel out of range [0..255]",
212 asge::error::ErrorSeverity::Recoverable
213 );
214 }
215
216 const std::size_t configured_mtu = _channel->mtu();
217 if (NET_HEADER_SIZE + data.size() > configured_mtu) {
218 throw asge::error::Error(
219 asge::error::ErrorType::Network,
220 "NetworkManager::send",
221 "Payload too large for current MTU (including header)",
222 asge::error::ErrorSeverity::Recoverable
223 );
224 }
225
226 std::vector<std::uint32_t> ids;
227 {
228 std::lock_guard<std::mutex> lg(_conn_mx);
229
230 ids.reserve(_addressToId.size());
231 for (auto &p : _addressToId)
232 ids.push_back(p.second);
233 }
234
235 if (channel == 0) {
236 for (auto id : ids)
237 sendReliable(id, data, 0);
238 } else {
239 for (auto id : ids)
240 sendBestEffort(id, data, static_cast<std::uint8_t>(channel));

364void NetworkManager::armAndSend(std::uint32_t connection_id,
365 const std::vector<std::uint8_t>& data, int channel)
366{
367 auto connection_iterator = _connections.find(connection_id);
368 if (connection_iterator == _connections.end()) {
369 throw asge::error::Error(
370 asge::error::ErrorType::Network,
371 "NetworkManager::armAndSend",
372 "Unknown connection id",
373 asge::error::ErrorSeverity::Recoverable
374 );
375 }
376 auto &connection = connection_iterator->second;
377
378 const auto configured_mtu = _channel->mtu();
379 if (NET_HEADER_SIZE + data.size() > configured_mtu) {
380 throw asge::error::Error(
381 asge::error::ErrorType::Network,
382 "NetworkManager::armAndSend",
383 "Payload too large for current MTU (including header)",
384 asge::error::ErrorSeverity::Recoverable
385 );
386 }
387
388 NetHeader header{};
389 header.seq = connection.nextSendSeq();
390 header.ack = 0;
391 header.flags = FLAG_RELIABLE;
392 header.channel = static_cast<std::uint8_t>(channel);
393
394 std::vector<std::uint8_t> packet;
395 packet.resize(NET_HEADER_SIZE + data.size());
396 writeHeader(packet.data(), header);
397 std::memcpy(packet.data() + NET_HEADER_SIZE, data.data(), data.size());
398
399 const auto &endpoint = connection.endpoint();
400 if (!_channel->sendTo(packet, endpoint.address, endpoint.port)) {
401 throw asge::error::Error(
402 asge::error::ErrorType::Network,
403 "NetworkManager::armAndSend",
404 "Channel::sendTo() failed",
405 asge::error::ErrorSeverity::Recoverable
406 );
407 }
408
409 if (_debug) {
410 logHeaderTx("RELIABLE", endpoint.address, endpoint.port, header, data.size());
411 }
412
413 connection.bumpSendSeq();
414
415 auto rto = connection.rtt().count() > 0 ? connection.rtt() :
416 std::chrono::milliseconds{200};
417
418 if (rto < _rtoMin)
419 rto = _rtoMin;
420 if (rto > _rtoMax)
421 rto = _rtoMax;
422
423 auto t0 = std::chrono::steady_clock::now();
424 InFlight inflight{
425 .packet = std::move(packet),
426 .lastSend = t0,
427 .rto = rto,
428 .seq = header.seq,
429 .channel = header.channel,
430 .address = endpoint.address,
431 .port = endpoint.port
432 };
433
434 {
435 std::lock_guard<std::mutex> lg(_tx_mx);
436 _tx[connection_id].in_flight = std::move(inflight);
437 }
438}

529void NetworkManager::processIncoming(
530 std::uint32_t connection_id,
531 const std::vector<std::uint8_t> &raw_data,
532 const std::string &address,
533 std::uint16_t port,
534 std::function<void(const std::vector<std::uint8_t> &)> on_receive)
535{
536 if (raw_data.size() < NET_HEADER_SIZE) {
537 throw asge::error::Error(
538 asge::error::ErrorType::Network,
539 "NetworkManager::processIncoming",
540 "Truncated packet (smaller than header)",
541 asge::error::ErrorSeverity::Recoverable
542 );
543 }
544
545 bool header_valid = false;
546 const NetHeader header = readHeader(raw_data.data(), raw_data.size(), header_valid);
547 if (!header_valid) {
548 throw asge::error::Error(
549 asge::error::ErrorType::Network,
550 "NetworkManager::processIncoming",
551 "Failed to parse header",
552 asge::error::ErrorSeverity::Recoverable
553 );
554 }
555
556 std::uint32_t lastRecvSeq = 0;
557 {
558 std::lock_guard<std::mutex> lg(_conn_mx);
559 auto it = _connections.find(connection_id);
560
561 if (it == _connections.end()){
562 throw;
563 }
564 lastRecvSeq = it->second.lastRecvSeq();
565 }
566
567 if (_debug) {
568 logHeaderRx("RECV", address, port, header, raw_data.size() - NET_HEADER_SIZE);
569 }
570
571 if (header.flags & FLAG_ACK_ONLY) {
572 std::optional<Pending> to_send_next;
573
574 {
575 std::lock_guard<std::mutex> lg(_tx_mx);
576 auto it = _tx.find(connection_id);
577
578 if (it != _tx.end() && it->second.in_flight.has_value()
579 && it->second.in_flight->seq == header.ack) {
580 auto rtt_sample = std::chrono::duration_cast<std::chrono::milliseconds>(
581 std::chrono::steady_clock::now() - it->second.in_flight->lastSend);
582 if (rtt_sample <= std::chrono::milliseconds{0}) {
583 rtt_sample = std::chrono::milliseconds{1};
584 }
585
586 {
587 std::lock_guard<std::mutex> lc(_conn_mx);
588 auto c = _connections.find(connection_id);
589
590 if (c != _connections.end())
591 c->second.updateRtt(rtt_sample);
592 }
593
594 it->second.in_flight.reset();
595 if (!it->second.queue.empty()) {
596 to_send_next = std::move(it->second.queue.front());
597 it->second.queue.pop_front();
598 }
599 }
600 }
601
602 if (to_send_next.has_value()) {
603 armAndSend(connection_id, to_send_next->data, to_send_next->channel);
604 }
605 return;
606 }
607
608 if (header.flags & FLAG_RELIABLE) {
609 if (header.seq > lastRecvSeq) {
610 {
611 std::lock_guard<std::mutex> lg(_conn_mx);
612 auto it = _connections.find(connection_id);
613
614 if (it != _connections.end())
615 it->second.noteRecvSeq(header.seq);
616 }
617
618 std::vector<std::uint8_t> payload(raw_data.begin() + NET_HEADER_SIZE, raw_data.end());
619 auto handler_iterator = _handlers.find(static_cast<int>(header.channel));
620 if (handler_iterator != _handlers.end()) {
621 PacketEnvelope envelope{payload, address, port, header.channel};
622 handler_iterator->second(envelope);
623 } else {
624 if (!on_receive) {
625 throw asge::error::Error(
626 asge::error::ErrorType::Network,
627 "NetworkManager::processIncoming",
628 "onReceive callback is null",
629 asge::error::ErrorSeverity::Recoverable
630 );
631 }
632 on_receive(payload);
633 }
634 } else {
635 if (_debug) {
636 std::cerr << "[NetworkManager] duplicate seq " << header.seq << " ignored\n";
637 }
638 }
639
640 NetHeader ack_header{};
641 ack_header.seq = 0;
642 ack_header.ack = header.seq;
643 ack_header.flags = FLAG_ACK_ONLY;
644 ack_header.channel = header.channel;
645
646 std::vector<std::uint8_t> ack_packet(NET_HEADER_SIZE);
647 writeHeader(ack_packet.data(), ack_header);
648 if (!_channel->sendTo(ack_packet, address, port)) {
649 throw asge::error::Error(
650 asge::error::ErrorType::Network,
651 "NetworkManager::processIncoming",
652 "Channel::sendTo() failed when sending ACK",
653 asge::error::ErrorSeverity::Recoverable
654 );
655 }
656
657 if (_debug) {
658 logHeaderTx("ACK", address, port, ack_header, 0);
659 }
660 return;
661 }
662
663 {
664 std::vector<std::uint8_t> payload(raw_data.begin() + NET_HEADER_SIZE, raw_data.end());
665 auto handler_iterator = _handlers.find(static_cast<int>(header.channel));
666 if (handler_iterator != _handlers.end()) {
667 PacketEnvelope envelope{payload, address, port, header.channel};
668 handler_iterator->second(envelope);
669 } else {
670 if (!on_receive) {
671 throw asge::error::Error(
672 asge::error::ErrorType::Network,
673 "NetworkManager::processIncoming",
674 "onReceive callback is null",
675 asge::error::ErrorSeverity::Recoverable
676 );
677 }
678 on_receive(payload);
679 }
680 }

683void NetworkManager::pumpRetransmit()
684{
685 if (!_channel) {
686 return;
687 }
688
689 const auto now = std::chrono::steady_clock::now();
690 std::vector<Todo> to_rexmit;
691
692 {
693 std::lock_guard<std::mutex> lg(_tx_mx);
694 for (auto &kv : _tx) {
695 const auto connId = kv.first;
696 auto &tx = kv.second;
697 if (!tx.in_flight.has_value())
698 continue;
699 const auto &inf = *tx.in_flight;
700 if (now - inf.lastSend < inf.rto)
701 continue;
702 to_rexmit.push_back(Todo{connId, inf.seq});
703 }
704 }
705
706 for (const auto &todo : to_rexmit) {
707 std::vector<std::uint8_t> pkt_copy;
708 std::string addr;
709 std::uint16_t port = 0;
710 bool should_send = false;
711
712 {
713 std::lock_guard<std::mutex> lg(_tx_mx);
714 auto it = _tx.find(todo.connId);
715 if (it != _tx.end() && it->second.in_flight.has_value()) {
716 auto &inf = *it->second.in_flight;
717
718 if (inf.seq == todo.seq) {
719 pkt_copy = inf.packet;
720 addr = inf.address;
721 port = inf.port;
722
723 inf.lastSend = now;
724 auto new_rto = std::chrono::milliseconds{ static_cast<int>(inf.rto.count() * 3 / 2) };
725 if (new_rto > _rtoMax) new_rto = _rtoMax;
726 inf.rto = new_rto;
727 should_send = true;
728 }
729 }
730 }
731
732 if (should_send) {
733 if (!_channel->sendTo(pkt_copy, addr, port)) {
734 throw asge::error::Error(
735 asge::error::ErrorType::Network,
736 "NetworkManager::pumpRetransmit",
737 "Channel::sendTo() failed during retransmit",
738 asge::error::ErrorSeverity::Recoverable
739 );
740 }
741 }
742 }
