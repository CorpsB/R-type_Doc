19 * Responsibilities:
20 * - Initialize and toggle the dropdown menu state.
21 * - Dynamically position option buttons relative to the main button.
22 * - Handle user interactions, clicks, and callbacks.
23 * - Render both the main button and option buttons when expanded.
24 *
25 * @note The layout is recomputed every frame to ensure consistency with main button movement.
26 * @warning Callbacks passed through @ref addOption must remain valid during execution.

42DropMenu &DropMenu::addOption(const std::string &option, const std::function<void()> &callBack)
43{
44 Rectangle mainBounds = _mainButton.getBounds();
45
46 _options.emplace_back(
47 mainBounds.x,
48 mainBounds.y + (_options.size() + 1) * (mainBounds.height + _optionSpacing),
49 mainBounds.width,
50 mainBounds.height,
51 option
52 );
53
54 _options.back().onClickAction([this, callBack]() {
55 _isOpen = false;
56 callBack();
57 });
58
59 return *this;
60}

62void DropMenu::update()
63{
64 _mainButton.update();
65
66 Rectangle mainBounds = _mainButton.getBounds();
67
68 for (std::size_t i = 0; i < _options.size(); i++) {
69 _options[i].setPosition(
70 mainBounds.x + mainBounds.width,
71 mainBounds.y + i * mainBounds.height
72 );
73 }
74
75 if (_isOpen) {
76 for (auto &opt : _options)
77 opt.update();
78 }
79}
