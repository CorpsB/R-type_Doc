38void Button::update()
39{
40 auto &settings = rtype::client::GameSettings::getInstance();
41
42 settings.syncWithWindow();
43 const float scale = settings.getScale();
44
45 Vector2 mouse = GetMousePosition();
46
47 Rectangle scaledBounds = {
48 _bounds.x * scale,
49 _bounds.y * scale,
50 _bounds.width * scale,
51 _bounds.height * scale
52 };
53
54 _isHovered = CheckCollisionPointRec(mouse, scaledBounds);
55 _isPressed = _isHovered && IsMouseButtonDown(MOUSE_BUTTON_LEFT);
56
57 if (_isHovered && IsMouseButtonReleased(MOUSE_BUTTON_LEFT) && _onClick)
58 _onClick();
59}
