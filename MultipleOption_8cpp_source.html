29void MultipleOption::update(bool selected, bool)
30{
31 auto &settings = GameSettings::getInstance();
32
33 const float scale = settings.getScale();
34
35 _realSize = {_size.x * scale, _size.y * scale};
36 _realPos = {(_position.x - _size.x / 2.0f) * scale, (_position.y - _size.y / 2.0f) * scale};
37 if (!selected)
38 return;
39 if (IsKeyPressed(KEY_LEFT)) {
40 if (_choiceId <= 0) _choices.size() == 0 ? _choiceId = 0 : _choiceId = _choices.size() - 1;
41 else _choiceId -= 1;
42 if (!_choices.empty() && _choices[_choiceId].callBack) _choices[_choiceId].callBack();
43 }
44 if (IsKeyPressed(KEY_RIGHT)) {
45 if (_choiceId >= _choices.size() - 1) _choiceId = 0;
46 else _choiceId += 1;
47 if (!_choices.empty() && _choices[_choiceId].callBack) _choices[_choiceId].callBack();
48 }
49}

79 Vector2 textSize = MeasureTextEx(font, text.c_str(), fontSize, 1);
80 Vector2 textPos = {_realPos.x, _realPos.y + (_realSize.y / 2.0f) - (textSize.y / 2.0f)};
81
82 DrawRectangle(_realPos.x, _realPos.y, _realSize.x, _realSize.y, boxCol);
83 DrawTextEx(font, text.c_str(), textPos, fontSize, 1, textCol);
84
85 if (!_choices.empty()) {
86 const std::string choiceText = "< " + langManager.getTranslation(_choices[_choiceId].textKey) + " >";
87 Vector2 choiceSize = MeasureTextEx(font, choiceText.c_str(), fontSize, 1);
88 Vector2 choicePos = {_realPos.x + _realSize.x - choiceSize.x, _realPos.y + (_realSize.y / 2.0f) - (choiceSize.y / 2.0f)};
89
90 DrawTextEx(font, choiceText.c_str(), choicePos, fontSize, 1, textCol);
