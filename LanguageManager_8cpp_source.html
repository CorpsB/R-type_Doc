42void LanguageManager::loadFile(const std::string &path)
43{
44 std::ifstream file(path);
45
46 if (!file.is_open())
47 throw rtype::exception::Error("LanguageManager", "Failed to load Assets/lang/lang.json");
48 json j;
49
50 file >> j;
51 _translations.clear();
52 for (auto &[key, langs] : j.items()) {
53 LangMap langMap;
54
55 for (auto &[lang, text] : langs.items()) {
56 if (text.is_string())
57 langMap[lang] = text.get<std::string>();
58 }
59 _translations[key] = std::move(langMap);
60 }
61}

63const std::string &LanguageManager::getTranslation(const std::string &key) const
64{
65 static const std::string missingKey = "[missing key]";
66 static const std::string missingTranslation = "[missing translation]";
67 auto &settings = GameSettings::getInstance();
68 auto keyIt = _translations.find(key);
69
70 if (keyIt == _translations.end())
71 return missingKey;
72 const auto &langMap = keyIt->second;
73 auto langIt = langMap.find(settings.getLangTag());
74
75 if (langIt == langMap.end())
76 return missingTranslation;
77 return langIt->second;
78}
