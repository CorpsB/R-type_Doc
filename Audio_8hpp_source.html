31/**
32 * @struct audio_attr_t
33 * @brief ECS component that stores sound playback state and resource key.
34 *
35 * @details
36 * The audio_attr_t component defines the minimal state required to control
37 * sound behavior for an entity. It is consumed by AudioSystem each frame
38 * to determine whether to start, stop, or loop a sound.
39 *
40 * ### Members:
41 * | Field | Type | Description |
42 * |------------|-------------|-------------|
43 * | soundKey | std::string | Key referencing a loaded Sound in ResourceManager |
44 * | play | bool | Set true to trigger playback (reset after use) |
45 * | loop | bool | Set true for continuous playback loop |
46 * | stop | bool | Set true to immediately stop playback |
47 * | playing | bool | True if the sound is currently active |
48 *
49 * @note The component does not store a Raylib Sound object directly,
50 * only a lookup key to ensure centralized resource management.
51 */
52struct audio_attr_t {
53 std::string soundKey; ///< Resource key referencing a Sound in ResourceManager.
54 bool play = false; ///< If true, play the sound once and reset to false.
55 bool loop = false; ///< If true, continuously loop the sound.
56 bool stop = false; ///< If true, stop the sound immediately.
57 bool playing = false; ///< True while the sound is currently playing.
58};
