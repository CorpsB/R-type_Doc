17 * @details
18 * This file provides the implementation of the @ref PluginHandle::unload method.
19 * It ensures that plugin instances are safely released while handling potential
20 * exceptions raised by the plugin or runtime environment.
21 *
22 * The unloading process:
23 * - Verifies that the plugin instance is valid.
24 * - Calls @ref IPlugin::unloadAll to release internal resources.
25 * - Marks the handle as no longer loaded.
26 * - Wraps any failure in a recoverable @ref asge::error::Error.
27 */

31void PluginHandle::unload()
32{
33 try {
34 if (!_plugin_instance) {
35 throw asge::error::Error(
36 asge::error::ErrorType::Plugin,
37 "PluginHandle::unload",
38 "Plugin instance is null",

40 );
41 }
42
43 _plugin_instance->unloadAll();
44 _is_loaded = false;
45 } catch (const std::exception &e) {

47 asge::error::ErrorType::Plugin,
48 "PluginHandle::unload",
49 std::string("Failed to unload plugin: ") + e.what(),
50 asge::error::ErrorSeverity::Recoverable

52 }
53}
