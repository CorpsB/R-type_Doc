45/**
46 * @class PlayerSystem
47 * @brief System that handles player input logic within the ECS.
48 *
49 * @details
50 * PlayerSystem manipulates movement, firing, and score for entities representing
51 * players. It retrieves relevant components through the Coordinator and applies
52 * per-frame updates according to the input type.
53 *
54 * ### Component requirements:
55 * - position_attr_t
56 * - direction_attr_t
57 * - speed_attr_t
58 * - score_attr_t
59 * - frequency_attr_t
60 *
61 * Complexity: O(n) per call where n is the number of controlled player entities.
62 */
63class PlayerSystem : public System {
64 public:

66 virtual ~PlayerSystem() = default;

68 /**
69 * @brief Apply player movement based on input direction.
70 *
71 * @param coord Reference to Coordinator for component access.
72 * @param ticks Number of ticks to apply (movement scaling factor).
73 * @param input Input command received from the player.
74 *
75 * @details
76 * The function updates the entity’s direction based on input and computes
77 * position deltas using speed and direction attributes. Missing components
78 * cause the entity to be skipped safely.
79 *
80 * @note Movement is deterministic and frame-rate independent.
81 */
82 void move(Coordinator &coord, float ticks, protocol::game::player_input_t input);

94 void updateScore(Coordinator &coord, int scoreToAdd);
