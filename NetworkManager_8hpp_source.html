32/**
33 * @struct NetHeader
34 * @brief On-wire header (12 bytes) prepended before each payload.
35 * @ingroup network
36 *
37 * @details
38 * Binary layout (little-endian):
39 * - seq (uint32) : sender sequence (reliable channel only).
40 * - ack (uint32) : last received/accepted sequence (ACK-only possible).
41 * - flags (uint8) : bitfield (FLAG_ACK_ONLY, FLAG_RELIABLE, ...).
42 * - channel (uint8) : 0 = reliable; >0 = best-effort channels.
43 * - reserved(uint16) : reserved for future extensions / alignment.
44 */
45struct NetHeader {
46 std::uint32_t seq{0};
47 std::uint32_t ack{0};
48 std::uint8_t flags{0};
49 std::uint8_t channel{0};
50 std::uint16_t reserved{0};
51};
52
53/** @brief Size in bytes of the serialized @ref NetHeader. */
54constexpr std::size_t NET_HEADER_SIZE = 12;
55/** @brief Flag bit: header carries ACK-only (no payload). */
56constexpr std::uint8_t FLAG_ACK_ONLY = 0x01;
57/** @brief Flag bit: packet is reliable (expects ACK/RTX). */
58constexpr std::uint8_t FLAG_RELIABLE = 0x02;
59
60// Handshake message types
61static constexpr std::uint8_t MSG_CONNECT_REQUEST = 0x10;
62static constexpr std::uint8_t MSG_CONNECT_ACCEPT = 0x11;

65 * @struct PacketEnvelope
66 * @brief Delivery context used by handlers (payload + source + channel).
67 * @ingroup network
68 *
69 * @details
70 * Passed to channel-specific handlers registered via @ref setHandler.
71 * `fromAddr` and `fromPort` identify the peer that originated the packet.

73struct PacketEnvelope {

75 std::string fromAddr;

82/**
83 * @class NetworkManager
84 * @brief High-level network service with reliability, sessions, and dispatch.
85 * @ingroup network
86 *
87 * @details
88 * Responsibilities:
89 * - Owns a @ref NetworkChannel (transport).
90 * - Manages per-peer @ref Connection objects (seq/RTT/activity).
91 * - Encodes/decodes @ref NetHeader and enforces MTU/flags.
92 * - Reliable sending on channel 0 (one in-flight per connection).
93 * - Best-effort for channels > 0.
94 * - ACK-only handling and periodic retransmission via RTO backoff.
95 *
96 * Lifecycle:
97 * 1. Construct with a transport channel.
98 * 2. startServer() or connectTo().
99 * 3. send(), poll() in the main loop.
