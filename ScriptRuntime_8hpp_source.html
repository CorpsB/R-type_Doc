28/**
29 * @class ScriptRuntime
30 * @brief Abstract interface for script runtimes.
31 *
32 * @details
33 * Implementations parse templates, convert payload strings into runtime values
34 * and resolve template inheritance by merging ScriptHandles. Concrete runtimes
35 * must ensure returned ScriptHandles are valid and throw or return std::nullopt
36 * on failure as appropriate.
37 */
38class ScriptRuntime {
39 public:
40 virtual ~ScriptRuntime() = default;

42 /**
43 * @brief Parse a script/template from a raw string.
44 *
45 * @param path Logical path the content was loaded from (used for diagnostics).
46 * @param content Raw file content to be parsed by the runtime.
47 * @returns Optional ScriptHandle containing parsed metadata and behavior
48 * on success, std::nullopt on parse/validation failure.
49 *
50 * @details
51 * Implementations should return std::nullopt if parsing fails. The returned
52 * ScriptHandle is expected to be self-contained (id, components, behavior).
53 */
54 virtual std::optional<ScriptHandle> parseString(const std::string &path,
55 const std::string &content) = 0;

57 /**
58 * @brief Parse a runtime payload (e.g. JSON string) into a ScriptValue.

60 * @param payload Raw payload string provided to host API handlers.

62 * if parsing failed.
63 *
64 * @details
65 * Used to convert event payloads or API arguments into a runtime-understood
66 * representation before invoking host functions.
67 */
68 virtual std::optional<ScriptValue> parsePayload(const std::string &payload) = 0;

70 /**
71 * @brief Merge a parent and child ScriptHandle resolving "extends".
72 *
73 * @param parent Parent (base) handle that provides defaults.
74 * @param child Child handle that overrides or extends the parent.
75 * @returns A merged ScriptHandle representing the resolved template.
76 *
77 * @details
78 * Implementations should combine components and behavior deterministically.
