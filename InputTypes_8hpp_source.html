76/**
77 * @struct PointerState
78 * @brief State of a single pointer/touch (mouse or touch).
79 */
80struct PointerState {
81 float x = 0.0f; ///< X position in pixels (or logical coordinates)
82 float y = 0.0f; ///< Y position
83 bool pressed = false; ///< true if primary button is pressed
84 int id = 0; ///< pointer id (touch index), 0 for mouse

87/**
88 * @struct DeviceState
89 * @brief Raw per-device input state (buttons, axes, pointers).
90 *
91 * @details
92 * Each InputDevice produces one DeviceState per frame. These are
93 * merged by InputManager into an InputSnapshot.
94 */
95struct DeviceState {
96 /// @brief Keys that were pressed (edge detection).
97 std::unordered_set<int> pressed;
98
99 /// @brief Keys currently held down.
100 std::unordered_set<int> down;
101
102 /// @brief Keys that were released this frame.
103 std::unordered_set<int> released;
104
105 /// @brief Named analog axes (e.g. joystick X/Y, triggers).
106 std::unordered_map<std::string, float> axes;
107
108 /// @brief Pointer devices (e.g. mouse position, touch coordinates).
109 /// pair<float,float> = (x, y)
110 std::unordered_map<std::string, std::pair<float, float>> pointers;
111
112 /// @brief Reset all transient data.
113 void clear()
114 {
115 pressed.clear();
116 released.clear();
117 down.clear();
118 axes.clear();
119 pointers.clear();
120 }
121};
122
123/**
124 * @struct InputSnapshot
125 * @brief Aggregated snapshot composed by InputManager at the end of update().
126 *
127 * @details
128 * Combines states from multiple devices into one read-only snapshot used by systems.
129 */
130struct InputSnapshot {
131 std::unordered_set<int> pressed; ///< raw scancodes pressed this frame
132 std::unordered_set<int> down; ///< raw scancodes currently down
133 std::unordered_set<int> released; ///< raw scancodes released this frame
134 std::unordered_map<std::string, float> axes; ///< logical axis names to values
135 std::vector<PointerState> pointers; ///< aggregated pointers
136};
