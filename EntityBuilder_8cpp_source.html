26 /**
27 * @brief Adds a component to an entity if the optional payload is present.
28 *
29 * @tparam ComponentType Component POD/type to be attached.
30 * @param services Engine services (used to reach the ECS world and debug).
31 * @param entityId Target entity identifier.
32 * @param component Optional component payload; if empty, nothing is added.
33 * @param componentName Human-readable component name for log messages.
34 *
35 * @details
36 * This helper is intentionally silent on success. When insertion fails, a
37 * warning is issued via the debug service (if available). No exception is
38 * thrown, allowing the builder to produce partial entities.
39 */
40 template <typename ComponentType>
41 void addComponentIfPresent(
42 asge::Services &services,
43 std::uint32_t entityId,
44 const std::optional<ComponentType> &component,
45 const std::string &componentName)
46 {
47 if (!component.has_value()) {
48 return;
49 }
50
51 if (!services.world().addComponent<ComponentType>(entityId, *component)) {
52 if (services._debug != nullptr) {
53 services.debug().log(
54 asge::debug::Level::Warning,
55 "EntityBuilder::build: addComponent<" + componentName + "> failed.");
56 }
57 }
58 }
59}

174uint32_t asge::script::EntityFactory::EntityBuilder::build(
175 const std::optional<uint32_t> &entity_id)
176{
177 if (_isBuilt) {
178 return _entityId;
179 }
180
181 if (!_services._world) {
182 if (_services._debug) {
183 _services.debug().log(
184 asge::debug::Level::Warning,
185 "EntityBuilder::build: world service missing.");
186 }
187 return asge::world::INVALID_ID;
188 }
189
190 try {
191 if (entity_id.has_value()) {
192 _services.world().createEntity(entity_id.value());
193 _entityId = entity_id.value();
194 } else {
195 _entityId = _services.world().createEntity();
196 }
197 }
198 catch (...) {
199 if (_services._debug) {
200 _services.debug().log(
201 asge::debug::Level::Warning,
202 "EntityBuilder::build: world.createEntity threw.");
203 }
204 return asge::world::INVALID_ID;
205 }
206
207 try {
208 components::template_attr_t templateComponent {};
209 templateComponent.template_id = _templateId;
210 if (!_services.world().addComponent<components::template_attr_t>(_entityId, templateComponent)) {
211 if (_services._debug) {
212 _services.debug().log(
213 asge::debug::Level::Warning,
214 "EntityBuilder::build: addComponent<template_attr_t> failed.");
215 }
216 }
217
218 addComponentIfPresent(_services, _entityId, _name, "name_attr_t");
219 addComponentIfPresent(_services, _entityId, _hitbox, "hitbox_attr_t");
220 addComponentIfPresent(_services, _entityId, _sprite, "sprite_attr_t");
221 addComponentIfPresent(_services, _entityId, _speed, "speed_attr_t");
222 addComponentIfPresent(_services, _entityId, _position, "position_attr_t");
223 addComponentIfPresent(_services, _entityId, _direction, "direction_attr_t");
224 addComponentIfPresent(_services, _entityId, _health, "health_attr_t");
225 addComponentIfPresent(_services, _entityId, _damage, "damage_attr_t");
226 addComponentIfPresent(_services, _entityId, _frequency, "frequency_attr_t");
227 }
228 catch (...) {
229 if (_services._debug) {
230 _services.debug().log(
231 asge::debug::Level::Warning,
232 "EntityBuilder::build: exception while adding components for eid=" + std::to_string(_entityId));
233 }
234 }
235
236 if (!_templateId.empty() && _templateId == "player") {
237 components::score_attr_t scoreAttributes {};
238 scoreAttributes.max_score = 0;
239 scoreAttributes.score = 0;
240
241 if (!_services.world().addComponent<components::score_attr_t>(_entityId, scoreAttributes)) {
242 if (_services._debug) {
243 _services.debug().log(
244 asge::debug::Level::Warning,
245 "EntityBuilder::build: addComponent<score_attr_t> failed.");
246 }
247 }
248
249 components::input_attr_t inputAttributes {};
250 inputAttributes.fire = false;
251 inputAttributes.pad = components::PLAYER_PAD::NO_PAD;
252
253 if (!_services.world().addComponent<components::input_attr_t>(_entityId, inputAttributes)) {
254 if (_services._debug) {
255 _services.debug().log(
256 asge::debug::Level::Warning,
257 "EntityBuilder::build: addComponent<input_attr_t> failed.");
258 }
259 }
260 }
261
262 try {
263 if (!_templateId.empty() && _services._scripts) {
264 _services.scripts().notifyEntityCreated(_entityId, _templateId);
265 }
266 }
267 catch (...) {
268 if (_services._debug) {
269 _services.debug().log(
270 asge::debug::Level::Warning,
271 "EntityBuilder::build: scripts.notifyEntityCreated threw for eid=" + std::to_string(_entityId));
272 }
273 }
274
275 _isBuilt = true;
276 return _entityId;
277}
