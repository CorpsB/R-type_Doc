12 * @ingroup client-scenes
13 * @see SceneManager.hpp, IScene

15 * @details
16 * This file implements the scene management system responsible for handling
17 * scene transitions (push, pop, replace) through a deferred stack-based model.
18 *
19 * Responsibilities:
20 * - Maintain a stack of active scenes.
21 * - Safely handle scene transitions between updates and draws.
22 * - Ensure lifecycle callbacks (@ref IScene::onEnter, @ref IScene::onExit, etc.)
23 * are properly executed during transitions.
24 *
25 * Deferred operations prevent invalid state changes during active frame updates.
26 * Transitions are committed through @ref flushTransitions() at controlled points
27 * in the frame loop.
28 */

63void SceneManager::flushTransitions()
64{
65 for (auto &op : _pending) {
66 switch (op.type) {
67 case OpType::Push: {
68 _stack.push_back(std::move(op.scene));
69 } break;
70 case OpType::Pop: {
71 if (!_stack.empty())
72 _stack.pop_back();
73 } break;
74 case OpType::Replace: {
75 if (!_stack.empty())
76 _stack.pop_back();
77 _stack.push_back(std::move(op.scene));
78 } break;
79 default:
80 break;
81 }
82 }
83 _pending.clear();
84}
