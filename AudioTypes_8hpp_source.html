26/**
27 * @struct PlayOptions
28 * @brief Runtime playback preferences for a sound or music track.
29 *
30 * @details
31 * Encapsulates volume, looping and pitch parameters that the audio backend
32 * interprets when starting playback of a resource.
33 */
34struct PlayOptions {
35 float volume = 1.0f; ///< Normalized volume modifier (0.0 = silent, 1.0 = default).
36 bool loop = false; ///< Set to true to keep the track looping indefinitely.
37 float pitch = 1.0f; ///< Pitch multiplier applied during playback.
38};

40/**
41 * @struct SongEntry
42 * @brief Metadata describing a loaded audio asset.
43 *
44 * @details
45 * Stores the resource path, native backend handle, last access timestamp and
46 * default playback options. Audio loaders populate this structure when caching
47 * songs in memory.
48 */
49struct SongEntry {
50 std::string path; ///< Logical or filesystem path for the resource.
51 bool loaded = false; ///< True when the audio data is fully loaded.
52 std::uintptr_t native_handle = 0; ///< Backend-specific handle (opaque to clients).
53 std::uint64_t last_access = 0; ///< Timestamp used for cache eviction.
54 PlayOptions options; ///< Default playback configuration for this entry.
55};
