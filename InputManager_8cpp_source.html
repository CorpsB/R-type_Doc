37InputManager::InputManager([[maybe_unused]] asge::Services &services,
38 std::vector<std::unique_ptr<InputDevice>> initial_devices)
39{
40 try {
41 if (!initial_devices.empty()) {
42 std::lock_guard<std::mutex> devices_lock(_devices_mtx);
43 for (auto &device : initial_devices) {
44 if (!device) {
45 continue;
46 }
47 _devices.emplace_back(std::move(device));
48 _next_device_id.fetch_add(1, std::memory_order_relaxed);
49 }
50 }
51
52 std::unique_lock<std::shared_mutex> snapshot_lock(_snapshot_mtx);
53 _front_snapshot = {};
54 _back_snapshot = {};
55 _frame_seq.store(0, std::memory_order_relaxed);
56 } catch (...) {
57 throw;
58 }
59}

273/**
274 * @copydoc InputManager::composeSnapshotFromDevices()
275 */
276void InputManager::composeSnapshotFromDevices()
277{
278 std::unique_lock<std::shared_mutex> snapshot_lock(_snapshot_mtx);
279
280 _back_snapshot.pressed.clear();
281 _back_snapshot.released.clear();
282 _back_snapshot.down.clear();
283 _back_snapshot.axes.clear();
284 _back_snapshot.pointers.clear();
285
286 std::lock_guard<std::mutex> devices_lock(_devices_mtx);
287 for (const auto &device : _devices) {
288 if (!device || !device->enabled()) {
289 continue;
290 }
291
292 const DeviceState state = device->snapshot();
293
294 _back_snapshot.pressed.insert(state.pressed.begin(), state.pressed.end());
295 _back_snapshot.released.insert(state.released.begin(), state.released.end());
296 _back_snapshot.down.insert(state.down.begin(), state.down.end());
297
298 for (const auto &[axis_name, axis_value] : state.axes) {
299 _back_snapshot.axes[axis_name] = axis_value;
300 }
301
302 for (const auto &[pointer_name, pointer_position] : state.pointers) {
303 PointerState pointer_state{};
304 pointer_state.x = pointer_position.first;
305 pointer_state.y = pointer_position.second;
306 pointer_state.pressed = false;
307 pointer_state.id = 0;
308 _back_snapshot.pointers.emplace_back(pointer_state);
309 }
310 }

331/**
332 * @copydoc InputManager::updatePlayerInputs(asge::Services &)
333 */
334void InputManager::updatePlayerInputs(asge::Services &services)
335{
336 try {
337 auto *input_component =
338 services.world().getComponent<components::input_attr_t>(services.player_id);
339
340 if (!input_component) {
341 return;
342 }
343
344 // Capture local keyboard state into the input component
345 input_component->fire = isPressed(input::KeyCode::SPACE);
346
347 const bool up_pressed = isDown(input::KeyCode::Z);
348 const bool left_pressed = isDown(input::KeyCode::Q);
349 const bool right_pressed = isDown(input::KeyCode::D);
350 const bool down_pressed = isDown(input::KeyCode::S);
351
352 // Resolve the directional pad using the same priority as before
353 if (up_pressed) {
354 if (left_pressed) {
355 input_component->pad = components::PLAYER_PAD::UP_LEFT;
356 } else if (right_pressed) {
357 input_component->pad = components::PLAYER_PAD::UP_RIGHT;
358 } else {
359 input_component->pad = components::PLAYER_PAD::UP;
360 }
361 } else if (down_pressed) {
362 if (left_pressed) {
363 input_component->pad = components::PLAYER_PAD::DOWN_LEFT;
364 } else if (right_pressed) {
365 input_component->pad = components::PLAYER_PAD::DOWN_RIGHT;
366 } else {
367 input_component->pad = components::PLAYER_PAD::DOWN;
368 }
369 } else if (left_pressed) {
370 input_component->pad = components::PLAYER_PAD::LEFT;
371 } else if (right_pressed) {
372 input_component->pad = components::PLAYER_PAD::RIGHT;
373 } else {
374 input_component->pad = components::PLAYER_PAD::NO_PAD;
375 }
376
377 // When running as a client, replicate the local input state to the server.
378 // We rely on the services.player_id to identify the player's entity in the
379 // authoritative world. Only send when networking is enabled and a valid
380 // player id has been assigned.
381 if (services.config().runMode == asge::EngineRunMode::Client &&
382 services.config().enableNetwork) {
383 try {
384 asge::net::Packet pkt;
385 services.protocol().encodeInputs(
386 services.player_id,
387 input_component->pad,
388 input_component->fire,
389 pkt,
390 /*channel=*/0,
391 /*flags=*/0);
392 services.network().send(pkt, 0);
393 }
394 catch (const std::exception &e) {
395 if (services._debug) {
396 services.debug().log(asge::debug::Level::Warning,
397 std::string("[InputManager] failed to send input packet: ") + e.what());
398 }
399 }
400 catch (...) {
401 if (services._debug) {
402 services.debug().log(asge::debug::Level::Warning,
403 "[InputManager] failed to send input packet: unknown error");
404 }
405 }
406 }
407 } catch (...) {
408 if (services._debug) {
409 services.debug().log(debug::Level::Error,
410 "Failure to retrieve and assign inputs to the player.");
411 }
412 }
413}
