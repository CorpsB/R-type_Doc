41/**
42 * @class Sprite
43 * @brief Represents a 2D textured sprite for UI rendering.
44 *
45 * @details
46 * The Sprite class encapsulates the logic and rendering of a texture-based UI component.
47 * It stores both logical and scaled (“real”) position and size to adapt to the window scale dynamically.
48 *
49 * @see rtype::client::Resources::ResourceManager, rtype::client::Settings::GameSettings
50 */
51class Sprite : public UIElement {
52 public:
53 /**
54 * @brief Constructs a Sprite object.
55 * @param position The logical position of the sprite.
56 * @param size The logical size of the sprite.
57 * @param textureTag The key used to fetch the sprite’s texture from the ResourceManager.
58 */
59 explicit Sprite(const std::string &textureTag) : _textureTag(textureTag), _textureRect(Rectangle{0, 0, 0, 0}) { _instances.reserve(10); };
60
61 void setRect(Rectangle rect) { _textureRect = rect; };
62
63 /**
64 * @brief Adds a new instance (logical position and size) to be rendered.
65 * @param position The logical position of the new instance.
66 * @param size The logical size of the new instance.
67 */
68 void addInstance(Vector2 position, Vector2 size) { _instances.push_back({position, size, position, size}); };
69
70 /**
71 * @brief Clears all sprite instances.
72 */
73 void clearInstances() { _instances.clear(); };
74
75 /**
76 * @brief Updates the sprite’s real position and size according to the game’s scaling settings.
77 * @details
78 * Retrieves the current scale factor from the GameSettings instance and recalculates
79 * the sprite’s display coordinates and dimensions.
80 */
81 void update() override;
82
83 /**
84 * @brief Draws the sprite on the screen.
85 * @details
86 * Fetches the texture from the ResourceManager and renders it using Raylib’s `DrawTexturePro` function.
87 * If the texture is invalid, an error message is printed in debug output.
88 */
89 void draw() const override;
90
91 private:
92 std::string _textureTag; ///< Identifier key for the associated texture.
93 Rectangle _textureRect;
94
95 std::vector<SpriteInstance> _instances; ///< List of logical sprite instances.
