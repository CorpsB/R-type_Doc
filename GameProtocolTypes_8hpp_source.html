27/**
28 * @namespace protocol::game
29 * @brief Game-level message IDs and payload definitions built on top of the core protocol.
30 *
31 * @details
32 * The `protocol::game` namespace defines the application-level message identifiers
33 * and the wire payload layouts used by the game facade. These types map directly
34 * onto the low-level `protocol::packet_header_t::message_type` field and provide
35 * strongly-typed payload structures (create/move/destroy, player input, snapshots).
36 *
37 * Use these types when encoding and decoding game messages; the game layer is
38 * responsible for packing the payload into a `protocol::packet_t` and setting
39 * the corresponding `message_type` value before sending.
40 *
41 * @since 0.1
42 * @ingroup protocol::game
43 */
44namespace protocol::game {

138/**
139 * @brief Variant payload carried by a game-level event.
140 *
141 * @details
142 * The variant holds one of:
143 * - std::monostate: no payload (NONE)
144 * - payload_create_entity_t: for CREATE_ENTITY events
145 * - payload_move_entity_t: for MOVE_ENTITY events
146 * - payload_destroy_entity_t: for DESTROY_ENTITY events
147 * - payload_player_input_t: for PLAYER_INPUT events
148 * - protocol::packet_t: for RAW_PACKET events (raw core packet)
149 * - protocol::transmit_failed_t: for TRANSMIT_FAILED events forwarded from core
150 * - protocol::udp::endpoint: for CONNECTED / DISCONNECTED events, carrying the peer endpoint
151 */
152using game_event_payload_t = std::variant<
153 std::monostate,
154 payload_create_entity_t,
155 payload_move_entity_t,
156 payload_destroy_entity_t,
157 payload_player_input_t,
158 protocol::packet_t,
159 protocol::transmit_failed_t,
160 protocol::udp::endpoint
161>;
162
163/**
164 * @struct game_event_t
165 * @brief Top-level game event exposed by GameProtocol.
166 *
167 * @details
168 * Instances of this struct are produced by the GameProtocol facade and delivered
169 * to the application via is_event / wait_event. The `type` field describes the
170 * logical event kind and the `payload` variant carries the strongly-typed data
171 * for that event (or std::monostate when there is no payload).
172 */
173struct game_event_t {
174 game_event_type_t type = game_event_type_t::NONE; ///< logical event kind
175 game_event_payload_t payload{}; ///< variant carrying event data
176};
