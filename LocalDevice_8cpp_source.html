44std::vector<uint8_t> asge::storage::LocalDevice::read(const std::string &path)
45{
46 std::ifstream file(path, std::ios::binary | std::ios::ate);
47 if (!file) {
48 if (_debugger != nullptr) {
49 _debugger->log(
50 debug::Level::Error,
51 "LocalDevice::read(): cannot open " + path);
52 }
53 return {};
54 }
55
56 std::streamoff fileSize = file.tellg();
57 if (fileSize <= 0) {
58 if (_debugger != nullptr) {
59 _debugger->log(
60 debug::Level::Warning,
61 "LocalDevice::read(): empty or invalid file " + path);
62 }
63 return {};
64 }
65
66 if (static_cast<size_t>(fileSize) > std::numeric_limits<std::size_t>::max()) {
67 if (_debugger != nullptr) {
68 _debugger->log(
69 debug::Level::Error,
70 "LocalDevice::read(): file too large " + path);
71 }
72 return {};
73 }
74
75 file.seekg(0, std::ios::beg);
76
77 std::vector<uint8_t> buffer(static_cast<std::size_t>(fileSize));
78 if (!file.read(reinterpret_cast<char *>(buffer.data()), fileSize)) {
79 std::streamsize readCount = file.gcount();
80
81 if (readCount != fileSize) {
82 if (_debugger != nullptr) {
83 _debugger->log(
84 debug::Level::Error,
85 "LocalDevice::read(): read failed or incomplete " + path);
86 }
87 return {};
88 }
89 }
90
91 return buffer;
92}

94bool asge::storage::LocalDevice::write(const std::string &path, const std::vector<uint8_t> &data)
95{
96 std::ofstream file(path, std::ios::binary | std::ios::trunc);
97
98 if (!file.is_open()) {
99 if (_debugger != nullptr) {
100 _debugger->log(
101 debug::Level::Error,
102 "LocalDevice::write(): write failed " + path);
103 }
104 return false;
105 }
106 file.write(reinterpret_cast<const char *>(data.data()), static_cast<std::streamsize>(data.size()));
107 if (!file.good()) {
108 if (_debugger != nullptr) {
109 _debugger->log(
110 debug::Level::Error,
111 "LocalDevice::write(): write failed " + path);
112 }
113 return false;
114 }
115 file.flush();
116 return true;
117}
