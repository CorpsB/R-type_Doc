115void UDPChannel::poll(std::function<void(const std::vector<std::uint8_t>&, const std::string&, std::uint16_t)> onPacket) {
116 for (;;) {
117 std::vector<std::uint8_t> buf(2048);
118 udp::endpoint from;
119 boost::system::error_code ec;
120
121 std::size_t n = _socket.receive_from(boost::asio::buffer(buf), from, 0, ec);
122 if (ec == boost::asio::error::would_block || ec == boost::asio::error::try_again) {
123 break;
124 }
125 if (ec) {
126 if (_debug)
127 std::cerr << "[UDPChannel] recv error: " << ec.message() << "\n";
128 break;
129 }
130 buf.resize(n);
131 onPacket(buf, from.address().to_string(), from.port());
132 }
133}
