164/**
165 * @brief Evicts the least recently used (LRU) loaded entry to respect cache size.
166 */
167static void evict_lru(std::unordered_map<std::string, asge::audio::SongEntry> &entries,
168 asge::audio::AudioDriver *driver)
169{
170 std::string evict_id;
171 std::uint64_t min_access = UINT64_MAX;
172 for (auto &kv : entries) {
173 if (!kv.second.loaded)
174 continue;
175 if (kv.second.last_access < min_access) {
176 min_access = kv.second.last_access;
177 evict_id = kv.first;
178 }
179 }
180
181 if (!evict_id.empty()) {
182 entries[evict_id].loaded = false;
183 entries[evict_id].native_handle = 0;
184 if (driver)
185 driver->destroyEntry(evict_id);
186 }
187}

189/**
190 * @brief Extracts a file extension from a path and returns it in uppercase.
191 */
192static std::string extension_to_file_type(const std::string &path)
193{
194 auto pos = path.find_last_of('.');
195
196 if (pos == std::string::npos)
197 return std::string();
198 std::string ext = path.substr(pos + 1);
199 for (auto &c : ext)
200 c = static_cast<char>(std::toupper(static_cast<unsigned char>(c)));
201 return ext;
202}

204bool asge::audio::AudioManager::createEntry(const std::string &id,
205 const std::string &path)
206{
207 try {
208 std::lock_guard<std::mutex> lk(_mtx);
209 auto it = _entries.find(id);
210 if (it != _entries.end()) {
211 it->second.path = path;
212 it->second.loaded = false;
213 it->second.native_handle = 0;
214 it->second.last_access = 0;
215 it->second.options = PlayOptions{};
216 if (_driver)
217 _driver->createEntry(id, path);
218 if (_dbg)
219 _dbg->log(debug::Level::Info, "AudioManager::createEntry overwritten id " + id);
220 return true;
221 }
222
223 SongEntry entry;
224 entry.path = path;
225 entry.loaded = false;
226 entry.native_handle = 0;
227 entry.last_access = 0;
228 entry.options = PlayOptions{};
229 _entries.emplace(id, std::move(entry));
230
231 if (_driver)
232 _driver->createEntry(id, path);
233 if (_dbg)
234 _dbg->log(debug::Level::Info, "AudioManager::createEntry created id " + id);
235 return true;
236 }
237 catch (const std::exception &e) {
238 if (_dbg)
239 _dbg->log(debug::Level::Error,
240 std::string("AudioManager::createEntry std::exception: ") + e.what());
241 throw error::Error(error::ErrorType::Audio, "AudioManager::createEntry",
242 std::string("std::exception: ") + e.what(), error::ErrorSeverity::Recoverable);
243 }

314void asge::audio::AudioManager::playSong(const std::string &id,
315 storage::IStorage &storage, const std::optional<PlayOptions> &opts,
316 const std::optional<std::string> &path)
317{
318 try {
319 std::lock_guard<std::mutex> lk(_mtx);
320
321 auto it = _entries.find(id);
322 if (it == _entries.end()) {
323 if (!path.has_value()) {
324 if (_dbg)
325 _dbg->log(debug::Level::Warning,
326 "AudioManager::playSong no entry for id " + id +
327 ", and yet no path given.");
328 return;
329 }
330
331 SongEntry e;
332 e.path = path.value();
333 e.loaded = false;
334 e.native_handle = 0;
335 e.last_access = 0;
336 e.options = PlayOptions{};
337 it = _entries.emplace(id, std::move(e)).first;
338
339 if (_driver)
340 _driver->createEntry(id, path.value());
341 }
342
343 SongEntry &entry = it->second;
344
345 if (!entry.loaded) {
346 std::vector<std::uint8_t> bytes = storage.readFile(entry.path);
347 if (bytes.empty()) {
348 if (_dbg)
349 _dbg->log(debug::Level::Warning,
350 "AudioManager::playSong read empty bytes for id " + id);
351 throw error::Error(error::ErrorType::Storage, "AudioManager::playSong",
352 "read empty file", error::ErrorSeverity::Recoverable);
353 }
354
355 bool loaded_by_driver = false;
356 if (_driver) {
357 if (auto *rd = dynamic_cast<asge::audio::RaylibDriver *>(_driver.get())) {
358 std::string file_type = extension_to_file_type(entry.path);
359 if (file_type.empty())
360 file_type = "WAV";
361 loaded_by_driver = rd->loadEntryFromMemory(id, bytes, file_type);
362 } else {
363 _driver->createEntry(id, entry.path);
364 loaded_by_driver = true;
365 }
366 }
367
368 if (!loaded_by_driver)
369 throw error::Error(error::ErrorType::Audio, "AudioManager::playSong",
370 "driver failed to load memory", error::ErrorSeverity::Recoverable);
371
372 entry.loaded = true;
373 entry.last_access = _access_counter++;
374
375 while (count_loaded(_entries) > MAX_CACHE_SOUNDS)
376 evict_lru(_entries, _driver.get());
377 } else {
378 entry.last_access = _access_counter++;
379 }
380
381 if (opts)
382 entry.options = *opts;
383
384 if (_driver) {
385 _driver->setSongOptions(id, entry.options);
386 if (!_driver->playSong(id, entry.options))
387 throw error::Error(error::ErrorType::Audio, "AudioManager::playSong",
388 "driver refused to play", error::ErrorSeverity::Recoverable);
389 }
390 }
391 catch (const std::exception &e) {
392 if (_dbg)
393 _dbg->log(debug::Level::Error,
394 std::string("AudioManager::playSong std::exception: ") + e.what());
395 throw error::Error(error::ErrorType::Audio, "AudioManager::playSong",
396 std::string("std::exception: ") + e.what(), error::ErrorSeverity::Recoverable);
397 }
398}
