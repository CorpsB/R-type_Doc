8/**
9 * @file SceneManager.hpp
10 * @brief Declaration of the @ref rtype::client::SceneManager class.
11 * @since 1.0
12 * @ingroup client-scenes
13 * @see IScene
14 *
15 * @details
16 * This file defines the @ref rtype::client::SceneManager, which owns and manages
17 * the lifecycle of all game scenes. The manager maintains a stack-based structure
18 * where the scene at the back of the stack is always the active one.
19 *
20 * Responsibilities:
21 * - Handle scene transitions (push, pop, replace) safely via deferred operations.
22 * - Maintain a stack of unique scenes, each implementing the @ref IScene interface.
23 * - Provide lifecycle synchronization between scenes (enter, exit, pause, resume).
24 *
25 * All modifications to the stack are deferred and processed in batch
26 * through @ref flushTransitions() to ensure stability during updates.
27 *
28 * @note This class follows the singleton pattern.
29 * @warning Scene transitions should only occur between update cycles.
30 */

45/**
46 * @class SceneManager
47 * @brief Manages and orchestrates a stack of active scenes.
48 *
49 * @details
50 * The SceneManager acts as the core control unit for the client-side
51 * scene system. Scenes are owned via unique pointers, ensuring safe
52 * and deterministic destruction order. Transitions (push, pop, replace)
53 * are deferred to prevent mid-frame modifications.
54 *
55 * Example usage:
56 * @code{.cpp}
57 * auto &sceneManager = rtype::client::SceneManager::getInstance();
58 * sceneManager.pushNew<MainMenuScene>();
59 * sceneManager.update(dt);
60 * sceneManager.draw();
61 * @endcode
62 *
63 * @see IScene, flushTransitions
64 */
65class SceneManager {

71 static SceneManager &getInstance();

73 /**
74 * @brief Queues a new scene to be pushed on top of the stack.
75 * @param scene Unique pointer to the scene to push.
76 * @details
77 * The scene is not immediately active; it will become so after
78 * @ref flushTransitions() is executed.
79 */
80 void push(ScenePtr scene);
81
82 /**
83 * @brief Queues a pop operation to remove the current active scene.
84 * @details
85 * The scene remains active until @ref flushTransitions() is called.
86 */
87 void pop();
88
89 /**
90 * @brief Queues a replace operation, replacing the current scene with another.
91 * @param scene Unique pointer to the replacement scene.
92 * @details
93 * The old scene is destroyed, and the new one becomes active after
94 * @ref flushTransitions() is processed.
95 */
96 void replace(ScenePtr scene);
97
98 /**
99 * @brief Updates the currently active scene.
100 * @param dt Delta time (in seconds) since the last frame.
101 *
102 * @details
103 * Calls @ref IScene::update() on the active scene and then
104 * processes any pending transitions.
105 */
106 void update(float dt);
107
108 /**
109 * @brief Draws the active scene.
110 * @details
111 * Calls @ref IScene::draw() on the scene at the top of the stack.
112 */
113 void draw() const;
114
115 /**
116 * @brief Applies all pending scene transitions.
117 *
118 * @details
119 * Executes all queued operations in FIFO order:
120 * - **Push**: Pauses current scene (if any), activates the new one.
121 * - **Pop**: Exits current scene, resumes the previous one.
122 * - **Replace**: Destroys the current scene and activates the new one.
123 *
124 * @note All pending operations are cleared afterward.
125 */
126 void flushTransitions();
127
128 /**
129 * @brief Creates and queues a new scene of type @p T.
130 *
131 * @tparam T Concrete type deriving from @ref IScene.
132 * @tparam Args Argument types for @p T’s constructor.
133 * @param args Constructor arguments forwarded to @p T.
134 *
135 * @details
136 * Wraps scene creation and queues it for activation via @ref push().
137 * The scene becomes active only after @ref flushTransitions() runs.
138 */
139 template<typename T, typename... Args>
140 void pushNew(Args&&... args)
141 {
142 push(std::make_unique<T>(std::forward<Args>(args)...));
143 }
144
145 /**
146 * @brief Creates and replaces the current scene with a new instance of type @p T.
147 *
148 * @tparam T Concrete type deriving from @ref IScene.
149 * @tparam Args Argument types for @p T’s constructor.
150 * @param args Constructor arguments forwarded to @p T.
151 *
152 * @details
153 * Wraps scene creation and queues it for replacement via @ref replace().
154 * The current active scene will be destroyed upon @ref flushTransitions().
155 */
156 template<typename T, typename... Args>
157 void replaceNew(Args&&... args)
158 {
159 replace(std::make_unique<T>(std::forward<Args>(args)...));
160 }

193 std::vector<ScenePtr> _stack; ///< Scene stack (back() is the active scene).

195};
