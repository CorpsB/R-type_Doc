23namespace asge::scene {
24
25/**
26 * @class SceneManager
27 * @brief Implements the IScene interface to manage a stack of scenes.
28 *
29 * @details
30 * The SceneManager is responsible for the main game loop's scene flow.
31 * It holds a stack of scenes, but only the scene at the top of the stack
32 * (the "active" scene) is updated and drawn.
33 *
34 * All transition operations (push, pop, replace) are **deferred**.
35 * They are queued as pending operations and executed only when
36 * flushTransitions() is called, typically at the end of a frame.
37 * This prevents modifying the scene stack during an update or draw call.
38 *
39 * @see IScene
40 * @see asge::scene::SceneManager::PendingOp
41 */
42class SceneManager : public IScene {
43 public:
44 /**
45 * @brief Constructs the SceneManager.
46 * @param services A reference to the core services (e.g., input, renderer)
47 * to be passed to scenes.
48 */
49 SceneManager(Services &services) : _s(&services) {};
50
51 /**
52 * @brief Queues a 'push' operation.
53 * @details Adds a new scene to the top of the stack.
54 * The new scene will become active, and the current scene will be paused.
55 * The operation is deferred until flushTransitions() is called.
56 * @param scene The ScenePtr (e.g., std::unique_ptr<IScene>) to push onto the stack.
57 */
58 void push(ScenePtr scene) override;
59
60 /**
61 * @brief Queues a 'pop' operation.
62 * @details Removes the currently active scene from the top of the stack.
63 * The previous scene in the stack will become active.
64 * The operation is deferred until flushTransitions() is called.
65 */
66 void pop() override;
67
68 /**
69 * @brief Queues a 'replace' operation.
70 * @details Replaces the currently active scene with a new one.
71 * This is equivalent to a 'pop' followed by a 'push'.
72 * The operation is deferred until flushTransitions() is called.
73 * @param scene The ScenePtr (e.g., std::unique_ptr<IScene>) to replace the current one.
74 */
75 void replace(ScenePtr scene) override;
76
77 /**
78 * @brief Updates the currently active scene (top of the stack).
79 * @details If the stack is not empty, this calls update() on the scene
80 * at the top of the stack.
81 * @param dt The delta time (time elapsed since the last frame).
82 */
83 void update(double dt) override;
84
85 /**
86 * @brief Draws the currently active scene (top of the stack).
87 * @details If the stack is not empty, this calls draw() on the scene
88 * at the top of the stack.
89 */
90 void draw() const override;
91
92 /**
93 * @brief Executes all pending scene transitions (push, pop, replace).
94 * @details This method iterates over the queue of pending operations
95 * and applies them to the scene stack in the order they were requested.
96 * This should be called at a safe point in the game loop,
97 * typically once per frame.
98 */
99 void flushTransitions() override;
100
101 /**
102 * @brief Checks if the scene stack is empty.
103 * @return true if the stack is empty, false otherwise.
104 */
105 bool empty() const override;
106
107 private:
108 Services *_s; ///< Pointer to the core engine services, passed to scenes.
109
110 ///< Defines the type of pending scene transition.
111 enum class OpType { None, Push, Pop, Replace };
112
113 /**
114 * @struct PendingOp
115 * @brief Represents a deferred scene transition operation.
116 * @details Stores the type of operation (Push, Pop, Replace) and
117 * the scene associated with it (for Push and Replace).
118 */
119 struct PendingOp {
120 OpType type{ OpType::None };
121 ScenePtr scene{ nullptr };
122 };
123
124 std::vector<ScenePtr> _stack; ///< The stack of active scenes. Only _stack.back() is updated/drawn.
125 std::vector<PendingOp> _pending; ///< The queue of pending operations to be executed by flushTransitions().
126};

128}
