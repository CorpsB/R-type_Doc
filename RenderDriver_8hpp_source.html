10/**
11 * @file RenderDriver.hpp
12 * @brief Minimal Raylib-backed driver exposing only high-level draw ops.
13 * @since 0.1
14 * @ingroup renderer
15 *
16 * @details
17 * Cette interface définit le contrat d'un pilote de rendu adossé à Raylib,
18 * limité aux opérations de haut niveau nécessaires au moteur :
19 * - Cycle de vie : @ref init(), @ref beginFrame(), @ref clear(), @ref present(), @ref shutdown().
20 * - Aides textures minimales pour permettre @ref drawSprite().
21 * - Primitives 2D : cercle, sprite, texte, rectangle, mesure de texte.
22 *
23 * @note Aucune abstraction multi-backend n'est proposée ici (pas d'OpenGL/Vulkan/Metal).
24 * @warning Tous les appels doivent être effectués depuis le thread de rendu.
25 */
26

34/**
35 * @class RenderDriver
36 * @brief Raylib-only driver: lifecycle + high-level drawing.
37 * @ingroup renderer
38 *
39 * @details
40 * Cycle de vie typique :
41 * @code
42 * driver.init(w, h, "title");
43 * while (running) {
44 * driver.beginFrame();
45 * driver.clear(0,0,0,1);
46 * draw calls...
47 * driver.present();
48 * }
49 * driver.shutdown();
50 * @endcode
51 *
52 * @note Les ressources GPU (textures) créées par @ref createTextureFromPixels()
53 * doivent être libérées via @ref destroyTexture().
54 */
55class RenderDriver {

65 virtual bool isFullscreen() const = 0;
66 virtual void toggleFullscreen() = 0;

68 /**
69 * @brief Initialise la fenêtre/contexte Raylib.
70 *
71 * @param width Largeur framebuffer (en pixels).
72 * @param height Hauteur framebuffer (en pixels).
73 * @param title Titre de la fenêtre (si applicable).
74 * @return `true` en cas de succès, `false` sinon.
75 *
76 * @pre Aucun contexte Raylib actif pour ce pilote.
77 * @post Le pilote est prêt à recevoir @ref beginFrame() et des appels de dessin.
78 * @warning Doit être appelé une seule fois avant tout rendu.
79 */
80 virtual bool init(std::uint32_t width, std::uint32_t height, const std::string &title) = 0;
81
82 /**
83 * @brief Démarre une nouvelle frame (équivalent à BeginDrawing).
84 *
85 * @pre @ref init() a été appelé avec succès.
86 * @post Le backend accepte des commandes de dessin jusqu'à @ref present().
87 * @warning Appel requis à chaque frame avant tout draw/clear.
88 */
89 virtual void beginFrame() = 0;
90
91 /**
92 * @brief Efface la cible de rendu avec une couleur RGBA (flottants 0..1).
93 *
94 * @param r Composante rouge (0..1).
95 * @param g Composante verte (0..1).
96 * @param b Composante bleue (0..1).
97 * @param a Composante alpha (0..1).
98 *
99 * @pre @ref beginFrame() a été invoqué sur la frame courante.
100 * @post Le framebuffer courant est nettoyé avec la couleur fournie.
101 */
102 virtual void clear(float r, float g, float b, float a) = 0;
103
104 /**
105 * @brief Termine et présente la frame (EndDrawing + swap buffers).
106 *
107 * @pre @ref beginFrame() a été appelé pour la frame courante.
108 * @post Les commandes de dessin de la frame sont soumises et affichées.
109 * @warning A appeler exactement une fois par frame.
110 */
111 virtual void present() = 0;
112
113 /**
114 * @brief Indique un redimensionnement de fenêtre/framebuffer.
115 *
116 * @param width Nouvelle largeur en pixels.
117 * @param height Nouvelle hauteur en pixels.
118 *
119 * @note No-op si Raylib gère déjà la taille en interne.
120 * @pre Le contexte d'affichage doit être initialisé.
121 */
122 virtual void resize(std::uint32_t width, std::uint32_t height) = 0;
123
124 /**
125 * @brief Ferme Raylib et libère les ressources associées.
126 *
127 * @post Aucune ressource GPU/OS détenue par le pilote.
128 * @warning Doit être appelé avant destruction du pilote si @ref init() a réussi.
129 */
130 virtual void shutdown() = 0;
131
132 /**
133 * @brief Crée une texture à partir de pixels en mémoire CPU.
134 *
135 * @param pixel_data Pointeur vers les données (non nul).
136 * @param texture_width Largeur de la texture (px).
137 * @param texture_height Hauteur de la texture (px).
138 * @param pixel_format Format de pixels (ex : RGBA8).
139 * @return Un @ref TextureHandle valide en cas de succès, invalide sinon.
140 *
141 * @pre `pixel_data != nullptr`, dimensions > 0.
142 * @post Une ressource GPU est allouée si succès.
143 * @warning La texture créée doit être libérée via @ref destroyTexture().
144 */
145 virtual TextureHandle createTextureFromPixels(const void *pixel_data,
146 std::uint32_t texture_width, std::uint32_t texture_height, PixelFormat pixel_format) = 0;
147
148 /**
149 * @brief Détruit une texture précédemment créée.
150 *
151 * @param texture_handle Handle de texture retourné par @ref createTextureFromPixels().
152 *
153 * @post Le handle devient invalide et la ressource GPU correspondante est libérée.
154 * @warning Ne pas utiliser le handle après destruction.
155 */
156 virtual void destroyTexture(TextureHandle texture_handle) = 0;
157
158 /**
159 * @brief Dessine un cercle rempli ou en contour.
160 *
161 * @param center_x Centre X.
162 * @param center_y Centre Y.
163 * @param radius Rayon en pixels.
164 * @param color Couleur (remplissage/contour).
165 * @param is_filled `true` pour rempli, `false` pour contour.
166 * @param border_thickness Épaisseur du trait (mode contour).
167 *

170 virtual void drawCircle(float center_x, float center_y, float radius,

172
173 /**
174 * @brief Dessine un sprite (quad texturé).
175 *
176 * @param texture_handle Handle de texture valide.
177 * @param destination_rect Rectangle destination écran.
178 * @param source_rect Rectangle source dans la texture (UV en pixels).
179 * @param origin_point Pivot de rotation (souvent centre ou top-left).
180 * @param rotation_degrees Angle en degrés.
181 * @param tint_color Teinte (blanc = sans teinte).

185 virtual void drawSprite(TextureHandle texture_handle, Rect destination_rect, Rect source_rect,
186 Vec2f origin_point, float rotation_degrees, RGBAColor tint_color) = 0;

188 /**

196 *

199 virtual void drawText(const std::string &text,

203 * @brief Dessine un rectangle rempli ou en contour (AABB).

209 *

212 virtual void drawRectangle(Rect rectangle, RGBAColor color,
