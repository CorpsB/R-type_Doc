13 * @version 1.0

16 * @details

37/**
38 * @struct EntityData
39 * @brief Stores metadata associated with an ECS entity.
40 *
41 * @details
42 * - `generation` increments every time an entity ID is reused
43 * - `signature` determines compatibility with systems
44 * - `active` indicates whether the entity is currently in use
45 */
46struct EntityData {
47 uint32_t generation = 0; ///< Number of times this entity ID has been reused.
48 std::bitset<MAX_COMPONENTS> signature; ///< Component inclusion mask.
49 bool active = false; ///< Indicates if the entity is currently alive.
50};

52/**
53 * @class EntityManager
54 * @brief Handles allocation, recycling, and signature access for entities.
55 *
56 * @details
57 * The manager does not store any component data. It only manages:
58 * - Which entity IDs are active
59 * - The signature bitmask of each active entity
60 */
61class EntityManager {
62 public:
63 /**
64 * @brief Constructs the entity manager and initializes ID freelist.

66 EntityManager();
67
68 /**
69 * @brief Virtual destructor for safe polymorphic use.
70 */
71 virtual ~EntityManager() = default;
72
73 /**
74 * @brief Creates a new entity and returns its ID.
75 * @return Newly allocated entity ID.
76 * @throw asge::error::Error If out of free entity slots.
77 */
78 virtual uint32_t createEntity();
79
80 /**
81 * @brief Activates an entity with a specific ID if it's free.
82 * @param id Entity ID to activate.
83 * @throw asge::error::Error If ID invalid or already active.
84 */
85 virtual void createEntity(uint32_t id);
86
87 /**
88 * @brief Destroys an entity and returns its ID to freelist.
89 * @param id ID of entity to destroy.
90 * @return true if entity was destroyed, false if already inactive.
91 * @throw asge::error::Error If ID out of range.
92 */
93 virtual bool destroyEntity(uint32_t id);
94
95 /**

100 */
101 virtual std::bitset<MAX_COMPONENTS> &getSignature(uint32_t id);
102
103 /**
104 * @brief Assigns signature mask to a given entity.
105 * @param id Entity ID.
106 * @param signature New signature value.
107 * @return true if applied successfully.
108 * @throw asge::error::Error If ID out of range.
109 */
110 virtual bool setSignature(uint32_t id, const std::bitset<MAX_COMPONENTS> &signature);
111
112#ifndef TESTING
113 private:
114#endif
115 std::vector<EntityData> _entities; ///< Storage for entity metadata.
116 std::vector<uint32_t> _free_ids; ///< Recycled ID pool for efficient reuse.
