168template<typename T>
169inline bool World::addComponent(uint32_t id, const T &component)
170{
171 if (!_component_manager->addComponent<T>(id, component))
172 return false;
173 auto &signature = _entity_manager->getSignature(id);
174 signature.set(_component_manager->getComponentTypeID<T>(), true);
175 _entity_manager->setSignature(id, signature);
176 _system_manager->entitySignatureChanged(id, signature);
177 return true;
178}

180template<typename T>
181inline bool World::removeComponent(uint32_t id)
182{
183 if (!_component_manager->removeComponent<T>(id))
184 return false;
185 auto &signature = _entity_manager->getSignature(id);
186 signature.set(_component_manager->getComponentTypeID<T>(), false);
187 _entity_manager->setSignature(id, signature);
188 _system_manager->entitySignatureChanged(id, signature);
189 return true;
190}
