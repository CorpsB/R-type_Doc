10/**
11 * @file Error.hpp
12 * @brief Exception type used by the engine for categorized errors.
13 *
14 * @details
15 * The asge::error::Error class wraps an error category (ErrorType), a textual
16 * location (where the error originated), a human readable message and a
17 * severity level (ErrorSeverity). It derives from std::exception to allow
18 * propagation via standard exception handling while providing rich diagnostic
19 * formatting via toString() and a lazy-cached what() implementation.
20 *
21 * The type / severity fields are intended to allow higher-level systems to
22 * decide how to respond to an error (log, retry, shutdown).
23 */
24

32/**
33 * @class Error
34 * @brief Rich, categorized runtime error type for the engine.
35 *
36 * @details
37 * Error instances are small, copyable exceptions that own their message and
38 * location strings. The what() message is lazily formatted and cached so that
39 * repeated calls are cheap. All accessor methods are noexcept and intended to
40 * be safe to call from destructors and logging code paths.
41 */
42class Error : public std::exception {
43 public:
44 /**
45 * @brief Construct a new Error.
46 * @param type Category of the error (see ErrorType).
47 * @param where Logical location where the error originated (module, file, function).
48 * @param what Human readable description of the failure.
49 * @param severity Severity indicating how the runtime should react.
50 *
51 * @note Constructor is noexcept so it can be used in error handling paths
52 * without throwing further exceptions. It should only be used for
53 * recoverable diagnostic creation; fatal actions should still be handled
54 * separately (e.g. by supervising systems).
55 */
56 Error(ErrorType type, const std::string &where, const std::string &what,
57 ErrorSeverity severity = ErrorSeverity::Warning) noexcept;
58
59 /**
60 * @brief Returns a C-style diagnostic string.
61 *
62 * @returns Pointer to an internal, null-terminated string suitable for
63 * use with std::exception interfaces. The returned pointer
64 * remains valid as long as the exception object exists.
65 *
66 * @note The string is lazily formatted on first call and cached for
67 * subsequent calls.
68 */
69 const char *what() const noexcept override;
