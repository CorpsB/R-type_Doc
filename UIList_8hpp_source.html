27/**
28 * @class UIList
29 * @brief A UI element that acts as a container for other UI elements.
30 * @details Inherits from UIElement and holds a map of child elements.
31 * It delegates update and draw calls to all elements it contains,
32 * implementing the Composite design pattern.
33 */
34class UIList : public UIElement {
35 public:
36 /**
37 * @brief Default constructor for UIList.

39 UIList() {};
40
41 /**
42 * @brief Updates all UI elements contained in this list.
43 * @details Iterates over the internal map and calls update() on each child element.
44 * @param services Reference to core engine services.
45 * @param dt Delta time since the last frame.
46 */
47 void update(Services &services, double dt) override;
48
49 /**
50 * @brief Draws all UI elements contained in this list.
51 * @details Iterates over the internal map and calls draw() on each child element.
52 * @param services Reference to core engine services (e.g., renderer).
53 */
54 void draw(Services &services) const override;
55
56 /**
57 * @brief Adds a new UI element to the list.
58 * @details Associates the element with a unique key for retrieval
59 * and takes ownership of it via std::unique_ptr.
60 * @param key The unique string identifier for the element.
61 * @param element A std::unique_ptr<UIElement> to the element to be added.
62 */
63 void addUIElement(const std::string &key, std::unique_ptr<UIElement> element) { _elements[key] = std::move(element); };
64
65 /**
66 * @brief Retrieve a pointer to a child UI element.
67 *
68 * This helper returns a raw pointer to the stored UIElement associated with
69 * a given key. It can be used by scenes to modify UIElement state at
70 * runtime, for example to update progress bars or text labels. The
71 * lifetime of the returned pointer is managed by the UIList and should
72 * not be deleted by the caller. If the key is not present, nullptr
73 * is returned.
74 *
75 * @param key The string identifier used when adding the UI element.
76 * @return UIElement* Pointer to the element or nullptr if not found.
77 */
78 UIElement *getUIElement(const std::string &key)
79 {
80 auto it = _elements.find(key);
81 return it != _elements.end() ? it->second.get() : nullptr;
82 }
83
84 /**
85 * @brief Retrieve a constant pointer to a child UI element.
86 *
87 * Const‑qualified overload of getUIElement for read‑only access. Returns
88 * nullptr if the key is not associated with any element.
89 *
90 * @param key The string identifier used when adding the UI element.
91 * @return const UIElement* Constant pointer to the element or nullptr.
92 */
93 const UIElement *getUIElement(const std::string &key) const
94 {
95 auto it = _elements.find(key);
96 return it != _elements.end() ? it->second.get() : nullptr;
97 }
98
99 private:
100 std::unordered_map<std::string, std::unique_ptr<UIElement>> _elements; ///< Map of child UI elements, keyed by name.
101};
