10 * @brief Implementation of AudioSystem — system for sound playback control.

12 * @details
13 * Implements AudioSystem::update and handles audio device lifecycle.
14 * The update loop synchronizes ECS audio components with Raylib playback:
15 * - plays sounds marked as `play`,
16 * - stops those marked as `stop`,

41void AudioSystem::update(Coordinator &coord)
42{
43 auto &resourceManager = rtype::client::ResourceManager::getInstance();
44
45 for (const uint32_t id : _filtered_entities) {
46 audio_attr_t *audio = coord.getComponent<audio_attr_t>(id);
47 if (!audio)
48 continue;
49
50 try {
51 Sound &snd = resourceManager.getSound(audio->soundKey);
52
53 if (audio->stop) {
54 StopSound(snd);
55 audio->playing = false;
56 audio->stop = false;
57 }
58
59 if (audio->play && !audio->playing) {
60 PlaySound(snd);
61 audio->playing = true;
62 audio->play = false;
63 }
64
65 if (audio->loop && !IsSoundPlaying(snd)) {
66 PlaySound(snd);
67 audio->playing = true;
68 }
69 } catch (...) {
70 continue;
71 }
72 }
73}
