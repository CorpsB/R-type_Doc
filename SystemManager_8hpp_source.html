132template<typename T, typename... Args>
133inline std::shared_ptr<T> SystemManager::registerSystem(ComponentManager &m, Args&&... args)
134{
135 unsigned int type_id = getSystemTypeID<T>();
136 if (_systems.find(type_id) != _systems.end())
137 return nullptr;
138 auto sys = std::make_shared<T>(std::forward<Args>(args)...);
139 _systems[type_id] = sys;
140 _systems[type_id]->setSignature(m);
141 return sys;
142}
143
144template<typename T>
145inline bool SystemManager::deregisterSystem()
146{
147 unsigned int type_id = getSystemTypeID<T>();
148 auto it = _systems.find(type_id);
149 if (it != _systems.end()) {
150 _systems.erase(it);
151 _exec_signatures.erase(type_id);
152 return true;
153 }
154 return false;
155}
