33/**
34 * @class SystemManager
35 * @brief Centralized registry for systems and their execution signatures.
36 *
37 * @details
38 * SystemManager maps small integer type ids to System instances and keeps a
39 * corresponding execution signature (bitset of required components) for each
40 * registered system. When an entity's signature changes the manager updates
41 * every system's filtered entity set to add or remove the entity as appropriate.
42 */
43class SystemManager {
44 public:
45 SystemManager() = default;
46
47 /**
48 * @brief Get a stable numeric id for system type T.
49 *
50 * @tparam T System type
51 * @return unsigned int Unique id assigned to the type T
52 *
53 * @details
54 * Uses a per-template static counter to ensure each distinct type T
55 * receives a unique incremental id at runtime.
56 */
57 template<typename T>
58 unsigned int getSystemTypeID()
59 {
60 static unsigned int type_id = _system_id_cpt++;
61
62 return type_id;
63 }
64
65 /**
66 * @brief Register and create a system instance of type T.
67 *
68 * @tparam T System type (derived from System)
69 * @tparam Args Parameter pack forwarded to T's constructor
70 * @param args Constructor arguments for T
71 * @return std::shared_ptr<T> Shared pointer to the created system, or nullptr if already registered
72 *
73 * @details
74 * The new system is stored in the internal _systems map indexed by its type id.
75 */
76 template<typename T, typename... Args>
77 std::shared_ptr<T> registerSystem(Args&&... args);
78
79 /**
80 * @brief Deregister a previously registered system type T.
81 *
82 * @tparam T System type to deregister

85 template<typename T>
86 bool deregisterSystem();

88 /**

94 template<typename T>

136 void removeEntityFromSystems(uint32_t id);

146template<typename T, typename... Args>
147inline std::shared_ptr<T> SystemManager::registerSystem(Args&&... args)
148{
149 unsigned int type_id = getSystemTypeID<T>();
150
151 if (_systems.find(type_id) != _systems.end())
152 return nullptr;
153
154 auto sys = std::make_shared<T>(std::forward<Args>(args)...);
155 _systems[type_id] = sys;
156 return sys;
157}
158
159template <typename T>
160inline bool SystemManager::deregisterSystem()
161{
162 unsigned int type_id = getSystemTypeID<T>();
163 auto it = _systems.find(type_id);
164
165 if (it != _systems.end()) {
166 _systems.erase(it);
167 _exec_signatures.erase(type_id);
168 return true;
169 }
170 return false;
171}

181template<typename T>
182inline bool SystemManager::isSystemEnabled()

185

189template<typename T>
190inline void SystemManager::setSignature(const std::bitset<MAX_COMPONENTS> &signature)
191{
192 unsigned int type_id = getSystemTypeID<T>();
193
194 _exec_signatures[type_id] = signature;
195}

197inline void SystemManager::entitySignatureChanged(uint32_t id,
198 const std::bitset<MAX_COMPONENTS> &entitySignature)
199{
200 for (auto &sys : _systems) {
201 uint32_t system_type = sys.first;
202 auto &system = sys.second;
203 auto &exec_signature = _exec_signatures[system_type];
204
205 if (exec_signature == 0)
206 return;
207 if ((entitySignature & exec_signature) == exec_signature) {
208 system->_filtered_entities.insert(id);
209 } else {
210 system->_filtered_entities.erase(id);
211 }
212 }
213}
