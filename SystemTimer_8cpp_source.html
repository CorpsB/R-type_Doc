96void asge::timer::SystemTimer::tick()
97{
98 try {
99 const TimePoint current = Clock::now();
100
101 std::lock_guard<std::mutex> lock(_mutex);
102 std::chrono::duration<double> duration = current - _last;
103
104 _delta = duration.count();
105 _last = current;
106 _ticks.fetch_add(1, std::memory_order_relaxed);
107 }
108 catch (const std::exception &exception) {
109 if (_debugger) {
110 _debugger->log(
111 asge::debug::Level::Error,

113 }
114
115 throw asge::error::Error(
