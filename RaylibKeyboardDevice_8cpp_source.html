78/**
79 * @brief Polls Raylib and refreshes the cached device state.
80 *
81 * @details
82 * - Clears previous key state containers before sampling.
83 * - Populates pressed/released/down sets for every interesting scan code.
84 * - Surfaces any unexpected backend failure through the debug service
85 * before rethrowing.
86 */
87void RaylibKeyboardDevice::poll()
88{
89 if (!_enabled) {
90 return;
91 }
92
93 try {
94 _last_state.pressed.clear();
95 _last_state.released.clear();
96 _last_state.down.clear();
97
98 for (int key : SCAN_KEYS) {
99 if (IsKeyPressed(key)) {
100 _last_state.pressed.insert(translateKey(key));
101 }
102 if (IsKeyReleased(key)) {
103 _last_state.released.insert(translateKey(key));
104 }
105 if (IsKeyDown(key)) {
106 _last_state.down.insert(translateKey(key));
107 }
108 }
109 } catch (const std::exception &exception) {
110 if (_debug) {
111 _debug->log(asge::debug::Level::Error,
112 std::string("[Input] RaylibKeyboardDevice::poll exception: ") + exception.what());
113 }
114 throw;
115 } catch (...) {
116 if (_debug) {
117 _debug->log(asge::debug::Level::Error,
118 "[Input] RaylibKeyboardDevice::poll unknown exception");
119 }
120 throw;
121 }
122}

176/**
177 * @brief Converts a Raylib key code into the engine's KeyCode domain.
178 *
179 * @note The mapping currently normalises AZERTY-friendly WASD controls.
180 */
181int RaylibKeyboardDevice::translateKey(int key) const
182{
183 switch (key) {
184 case KeyboardKey::KEY_W:
185 return static_cast<int>(input::KeyCode::Z);
186 case KeyboardKey::KEY_S:
187 return static_cast<int>(input::KeyCode::S);
188 case KeyboardKey::KEY_A:
189 return static_cast<int>(input::KeyCode::Q);
190 case KeyboardKey::KEY_D:
191 return static_cast<int>(input::KeyCode::D);
192 case KeyboardKey::KEY_SPACE:
193 return static_cast<int>(input::KeyCode::SPACE);
194 default:
195 return static_cast<int>(input::KeyCode::UNKNOWN);
196 }
197}
