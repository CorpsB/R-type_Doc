69bool asge::event::EventManager::unsubscribe(const std::string &type, uint64_t id)
70{
71 if (type.empty()) {
72 if (_debugger)
73 _debugger->log(asge::debug::Level::Warning,
74 "EventManager::unsubscribe called with empty type");
75 }
76
77 std::lock_guard<std::mutex> lock(_mutex);
78
79 auto type_iter = _subs.find(type);
80 if (type_iter == _subs.end()) {
81 if (_debugger)
82 _debugger->log(asge::debug::Level::Warning,
83 "EventManager::unsubscribe: no such event type '" + type + "'");
84 return false;
85 }
86
87 auto &handlers = type_iter->second;
88 auto handler_iter = handlers.find(id);
89
90 if (handler_iter == handlers.end()) {
91 std::ostringstream message;
92 message << "EventManager::unsubscribe: handler id " << id
93 << " not found for type '" << type << "'";
94 if (_debugger)
95 _debugger->log(asge::debug::Level::Warning, message.str());
96 return false;
97 }
98
99 handlers.erase(handler_iter);
100 if (handlers.empty())
101 _subs.erase(type_iter);
102
103 return true;
104}

167void asge::event::EventManager::flush()
168{
169 while (true)
170 {
171 Event next_event;
172
173 {
174 std::lock_guard<std::mutex> lock(_mutex);
175
176 if (_deferred.empty())
177 break;
178
179 next_event = std::move(_deferred.front());
180 _deferred.pop_front();
181 }
182
183 dispatch(next_event);
184 }

187void asge::event::EventManager::dispatch(const Event &event)
188{
189 std::vector<Handler> snapshot;
190
191 {
192 std::lock_guard<std::mutex> lock(_mutex);
193
194 auto type_iter = _subs.find(event.first);
195 if (type_iter == _subs.end())
196 return;
197
198 snapshot.reserve(type_iter->second.size());
199
200 for (const auto &entry : type_iter->second)
201 snapshot.push_back(entry.second);
202 }
203
204 for (const auto &handler : snapshot)
205 {
206 try {
207 handler(event.second);
208 }
209 catch (const std::exception &exception) {
210 std::ostringstream message;
211 message << "EventManager::dispatch: exception while handling event type '"
212 << event.first << "': " << exception.what();
213
214 if (_debugger)
215 _debugger->log(asge::debug::Level::Error, message.str());
216 }
217 catch (...) {
218 std::ostringstream message;
219 message << "EventManager::dispatch: unknown exception while handling event type '"
220 << event.first << "'";
221
222 if (_debugger)
223 _debugger->log(asge::debug::Level::Error, message.str());
224 }
225 }
