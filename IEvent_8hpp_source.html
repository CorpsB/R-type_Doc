28/**
29 * @class IEvent
30 * @brief Minimal abstract interface for an event manager / event bus.
31 *
32 * @details
33 * IEvent defines the contract used by subsystems to publish and subscribe
34 * to string-identified events. Handlers receive an opaque string payload.
35 *
36 * Typical usage:
37 * - subscribe("PLAYER_HIT", handler) returns subscription id
38 * - publish({"PLAYER_HIT", "payload"})
39 * - publishDeferred(...) queues the event for later dispatch via flush()
40 *
41 * Implementations are free to define the exact semantics of "deferred" publication.
42 */
43class IEvent {
44 public:
45 /// Handler signature: receives the event payload as a string
46 using Handler = std::function<void(const std::string &)>;
47 /// Event pair: (type, payload)
48 using Event = std::pair<std::string, std::string>;
49
50 virtual ~IEvent() = default;
51
52 virtual void clearDebug() = 0;
53
54 /**
55 * @brief Publish an event immediately (synchronous dispatch).
56 * @param event Event pair (type, payload)
57 *
58 * @details
59 * If the event type string is empty a warning is logged and the event is still dispatched.
60 */
61 virtual void publish(const Event &event) = 0;
62
63 /**
64 * @brief Queue an event for deferred dispatch (processed by flush()).
65 * @param event Event pair (type, payload)
66 *
67 * @details
68 * The function ensures the deferred queue growth is monitored and logs a warning
69 * if the queue becomes large. Exceptions during queueing are logged and wrapped.
70 */
71 virtual void publishDeferred(const Event &event) = 0;
72
73 /**
74 * @brief Subscribe to events of a given type.
75 * @param type String event type
76 * @param handler Handler function called when events of @p type are published
77 *
78 * @details
79 * Allocates a unique id for the subscription and stores the handler in the registry.
80 * Logs and throws a recoverable error if insertion fails.
81 * @return A subscription id (opaque) that can be used to unsubscribe
82 */
83 virtual uint64_t subscribe(const std::string &type, Handler handler) = 0;
84
85 /**
86 * @brief Unsubscribe a previously registered handler.
87 * @param type Event type
88 * @param id Subscription id returned by subscribe()
89 * @return true on success, false if nothing to remove
90 *
91 * @details
92 * If the event type has no remaining handlers after removal the type entry is erased.
93 */
94 virtual bool unsubscribe(const std::string &type, uint64_t id) = 0;
95
96 /**
97 * @brief Flush and dispatch all deferred events.
98 *
99 * @details
100 * The function pops events from the deferred queue and dispatches them one by one.
101 * It takes the minimal lock to pop an event and dispatches outside the lock to prevent
102 * handlers from blocking the queue or causing deadlocks.
103 */
104 virtual void flush() = 0;
105};
106
107}
