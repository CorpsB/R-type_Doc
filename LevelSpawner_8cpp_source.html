8/**
9 * @file LevelSpawner.cpp
10 * @brief Implementation of the script-driven progressive level spawner.
11 * @since 1.0
12 * @ingroup scripting
13 *
14 * @details
15 * The spawner pulls spawn data from `ScriptManager` (via `MapAsset`) and
16 * instantiates entities using the script-side `EntityFactory`. For each spawn,
17 * an optional position override is read from the JSON payload. After successful
18 * creation, the network protocol is notified so clients receive the entity.
19 * Errors are logged via the debug service and do not throw.
20 */

27/**
28 * @brief Constructs a level spawner and loads the referenced map asset.
29 *
30 * @param services Engine services used (scripts, world, network, debug).
31 * @param map_id Map identifier to resolve via the script manager.
32 *
33 * @details
34 * On success, the spawn list is copied and sorted by ascending distance.
35 * On failure (unknown map id), `_valid` is set to false and a warning is logged.
36 */
37asge::script::LevelSpawner::LevelSpawner(Services &services, const std::string &map_id)
38 : _S(services), _map_id(map_id)
39{
40 const MapAsset *map = _S.scripts().getMap(map_id);
41 if (!map) {
42 if (_S._debug)
43 _S.debug().log(debug::Level::Warning, "LevelSpawner: map '" + map_id + "' not found.");
44 _valid = false;
45 return;
46 }
47
48 _sorted_spawns = map->spawns;
49 std::sort(_sorted_spawns.begin(), _sorted_spawns.end(),
50 [](const MapSpawn &a, const MapSpawn &b){ return a.distance < b.distance; });
51
52 _next_index = 0;
53 _last_progress = 0;
54 _valid = true;
55
56 if (_S._debug)
57 _S.debug().log(debug::Level::Info, "LevelSpawner: loaded map '" + map_id +
58 "' spawns=" + std::to_string(_sorted_spawns.size()));
59}

74/**
75 * @brief Immediately spawns all remaining entries in order.
76 *
77 * @details
78 * For each remaining spawn, this method:
79 * - parses optional `position` overrides from the spawn JSON,
80 * - retrieves the template handle by archetype,
81 * - creates the entity via `EntityFactory::createFromTemplate`,
82 * - notifies the network through `EngineProtocol::sendEntityCreate`.
83 *
84 * Failures (unknown archetype, factory exception, invalid id) are logged as
85 * warnings and do not abort iteration. On completion, the cursor advances to
86 * the end of the spawn list.
87 */
88void asge::script::LevelSpawner::spawnAll()
89{
90 if (!_valid) return;
91
92 for (size_t i = _next_index; i < _sorted_spawns.size(); ++i) {
93 const auto &s = _sorted_spawns[i];
94 std::optional<double> px = std::nullopt;
95 std::optional<double> py = std::nullopt;
96 if (s.position.is_object()) {
97 if (s.position.contains("x")) px = s.position["x"].get<double>();
98 if (s.position.contains("y")) py = s.position["y"].get<double>();
99 }
100
101 const ScriptHandle *templ = _S.scripts().getTemplate(s.archetype);
102 if (!templ) {
103 if (_S._debug)
104 _S.debug().log(debug::Level::Warning, "LevelSpawner.spawnAll: unknown archetype '" + s.archetype + "'");
105 continue;
106 }
107
108 uint32_t eid = world::INVALID_ID;
109 try {
110 eid = EntityFactory::createFromTemplate(_S, *templ, px, py);
111 network::EngineProtocol::sendEntityCreate(_S, eid);
112
113 std::vector<uint16_t> comp_ids = { net::GameProtocol::AvailableComponents::POSITION };
114 network::EngineProtocol::sendEntityComponents(_S, eid, comp_ids);
115 }
116 catch (...) {
117 if (_S._debug)
118 _S.debug().log(debug::Level::Warning, "LevelSpawner.spawnAll: exception when creating '" + s.archetype + "'");
119 continue;
120 }
121
122 if (eid == world::INVALID_ID) {
123 if (_S._debug)
124 _S.debug().log(debug::Level::Warning, "LevelSpawner.spawnAll: factory failed for '" + s.archetype + "'");
125 continue;
126 }
127 // optional: record layer or other spawn metadata via a component or event
128 }
129 _next_index = _sorted_spawns.size();
130}

132/**
133 * @brief Progressively spawns entries up to the given progress threshold.
134 *
135 * @param current_progress Progress value used to unlock spawns (monotonic non-strict).
136 *
137 * @details
138 * Spawns all entries where `distance <= current_progress`, starting at the
139 * current cursor. For each, it performs the same creation and notification
140 * sequence as `spawnAll()`. Unknown archetypes, exceptions, and invalid ids
141 * are logged as warnings. The cursor advances for every entry processed, and
142 * `_last_progress` is updated to the latest value (never forces respawn).
143 */
144void asge::script::LevelSpawner::update(std::uint64_t current_progress)
145{
146 if (!_valid) return;
147
148 if (current_progress < _last_progress) {
149 _last_progress = current_progress;
150 }
151
152 while (_next_index < _sorted_spawns.size() &&
153 _sorted_spawns[_next_index].distance <= current_progress) {
154 const auto &s = _sorted_spawns[_next_index];
155
156 std::optional<double> px, py;
157 if (s.position.is_object()) {
158 if (s.position.contains("x")) px = s.position["x"].get<double>();
159 if (s.position.contains("y")) py = s.position["y"].get<double>();
160 }
161
162 const ScriptHandle *templ = _S.scripts().getTemplate(s.archetype);
163 if (!templ) {
164 if (_S._debug)
165 _S.debug().log(debug::Level::Warning, "LevelSpawner.update: unknown archetype '" + s.archetype + "'");
166 ++_next_index;
167 continue;
168 }
169
170 uint32_t eid = world::INVALID_ID;
171 try {
172 eid = EntityFactory::createFromTemplate(_S, *templ, px, py);
173 network::EngineProtocol::sendEntityCreate(_S, eid);
174
175 std::vector<uint16_t> comp_ids = { net::GameProtocol::AvailableComponents::POSITION };
176 network::EngineProtocol::sendEntityComponents(_S, eid, comp_ids);
177 }
178 catch (...) {
179 if (_S._debug)
180 _S.debug().log(debug::Level::Warning, "LevelSpawner.update: exception when creating '" + s.archetype + "'");
181 ++_next_index;
182 continue;
183 }
184
185 if (eid == world::INVALID_ID) {
186 if (_S._debug)
187 _S.debug().log(debug::Level::Warning, "LevelSpawner.update: factory failed for '" + s.archetype + "'");
188 }
189 else {
190 if (_S._debug)
191 _S.debug().log(debug::Level::Info, "LevelSpawner.update: spawned '" + s.archetype + "' eid=" + std::to_string(eid));
192 }
193
194 ++_next_index;
195 }
196
197 _last_progress = current_progress;
198}
