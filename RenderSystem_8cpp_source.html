21void asge::world::RenderSystem::render(Services &services)
22{
23 if (!services._renderer || !services._scripts) {
24 return;
25 }
26
27 auto &world = services.world();
28
29 for (uint32_t entity_id : _filtered_entities) {
30 auto *position_component = world.getComponent<components::position_attr_t>(entity_id);
31 auto *sprite_component = world.getComponent<components::sprite_attr_t>(entity_id);
32
33 if (!position_component || !sprite_component) {
34 continue;
35 }
36
37 const float position_x = position_component->x;
38 const float position_y = position_component->y;
39
40 if (position_x < 0 || position_x > VIRTUAL_SCREEN_WIDTH ||
41 position_y < 0 || position_y > VIRTUAL_SCREEN_HEIGHT) {
42 continue;
43 }
44
45 const auto *sprite_asset = services.scripts().getSprite(sprite_component->sprite_id);
46 if (!sprite_asset) {
47 if (services._debug) {
48 services.debug().log(asge::debug::Level::Warning,
49 "RenderSystem: missing sprite asset " + sprite_component->sprite_id);
50 }
51 continue;
52 }
53
54 auto cache_entry = _texture_cache.find(sprite_component->sprite_id);
55 asge::renderer::TextureHandle texture_handle;
56 if (cache_entry == _texture_cache.end()) {
57 std::string texture_path = services.scripts().getAssetsRoot() + "/" + sprite_asset->link;
58
59 texture_handle = services.renderer().loadTextureFromFile(texture_path, services.storage());
60 if (!texture_handle) {
61 if (services._debug) {
62 services.debug().log(asge::debug::Level::Warning,
63 "RenderSystem: failed to load texture " + texture_path);
64 }
65 continue;
66 }
67 _texture_cache.emplace(sprite_component->sprite_id, texture_handle);
68 }
69 else {
70 texture_handle = cache_entry->second;
71 }
72
73 renderer::Rect source_rect;
74 source_rect.x = static_cast<float>(sprite_asset->left);
75 source_rect.y = static_cast<float>(sprite_asset->top);
76 source_rect.w = static_cast<float>(sprite_asset->width);
77 source_rect.h = static_cast<float>(sprite_asset->height);
78
79 float scale_factor = static_cast<float>(sprite_asset->scale);
80 renderer::Rect destination_rect;
81 destination_rect.x = position_x;
82 destination_rect.y = position_y;
83 destination_rect.w = source_rect.w * scale_factor;
84 destination_rect.h = source_rect.h * scale_factor;
85
86 renderer::Vec2f origin_offset{ destination_rect.w * 0.5f, destination_rect.h * 0.5f };
87 renderer::RGBAColor sprite_tint{255, 255, 255, 255};
88
89 try {
90 services.renderer().drawSprite(texture_handle, destination_rect, source_rect, origin_offset, 0.0f, sprite_tint);
91 }
92 catch (...) {
93 if (services._debug) {
94 services.debug().log(asge::debug::Level::Error,
95 "RenderSystem: exception during draw for eid " +
96 std::to_string(entity_id));
97 }
98 }
99 }
100}
