8/**
9 * @file INetwork.hpp
10 * @brief Abstract interface defining the public network API.
11 * @since 0.1
12 * @ingroup network
13 *
14 * @details
15 * This interface provides a unified API for all network backends
16 * (e.g. @ref NullNetwork, @ref NetworkManager). It abstracts away
17 * transport-layer specifics while exposing essential lifecycle
18 * operations: starting, connecting, sending, polling, and shutting down.
19 *
20 * Each concrete implementation must ensure thread safety and
21 * consistent lifecycle semantics.
22 */
23

33/**
34 * @class INetwork
35 * @brief Base interface for all networking implementations.
36 * @ingroup network
37 *
38 * @details
39 * Defines the minimal set of methods needed by the engine to communicate
40 * over the network. Implementations can be:
41 * - Real transports (e.g. UDP, TCP, reliable layer).
42 * - Simulated or null (for headless or test runs).
43 *
44 * Typical usage pattern:
45 * @code
46 * std::unique_ptr<INetwork> net = std::make_unique<UDPNetwork>();
47 * net->startServer(4200);
48 * net->poll([](const std::vector<std::uint8_t>& data) { handlePacket(data); });
49 * @endcode
50 */
51class INetwork {
52 public:
53 /** @brief Virtual destructor for safe polymorphic cleanup. */
54 virtual ~INetwork() = default;
55
56 /**
57 * @brief Start a network server listening on a given port.
58 * @param port Port number to bind (must be > 0).
59 * @return True if the server was successfully started.
60 * @throw asge::error::Error On invalid parameters or I/O failure.
61 */
62 virtual bool startServer(std::uint16_t port) = 0;
63
64 virtual void sendBestEffort(std::uint32_t connection_id,
65 const std::vector<std::uint8_t> &data, int channel = 1) = 0;
66
67 virtual void sendReliable(std::uint32_t connection_id,
68 const std::vector<std::uint8_t> &data, int channel) = 0;
69
70 virtual std::vector<std::uint32_t> connectionIds() const = 0;
71
72 /**
73 * @brief Connect to a remote endpoint.
74 * @param address Target IP or hostname.
75 * @param port Target port.
76 * @return True if the connection setup succeeded.
77 * @throw asge::error::Error On invalid endpoint or failed initialization.
78 */
79 virtual bool connectTo(const std::string& address, std::uint16_t port) = 0;
80
81 /**
82 * @brief Send a data buffer to the remote peer.
83 * @param data Byte payload to send.
84 * @param channel Logical channel index (default: 0).
85 *
86 * @details
87 * Channels may represent separate reliability or priority classes.
88 * The actual behavior depends on the implementation.
89 *
90 * @throw asge::error::Error If called before starting/connecting.
91 */
92 virtual void send(const std::vector<std::uint8_t>& data, int channel = 0) = 0;
93
94 /**
95 * @brief Poll the network for incoming packets.
96 * @param onReceive Callback invoked for each received packet.
97 *
98 * @details
99 * The callback receives a raw byte vector for each complete packet.
100 * Non-blocking by design; returns immediately if no data is available.
101 *
102 * @throw asge::error::Error On fatal network I/O failure.
103 */
104 virtual void poll(std::function<void(const std::vector<std::uint8_t>&)> onReceive) = 0;
105
106 /**
107 * @brief Shut down the network subsystem.
108 *
109 * @details
110 * Closes sockets, terminates background threads (if any),
111 * and releases associated resources. Safe to call multiple times.
112 */
113 virtual void shutdown() = 0;
114};
