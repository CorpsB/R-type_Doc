58/**
59 * @brief Helper template to add a component and destroy the entity on failure.
60 *
61 * @tparam T Component type.
62 * @param coord Reference to the ECS Coordinator.
63 * @param id Entity identifier.
64 * @param component Component instance to add.
65 * @return true if added successfully, false if rollback occurred.
66 */
67template<typename T>
68inline bool add_or_rollback(Coordinator &coord, uint32_t id, const T &component)
69{
70 if (!coord.addComponent<T>(id, component)) {
71 coord.destroyEntity(id);
72 return false;
73 }
74 return true;
75}
