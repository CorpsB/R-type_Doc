165void StorageManager::workerLoop()
166{
167 while (true) {
168 std::pair<std::string, std::function<void(std::vector<uint8_t>)>> task;
169 {
170 std::unique_lock lock(_mutex);
171
172 _cv.wait(lock, [&] { return !_asyncQueue.empty() || !_running; });
173 if (!_running && _asyncQueue.empty()) {
174 break;
175 }
176 task = _asyncQueue.front();
177 _asyncQueue.pop();
178 }
179 try {
180 auto data = _device->read(task.first);
181
182 task.second(data);
183 }
184 catch (const std::exception &exception) {
185 if (_debugger != nullptr) {
186 _debugger->log(
187 debug::Level::Error,
188 std::string("Async read failed: ") + exception.what());
189 }
190 }
191 }
192}
