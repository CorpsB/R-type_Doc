51std::error_code protocol::Protocol::init(const std::string &local_address,
52 uint16_t local_port) noexcept
53{
54 #if defined(_WIN32) || defined(_WIN64)
55 if (!_winsock_started) {
56 WSADATA wsaData;
57 int r = WSAStartup(MAKEWORD(2,2), &wsaData);
58 if (r != 0) return std::make_error_code(std::errc::not_supported);
59 _winsock_started = true;
60 }
61 #endif
62
63 try {
64 boost::system::error_code err;
65
66 _internal_io = std::make_unique<boost::asio::io_context>();
67 _socket = std::make_unique<boost::asio::ip::udp::socket>(*_internal_io);
68
69 if (local_address.empty()) {
70 _socket->open(boost::asio::ip::udp::v4(), err);
71 if (err)
72 return std::make_error_code(std::errc::address_not_available);
73 _socket->bind(boost::asio::ip::udp::endpoint(boost::asio::ip::udp::v4(), local_port), err);
74 if (err)
75 return std::make_error_code(std::errc::address_in_use);
76 } else {
77 boost::asio::ip::address addr = boost::asio::ip::make_address(local_address, err);
78 if (err)
79 return std::make_error_code(std::errc::invalid_argument);
80 _socket->open(addr.is_v4() ? boost::asio::ip::udp::v4() : boost::asio::ip::udp::v6(), err);
81 if (err)
82 return std::make_error_code(std::errc::address_not_available);
83 _socket->bind(boost::asio::ip::udp::endpoint(addr, local_port), err);
84 if (err)
85 return std::make_error_code(std::errc::address_in_use);
86 }
87
88 if (_receive_buffer.size() < MAX_PACKET_SIZE_BYTES) {
89 _receive_buffer.resize(MAX_PACKET_SIZE_BYTES);
90 }
91
92 schedule_receive();
93
94 _initialized.store(true, std::memory_order_release);
95 }
96 catch (...) {
97 return std::make_error_code(std::errc::io_error);
98 }
99
100 return {};
101}

103void protocol::Protocol::shutdown() noexcept
104{
105 if (!_initialized.load(std::memory_order_acquire))
106 return;
107
108 try {
109 if (_socket) {
110 boost::system::error_code err;
111
112 _socket->cancel(err);
113 _socket->close(err);
114 _socket.reset();
115 }
116 } catch (...) {}
117
118 {
119 std::lock_guard<std::mutex> lk(_pending_mutex);
120 _pending_sends.clear();
121 }
122
123 {
124 std::lock_guard<std::mutex> lk(_peer_mutex);
125 _peer_states.clear();
126 }
127
128 {
129 std::lock_guard<std::mutex> qlk(_event_mutex);
130 _event_queue.clear();

149{
150 if (!_internal_io || _running.load())
151 return;
152
153 _internal_io->restart();

155 _work_guard.emplace(_internal_io->get_executor());

166{

168 return;
169
170 if (_work_guard.has_value()) {
171 _work_guard.reset();
172 }
173

175 _internal_io->stop();

214/**
215 * @brief Serialize a packet_t into a contiguous send buffer with piggybacked ACK info.
216 * @param packet The packet object (header fields expected in host byte order).
217 * @param peer_state Optional pointer to the per-peer state used to fill ack/ack_bits.
218 * @return Shared_ptr of [packet_header_t (network byte order)] [payload bytes...].
219 *
220 * @details
221 * The function converts multi-byte header fields to network byte order, inserts
222 * optional piggybacked ACK/ACK_BITS taken from @p peer_state (if non-null), and
223 * appends the payload bytes after the header.
224 */
225std::shared_ptr<std::vector<uint8_t>> serialize_packet_ack(const protocol::packet_t &packet,
226 const protocol::peer_receive_state_t *peer_state)
227{
228 auto buffer = std::make_shared<std::vector<uint8_t>>();
229 buffer->resize(sizeof(protocol::packet_header_t) + packet.payload.size());
230
231 protocol::packet_header_t packet_header;
232 packet_header.message_type = packet.header.message_type;
233 packet_header.flags = static_cast<protocol::flag_t>(packet.header.flags);
234 packet_header.sequence = protocol::utils::to_network_u32(packet.header.sequence);
235 if (peer_state) {
236 packet_header.ack = protocol::utils::to_network_u32(peer_state->last_received);
237 packet_header.ack_bits = protocol::utils::to_network_u32(peer_state->ack_bits);
238 } else {
239 packet_header.ack = 0u;
240 packet_header.ack_bits = 0u;
241 }
242 packet_header.payload_length = htons(static_cast<uint16_t>(packet.payload.size()));
243 packet_header.message_type = htons(packet_header.message_type);
244
245 std::memcpy(buffer->data(), &packet_header, sizeof(packet_header));
246 if (!packet.payload.empty()) {
247 std::memcpy(buffer->data() + sizeof(packet_header), packet.payload.data(),
248 packet.payload.size());
249 }
250 return buffer;
251}

289std::error_code protocol::Protocol::send_packet(const udp::endpoint &remote,
290 uint16_t message_type, flag_t reliability, const byte_buffer_t &payload) noexcept
291{
292 if (!_initialized.load(std::memory_order_acquire)) {
293 return std::make_error_code(std::errc::not_connected);
294 }
295 if (!_socket || !_socket->is_open()) {
296 return std::make_error_code(std::errc::not_connected);
297 }
298
299 uint32_t seq = 0u;
300 packet_t packet;
301 packet.header.message_type = message_type;
302 packet.header.flags = reliability;
303 if (packet.header.flags == flag_t::UNRELIABLE) {
304 packet.header.sequence = 0u;
305 } else {
306 seq = _next_sequence.fetch_add(1u, std::memory_order_relaxed);
307 packet.header.sequence = seq;
308 }
309 packet.payload = payload;
310 packet.remote = remote;
311
312 send_buffered_packet(packet);
313
314 if (packet.header.flags != flag_t::UNRELIABLE) {
315 pending_send_t pending;
316 pending.packet = packet;
317 pending.attempts = 1;
318 pending.last_sent = std::chrono::steady_clock::now();
319
320 {
321 std::lock_guard<std::mutex> lk(_pending_mutex);
322 _pending_sends.emplace(seq, std::move(pending));
323 }
324 }
325 return {};
326}

348void protocol::Protocol::process_pending(steady_timepoint now) noexcept
349{
350 std::vector<uint32_t> to_resend;
351 std::vector<uint32_t> to_fail;
352
353 {
354 std::lock_guard<std::mutex> lk(_pending_mutex);
355
356 for (auto it = _pending_sends.begin(); it != _pending_sends.end(); ++it) {
357 uint32_t seq = it->first;
358 pending_send_t &p = it->second;
359 auto age = std::chrono::duration_cast<std::chrono::milliseconds>(now - p.last_sent).count();
360
361 if (p.attempts >= MAX_RETRANSMIT_ATTEMPTS) {
362 to_fail.push_back(seq);
363 } else if (static_cast<uint32_t>(age) >= RESEND_MS_INTERVAL) {
364 to_resend.push_back(seq);
365 p.attempts += 1;
366 p.last_sent = now;
367 }
368 }
369 }
370
371 // handle errors
372 for (uint32_t seq : to_fail) {
373 std::unique_lock<std::mutex> lk(_pending_mutex);
374 auto it = _pending_sends.find(seq);
375
376 if (it != _pending_sends.end()) {
377 packet_t orig = it->second.packet;
378
379 _pending_sends.erase(it);
380 lk.unlock();
381 emit_transmit_failed(seq, orig);
382 } else {
383 lk.unlock();
384 }
385 }
386
387 // handle retransmit
388 for (uint32_t seq : to_resend) {
389 std::unique_lock<std::mutex> lk(_pending_mutex);
390 auto it = _pending_sends.find(seq);
391
392 if (it != _pending_sends.end()) {
393 packet_t packet = it->second.packet; // copy before unlock
394
395 lk.unlock();
396 send_buffered_packet(packet);
397 } else {
398 lk.unlock();
399 }
400 }

403bool protocol::Protocol::retransmit_sequence(uint32_t sequence) noexcept
404{
405 std::unique_lock<std::mutex> lk(_pending_mutex);
406 auto it = _pending_sends.find(sequence);
407
408 if (it == _pending_sends.end())
409 return false;
410
411 packet_t packet_cpy = it->second.packet;
412
413 if (it->second.attempts >= MAX_RETRANSMIT_ATTEMPTS) {
414 _pending_sends.erase(it);
415 lk.unlock();
416 emit_transmit_failed(sequence, packet_cpy);
417 return false;
418 }
419 it->second.last_sent = std::chrono::steady_clock::now();
420
421 packet_t to_send = packet_cpy;
422 lk.unlock();
423 return send_buffered_packet(to_send);
424}

470/**
471 * @brief Update the peer ack window (last_received / ack_bits) for piggybacking.
472 */
473void update_peer_ack_window(protocol::peer_receive_state_t &state,
474 uint32_t seq) noexcept
475{
476 if (state.last_received == 0u) {
477 state.last_received = seq;
478 state.ack_bits = 0u;
479 return;
480 }
481
482 if (protocol::utils::is_seq_newer(seq, state.last_received)) {
483 uint32_t diff = seq - state.last_received;
484
485 if (diff >= 32u) {
486 state.ack_bits = 0u;
487 } else {
488 // shift left by diff, set the bit for seq-1
489 state.ack_bits = (state.ack_bits << diff) | ((diff >= 1) ? (1u << (diff - 1)) : 0u);
490 }
491 state.last_received = seq;
492 } else {
493 uint32_t dist = state.last_received - seq;
494
495 if (dist >= 1 && dist <= 32) {
496 state.ack_bits |= (1u << (dist - 1));
497 }
498 }
499}

501/**
502 * @brief Flush contiguous buffered packets starting at expected_sequence.
503 *
504 * @details
505 * The function will call enqueue_fn for every flushed packet.
506 * Caller must hold _peer_mutex while modifying state.
507 */
508void flush_buffered_contiguous(protocol::peer_receive_state_t &state,
509 const std::function<void(protocol::packet_t &&)> &enqueue_fn) noexcept
510{
511 while (true) {
512 auto itbuf = state.buffered.find(state.expected_sequence);
513 if (itbuf == state.buffered.end())
514 break;
515
516 enqueue_fn(std::move(itbuf->second.first));
517 state.buffered.erase(itbuf);
518 state.expected_sequence++;
519 state.waiting_since = std::chrono::steady_clock::now();
520 }
521}

523/**
524 * @brief Handle ordered-delivery semantics for incoming packet.
525 * @param state peer state to update
526 * @param packet incoming packet (will be moved into buffer or delivered)
527 * @param now current timepoint
528 * @param enqueue_fn function to call to deliver a packet_t
529 *
530 * @details
531 * - delivers in-order packets immediately (enqueue_fn)
532 * - buffers out-of-order packets
533 * - ignores duplicates
534 * - advances expected_sequence after a long wait (application-driven progress)
535 */
536void handle_ordered_delivery(protocol::peer_receive_state_t &state,
537 protocol::packet_t &&packet, protocol::steady_timepoint now,
538 const std::function<void(protocol::packet_t &&)> &enqueue_fn) noexcept
539{
540 uint32_t seq = packet.header.sequence;
541
542 if (seq == state.expected_sequence) {
543 enqueue_fn(std::move(packet));
544 state.expected_sequence++;
545 state.waiting_since = now;
546 flush_buffered_contiguous(state, enqueue_fn);
547 return;
548 }
549
550 if (protocol::utils::is_seq_newer(seq, state.expected_sequence)) {
551 state.buffered.emplace(seq, std::make_pair(std::move(packet), now));
552 return;
553 }
554
555 auto waited_ms = std::chrono::duration_cast<std::chrono::milliseconds>(now - state.waiting_since).count();
556 if (waited_ms >= static_cast<int64_t>(protocol::RESEND_MS_INTERVAL) * 10) {
557 if (!state.buffered.empty()) {
558 uint32_t lowest = state.buffered.begin()->first;
559
560 if (lowest > state.expected_sequence) {
561 state.expected_sequence = lowest;
562 } else {
563 state.expected_sequence++;
564 }
565 flush_buffered_contiguous(state, enqueue_fn);
566 } else {
567 state.expected_sequence++;
568 }
569 state.waiting_since = now;
570 }
571
572 auto expire_threshold = now - std::chrono::milliseconds(static_cast<int64_t>(protocol::RESEND_MS_INTERVAL) * 100);
573 for (auto it = state.buffered.begin(); it != state.buffered.end(); ) {
574 if (it->second.second < expire_threshold) {
575 it = state.buffered.erase(it);
576 } else {
577 ++it;
578 }
579 }

582void protocol::Protocol::handle_receive(const boost::system::error_code &err,
583 std::size_t bytes_transferred, const boost::asio::ip::udp::endpoint &remote)
584{
585 if (err) {
586 core_event_t ev;
587 ev.type = event_type_t::CONNECTION_LOST;
588 ev.payload = std::monostate{};
589
590 {
591 std::lock_guard<std::mutex> qlk(_event_mutex);
592
593 _event_queue.push_back(std::move(ev));
594 _event_cv.notify_one();
595 }
596
597 if (_socket && _socket->is_open())
598 schedule_receive();
599 return;
600 }
601
602 if (bytes_transferred < sizeof(packet_header_t)) {
603 if (_socket && _socket->is_open())
604 schedule_receive();
605 return;
606 }
607
608 packet_header_t packet_header;
609 std::memcpy(&packet_header, _receive_buffer.data(), sizeof(packet_header));
610
611 packet_t packet;
612 packet.header.message_type = ntohs(packet_header.message_type);
613 packet.header.flags = packet_header.flags;
614 packet.header.sequence = utils::from_network_u32(packet_header.sequence);
615 packet.header.ack = utils::from_network_u32(packet_header.ack);
616 packet.header.ack_bits = utils::from_network_u32(packet_header.ack_bits);
617 packet.header.payload_length = ntohs(packet_header.payload_length);
618
619 std::size_t expected_payload = static_cast<std::size_t>(packet.header.payload_length);
620 std::size_t available = (bytes_transferred > sizeof(packet_header)) ?
621 (bytes_transferred - sizeof(packet_header)) : 0;
622 std::size_t to_copy = std::min(expected_payload, available);
623
624 packet.payload.assign(_receive_buffer.begin() + sizeof(packet_header),
625 _receive_buffer.begin() + sizeof(packet_header) + to_copy);
626 packet.remote = remote;
627
628 steady_timepoint now = std::chrono::steady_clock::now();
629
630 {
631 uint32_t ack = packet.header.ack;
632
633 if (ack != 0) {
634 std::lock_guard<std::mutex> lk(_pending_mutex);
635
636 // remove ack
637 auto it = _pending_sends.find(ack);
638 if (it != _pending_sends.end())
639 _pending_sends.erase(it);
640
641 // remove ack bits
642 for (uint32_t bit = 0; bit < 32; ++bit) {
643 if ((packet.header.ack_bits & (1u << bit)) && bit < ack) {
644 uint32_t seq = ack - 1u - bit;
645 auto it2 = _pending_sends.find(seq);
646
647 if (it2 != _pending_sends.end())
648 _pending_sends.erase(it2);
649 }
650 }
651 }
652 }
653
654 std::string key = protocol::utils::endpoint_to_string(remote);
655 std::vector<packet_t> to_deliver;
656 {
657 std::lock_guard<std::mutex> plk(_peer_mutex);
658 auto &state = _peer_states[key];
659 uint32_t seq = packet.header.sequence;
660
661 update_peer_ack_window(state, seq);
662
663 if (packet.header.flags == flag_t::RELIABLE_ORDERED) {
664 auto enqueue_local = [&to_deliver](packet_t &&p){
665 to_deliver.push_back(std::move(p));
666 };
667 handle_ordered_delivery(state, std::move(packet), now, enqueue_local);
668 } else {
669 to_deliver.push_back(std::move(packet));
670 }
671 }
672 for (auto &p : to_deliver)
673 enqueue_packet_event(std::move(p));
674
675 if (_socket && _socket->is_open())
676 schedule_receive();
677}
