106void Connection::updateRtt(std::chrono::milliseconds sample)
107{
108 if (sample <= std::chrono::milliseconds{0}) {
109 sample = RTT_MIN;
110 }
111
112 if (sample < RTT_MIN) {
113 sample = RTT_MIN;
114 }
115 if (sample > RTT_MAX) {
116 sample = RTT_MAX;
117 }
118
119 if (_rtt.count() == 0) {
120 _rtt = sample;
121 } else {
122 auto smoothed_rtt = std::chrono::milliseconds{(_rtt.count() * 7 + sample.count()) / 8};
123 if (smoothed_rtt < RTT_MIN) {
124 smoothed_rtt = RTT_MIN;
125 }
126 if (smoothed_rtt > RTT_MAX) {
127 smoothed_rtt = RTT_MAX;
128 }
129 _rtt = smoothed_rtt;
130 }
131}
