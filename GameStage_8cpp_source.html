23GameStage::GameStage(asge::EngineCore &engine)
24{
25 init(engine);
26 engine.services()._gameRunning.store(true);
27 engine.services().renderer().clear(0, 0, 0, 0);
28
29 std::string &addr = engine.services().config().addr;
30 unsigned short port = engine.services().config().port;
31
32 while (!engine.connectToServer(addr, port)) {
33 std::cerr << "[client/main] connect to server " << addr
34 << ":" << port << " fail. retrying..." << std::endl;
35
36 std::this_thread::sleep_for(std::chrono::seconds(1));
37 }
38 std::cout << "[client/main] connected to " << addr << ":" << port << std::endl;
39}
