10 * @brief Implements the entity creation logic and component registration.
11 * @ingroup ecs
12 *
13 * @details
14 * This file defines the functions declared in `EntityFactory.hpp`.
15 * It contains helper routines to register ECS components with the
16 * `Coordinator` and to create archetypes such as player, enemy,
17 * and projectile entities with pre-defined attributes.
18 *
19 * Each creation function uses `add_or_rollback()` to ensure atomic
20 * component setup: if adding a component fails, the entity is safely
21 * destroyed to prevent dangling or partially built entities.
22 */

40bool rtype::ecs::factory::set_skin(Coordinator &coord,

42 const std::optional<sprite_attr_t> &sprite)
43{
44 if (box.has_value()) {
45 if (!add_or_rollback(coord, id, box.value()))

47 }
48 if (sprite.has_value()) {

75uint32_t rtype::ecs::factory::create_player(Coordinator &coord,
76 const entity_schema &schema)
77{
78 const uint32_t id = coord.createEntity();
79
80 if (id == INVALID_ID)
81 return INVALID_ID;
82
83 health_attr_t health { 12, 12 };
84 score_attr_t score { 0, 0 };
85 name_attr_t name { "Player" };
86
87 position_attr_t pos {schema.x, schema.y};
88 direction_attr_t dir { 0.0, 1.0 };
89 speed_attr_t spd { 300 };
90
91 damage_attr_t dmg { 1 };
92 frequency_attr_t freq { 1.0f, 0.0f };
93 hitbox_attr_t box { 17 * 3, 33 * 3 };
94 sprite_attr_t sprite { "PLAYER", Vector2{33 * 3, 17 * 3}, Rectangle{66, 0, 33, 17} };
95
96 if (!add_or_rollback(coord, id, name) ||
97 !add_or_rollback(coord, id, score) ||
98 !add_or_rollback(coord, id, health) ||
99 !set_movement(coord, id, pos, dir, spd) ||
100 !set_weapon(coord, id, dmg, freq) ||
101 !set_skin(coord, id, box, sprite))
102 return INVALID_ID;
103 return id;
104}

156uint32_t rtype::ecs::factory::create_entity(Coordinator &coord,
157 const entity_schema &schema)
158{
159 switch (schema.entity_type) {
160 case protocol::game::entity_type_t::PLAYER:
161 return create_player(coord, schema);
162 case protocol::game::entity_type_t::ENEMY:
163 return create_enemy(coord, schema);
164 case protocol::game::entity_type_t::PROJECTILE:
165 return create_projectile(coord, schema);
166 default:
167 return INVALID_ID;
168 }
169}
