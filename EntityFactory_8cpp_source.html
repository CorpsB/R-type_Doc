83 std::optional<asge::components::hitbox_attr_t> parseHitboxComponent(const nlohmann::json &json)
84 {
85 if (!json.is_object()) {
86 return std::nullopt;
87 }
88 asge::components::hitbox_attr_t hitbox {};
89 auto vertical = getOptionalUnsigned(json, "v");
90 auto horizontal = getOptionalUnsigned(json, "h");
91 if (vertical && horizontal) {
92 hitbox.height = *vertical;
93 hitbox.width = *horizontal;
94 return hitbox;
95 }
96 auto width = getOptionalUnsigned(json, "width");
97 if (width) {
98 hitbox.width = *width;
99 hitbox.height = getOptionalUnsigned(json, "height").value_or(hitbox.width);
100 return hitbox;
101 }
102 return std::nullopt;
103 }

259uint32_t asge::script::EntityFactory::createFromTemplate(
260 Services &services,
261 const ScriptHandle &templateHandle,
262 const std::optional<double> &positionX,
263 const std::optional<double> &positionY,
264 const std::optional<uint32_t> &entity_id)
265{
266 if (!services._world) {
267 if (services._debug) {
268 services.debug().log(
269 asge::debug::Level::Warning,
270 "EntityFactory::createFromTemplate: missing world service.");
271 }
272 return asge::world::INVALID_ID;
273 }
274
275 EntityBuilder builder(services);
276 builder.withTemplateId(templateHandle.id);
277
278 const nlohmann::json &components = templateHandle.components;
279
280 if (!components.is_null()) {
281 if (components.contains("name")) {
282 if (auto nameAttributes = parseNameComponent(components["name"])) {
283 builder.withName(*nameAttributes);
284 } else if (auto nameAttributes = parseNameComponent(components)) {
285 builder.withName(*nameAttributes);
286 }
287 }
288
289 if (components.contains("hitbox")) {
290 if (auto hitboxAttributes = parseHitboxComponent(components["hitbox"])) {
291 builder.withHitbox(*hitboxAttributes);
292 }
293 }
294
295 if (components.contains("sprite")) {
296 if (auto spriteAttributes = parseSpriteComponent(components["sprite"])) {
297 builder.withSprite(*spriteAttributes);
298 }
299 }
300
301 if (components.contains("speed")) {
302 if (auto speedAttributes = parseSpeedComponent(components["speed"])) {
303 builder.withSpeed(*speedAttributes);
304 }
305 }
306
307 if (components.contains("position")) {
308 if (auto positionAttributes = parsePositionComponent(components["position"])) {
309 builder.withPosition(*positionAttributes);
310 }
311 }
312
313 if (components.contains("direction")) {
314 if (auto directionAttributes = parseDirectionComponent(components["direction"])) {
315 builder.withDirection(*directionAttributes);
316 }
317 }
318
319 if (components.contains("health") || components.contains("max_health")) {
320 if (components.contains("health") && components["health"].is_number()) {
321 asge::components::health_attr_t healthAttributes {};
322 healthAttributes.health = components["health"].get<std::uint32_t>();
323 healthAttributes.max_health =
324 components.contains("max_health") && components["max_health"].is_number()
325 ? components["max_health"].get<std::uint32_t>()
326 : healthAttributes.health;
327 builder.withHealth(healthAttributes);
328 } else if (auto healthAttributes = parseHealthComponent(components)) {
329 builder.withHealth(*healthAttributes);
330 }
331 }
332
333 if (components.contains("damage")) {
334 if (auto damageAttributes = parseDamageComponent(components["damage"])) {
335 builder.withDamage(*damageAttributes);
336 }
337 }
338
339 if (components.contains("frequency") ||
340 components.contains("cooldown_seconds") ||
341 components.contains("rate_sec")) {
342 if (components.contains("frequency") && components["frequency"].is_number()) {
343 asge::components::frequency_attr_t frequencyAttributes {};
344 frequencyAttributes.cooldown_seconds = static_cast<float>(components["frequency"].get<double>());
345 builder.withFrequency(frequencyAttributes);
346 } else if (auto frequencyAttributes = parseFrequencyComponent(components)) {
347 builder.withFrequency(*frequencyAttributes);
348 }
349 }
350 }
351
352 if (positionX.has_value() || positionY.has_value()) {
353 asge::components::position_attr_t positionAttributes {};
354 if (components.contains("position")) {
355 if (auto templatePosition = parsePositionComponent(components["position"])) {
356 positionAttributes = *templatePosition;
357 }
358 }
359 if (positionX.has_value()) {
360 positionAttributes.x = static_cast<float>(*positionX);
361 }
362 if (positionY.has_value()) {
363 positionAttributes.y = static_cast<float>(*positionY);
364 }
365 builder.withPosition(positionAttributes);
366 }
367
368 uint32_t entityId = builder.build(entity_id);
369 if (entityId != asge::world::INVALID_ID) {
370 try {
371 services.scripts().notifyEntityCreated(entityId, templateHandle.id);
372 }
373 catch (...) {
374 }
375 }
376 return entityId;
