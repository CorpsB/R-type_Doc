14/**
15 * @file PluginManager.cpp
16 * @brief Implementation of the PluginManager class.
17 * @since 0.1
18 * @ingroup plugin
19 * @see PluginManager, PluginLoader, PluginHandle
20 *
21 * @details
22 * This file defines the runtime logic of the @ref PluginManager. It handles the
23 * complete plugin lifecycle — loading, unloading, and retrieval — while
24 * delegating low-level dynamic library management to @ref PluginLoader.
25 *
26 * Each plugin is uniquely identified by a string ID and tracked internally in
27 * a registry. Upon destruction, the manager automatically calls @ref unloadAll()
28 * to ensure that all active plugins are properly released.
29 *
30 * Exceptions thrown during plugin operations are wrapped in
31 * @ref asge::error::Error and logged via the provided @ref debug::IDebug
32 * interface.
33 */

51std::shared_ptr<PluginHandle> PluginManager::loadPlugin(const std::string &id, const std::string &path)
52{
53 try {
54 if (id.empty() || path.empty()) {
55 throw asge::error::Error(
56 asge::error::ErrorType::Plugin,
57 "PluginManager::loadPlugin",
58 "Invalid plugin ID or path (empty string)",
59 asge::error::ErrorSeverity::Recoverable
60 );
61 }
62
63 if (_plugin_registry.find(id) != _plugin_registry.end()) {
64 if (_debug_interface) {
65 _debug_interface->log(debug::Level::Warning, "PluginManager::loadPlugin - Plugin already loaded: " + id);
66 }
67 return _plugin_registry[id];
68 }
69
70 auto plugin_instance = _plugin_loader->load(path);
71
72 if (!plugin_instance) {
73 if (_debug_interface) {
74 _debug_interface->log(debug::Level::Error, "PluginManager::loadPlugin - Failed to load: " + path);
75 }
76 throw asge::error::Error(
77 asge::error::ErrorType::Plugin,
78 "PluginManager::loadPlugin",
79 "Failed to load plugin: " + path,
80 asge::error::ErrorSeverity::Recoverable
81 );
82 }
83
84 auto plugin_handle = std::make_shared<PluginHandle>(id, plugin_instance, path);
85 _plugin_registry[id] = plugin_handle;
86
87 if (_debug_interface) {
88 _debug_interface->log(debug::Level::Info, "Plugin loaded successfully: " + id);
89 }
90 return plugin_handle;
91 } catch (const asge::error::Error &) {
92 throw;
93 } catch (const std::exception &e) {
94 if (_debug_interface) {
95 _debug_interface->log(debug::Level::Error, std::string("PluginManager::loadPlugin - Exception: ") + e.what());
96 }
97 throw asge::error::Error(
98 asge::error::ErrorType::Plugin,
99 "PluginManager::loadPlugin",
100 std::string("Unexpected error while loading plugin: ") + e.what(),
101 asge::error::ErrorSeverity::Recoverable

106bool PluginManager::unloadPlugin(const std::string &id)
107{
108 try {
109 auto registry_iterator = _plugin_registry.find(id);
110
111 if (registry_iterator == _plugin_registry.end()) {
112 if (_debug_interface) {
113 _debug_interface->log(debug::Level::Warning, "PluginManager::unloadPlugin - Unknown plugin: " + id);
114 }
115 return false;
116 }
117 registry_iterator->second->unload();
118 _plugin_registry.erase(registry_iterator);
119 if (_debug_interface) {
120 _debug_interface->log(debug::Level::Info, "Plugin unloaded: " + id);
121 }
122 return true;
123 } catch (const std::exception &e) {
124 if (_debug_interface) {
125 _debug_interface->log(debug::Level::Error, std::string("PluginManager::unloadPlugin - ") + e.what());
126 }
127 throw asge::error::Error(
128 asge::error::ErrorType::Plugin,
129 "PluginManager::unloadPlugin",
130 std::string("Failed to unload plugin: ") + e.what(),
131 asge::error::ErrorSeverity::Recoverable
132 );
133 }
134}

136void PluginManager::unloadAll()
137{
138 try {
139 for (auto &registry_entry : _plugin_registry) {
140 try {
141 registry_entry.second->unload();
142 } catch (const std::exception &e) {
143 if (_debug_interface) {
144 _debug_interface->log(
145 debug::Level::Error,
146 std::string("PluginManager::unloadAll - Failed to unload plugin: ") + registry_entry.first + " - "
147 + e.what()
148 );
149 }
150 }
151 }
152 _plugin_registry.clear();
153 _plugin_loader->unloadAll();
154 } catch (const std::exception &e) {
155 throw asge::error::Error(
156 asge::error::ErrorType::Plugin,
157 "PluginManager::unloadAll",
158 std::string("Failed to unload all plugins: ") + e.what(),
159 asge::error::ErrorSeverity::Recoverable
160 );
161 }
162}
