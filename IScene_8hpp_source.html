35/**
36 * @class IScene
37 * @brief Abstract interface for scene stack management.
38 *
39 * @details
40 * Provides basic operations for a scene stack: push, pop, replace, update, draw,
41 * and flush transitions. Also provides templated helpers for creating and adding
42 * new scenes in a single call.
43 */
44class IScene {
45 public:
46 /**
47 * @brief Virtual destructor for safe polymorphic deletion.
48 */
49 virtual ~IScene() = default;
50
51 /**
52 * @brief Push a new scene onto the stack.
53 * @param scene Scene pointer to push.
54 */
55 virtual void push(ScenePtr scene) = 0;
56
57 /**
58 * @brief Pop the top scene from the stack.
59 */
60 virtual void pop() = 0;
61
62 /**
63 * @brief Replace the top scene with a new one.
64 * @param scene Scene pointer to replace the top with.
65 */
66 virtual void replace(ScenePtr scene) = 0;
67
68 /**
69 * @brief Update the current active scene.
70 * @param dt Delta time since last update.
71 */
72 virtual void update(double dt) = 0;
73
74 /**
75 * @brief Draw the current active scene.
76 */
77 virtual void draw() const = 0;
78
79 /**
80 * @brief Execute all pending push/pop/replace operations.
81 */
82 virtual void flushTransitions() = 0;
83
84 /**
85 * @brief Push a new scene by constructing it in-place.
86 * @tparam T Scene type to create.
87 * @tparam Args Constructor argument types.
88 * @param args Constructor arguments.
89 */
90 template<typename T, typename... Args>
91 void pushNew(Args&&... args)
92 {
93 push(std::make_unique<T>(std::forward<Args>(args)...));
94 }
95
96 /**
97 * @brief Replace the top scene by constructing a new one in-place.
98 * @tparam T Scene type to create.
99 * @tparam Args Constructor argument types.
100 * @param args Constructor arguments.
101 */
102 template<typename T, typename... Args>
103 void replaceNew(Args&&... args)
104 {
105 replace(std::make_unique<T>(std::forward<Args>(args)...));
106 }
107
108 /**
109 * @brief Check if the scene stack is empty.
110 * @return true if empty, false otherwise.
111 */
112 virtual bool empty() const = 0;
113};
