39class IStorage
40{
41 public:
42 virtual ~IStorage() = default;
43
44 virtual void clearDebug() = 0;
45
46 /**
47 * @brief Checks if a file exists at the given path.
48 * @param path Path to the file.
49 * @return true if the file exists, false otherwise.
50 */
51 virtual bool exists(const std::string &path) const = 0;
52
53 /**
54 * @brief Reads the contents of a file.
55 * @param path Path to the file.
56 * @return A vector of bytes representing the file contents.
57 * @throw asge::error::Error on read failure.
58 */
59 virtual std::vector<uint8_t> readFile(const std::string &path) = 0;
60
61 /**
62 * @brief Writes data to a file.
63 * @param path Path to the destination file.
64 * @param data Data buffer to write.
65 * @throw asge::error::Error if the write operation fails.
66 */
67 virtual void writeFile(const std::string &path, const std::vector<uint8_t> &data) = 0;
68
69 /**
70 * @brief Reads a file asynchronously.
71 * @param path Path to the file.
72 * @param callback Callback invoked with file data upon completion.
73 */
74 virtual void readAsync(const std::string &path, std::function<void(std::vector<uint8_t>)> callback) = 0;
75
76 /**
77 * @brief Lists all files in a directory.
78 * @param directory Directory to list.
79 * @return A vector containing the names or paths of files.
80 */
81 virtual std::vector<std::string> list(const std::string &directory) const = 0;
