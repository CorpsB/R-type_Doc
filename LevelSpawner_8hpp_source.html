21/**
22 * @file LevelSpawner.hpp
23 * @brief Progressive level spawner driven by script-defined map assets.
24 * @since 1.0
25 * @ingroup scripting
26 *
27 * @details
28 * `LevelSpawner` consumes a `MapAsset` (retrieved from the script manager) and
29 * instantiates entities at predefined distances, using the script-side
30 * `EntityFactory`. It supports two modes:
31 * - Progressive: call `update(current_progress)` each frame; all spawns whose
32 * distance threshold is reached are created.
33 * - Immediate: call `spawnAll()` to synchronously create every remaining spawn.
34 *
35 * The spawner is stateless with respect to the world itself; it tracks only
36 * the next spawn index, last progress value, and the map identifier. Errors
37 * and abnormal conditions (unknown archetype, factory failure) are reported
38 * as warnings through the debug service without throwing.
39 *
40 * Typical usage:
41 * @code
42 * // after scripts have been loaded and world initialized
43 * asge::script::LevelSpawner spawner(services, "level_1");
44 * if (spawner.valid()) {
45 * // in the main loop
46 * spawner.update(distance_travelled);
47 * }
48 * @endcode
49 */
50
51/**
52 * @class LevelSpawner
53 * @brief Spawns entities from a map's spawn table as the player advances.
54 *
55 * @details
56 * The constructor loads and sorts the map's spawn list by ascending distance.
57 * Each call to `update()` advances the spawn cursor, creating all spawns with
58 * `distance <= current_progress`. Calling `spawnAll()` forces creation of all
59 * remaining spawns in order. The `reset()` method resets the cursor and last
60 * progress, allowing the same map to be processed again (e.g., on reload).
61 */
62class LevelSpawner {
63 public:
64 /**
65 * @brief Constructs a spawner bound to a specific map.
66 * @param services Engine services used to access scripts, world, network and debug.
67 * @param map_id Identifier of the scripted map asset to load.
68 *
69 * @post `valid()` indicates whether the map was successfully loaded.
70 * @warning The constructor never throws; it logs warnings instead.
71 */
72 LevelSpawner(Services &services, const std::string &map_id);
73
74 // non-copy
75 /**
76 * @brief Deleted copy constructor.
77 * @details `LevelSpawner` instances are not copyable.
78 */
79 LevelSpawner(const LevelSpawner&) = delete;
80
81 /**
82 * @brief Deleted copy assignment.
83 * @details `LevelSpawner` instances are not copy-assignable.
84 */
85 LevelSpawner& operator=(const LevelSpawner&) = delete;
86
87 /**
88 * @brief Progressively spawns entries whose distance has been reached.
89 * @param current_progress Current travel/progress value in the same unit as the map distances.
90 *
91 * @details
92 * For every spawn whose `distance <= current_progress`, the spawner:
93 * 1. Resolves an optional position override from the spawn JSON,
94 * 2. Looks up the template (`archetype`) via the script manager,
95 * 3. Creates the entity through `EntityFactory`,
96 * 4. Notifies the network layer (`EngineProtocol`) of the new entity.
97 *
98 * Unknown archetypes, factory failures, and exceptions are logged and
99 * skipped without aborting the loop. If `current_progress` goes
100 * backwards, the spawner will not respawn already-processed entries.
101 */
102 void update(std::uint64_t current_progress);
103
104 /**
105 * @brief Synchronously spawns all remaining entries regardless of progress.
106 * @details
107 * Iterates the remaining spawn list from the current cursor and applies
108 * the same creation process as `update()`. This function is synchronous.
109 */
110 void spawnAll();
111
112 /**
113 * @brief Resets internal cursor and last progress.
114 * @details
115 * After `reset()`, the spawner behaves as if newly constructed (for the
116 * same map), and will re-process all spawns on subsequent `update()`.
117 */
118 void reset();
119
120 /**
121 * @brief Indicates whether the target map loaded successfully.
122 * @return `true` if the constructor resolved and sorted the spawns; `false` otherwise.
123 */
124 bool valid() const noexcept { return _valid; }
125
126 /**
127 * @brief Returns the configured map identifier.
128 * @return Constant reference to the map id string.
129 */
130 const std::string& mapId() const noexcept { return _map_id; }
131
132 private:
133 /// @brief Reference to engine services (not owned).
134 Services &_S;
135 /// @brief Scripted map identifier associated with this spawner.
136 std::string _map_id;
137 /// @brief True if the map asset was resolved and spawns sorted.
138 bool _valid = false;
139
140 /// @brief Spawn entries sorted by ascending distance.
141 std::vector<asge::script::MapSpawn> _sorted_spawns;
142 /// @brief Index of the next spawn to evaluate (cursor into @ref _sorted_spawns).
143 size_t _next_index = 0;
144 /// @brief Last progress value observed by @ref update (monotonic non-strict).
145 std::uint64_t _last_progress = 0;
146};
