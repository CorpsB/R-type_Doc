27void Sprite::update()
28{
29 auto &manager = GameManager::instance();
30 auto &settings = manager.gameSettings();
31
32 const float scale = settings.getScale();
33
34 for (auto &instance : _instances) {
35 instance.realSize = { instance.size.x * scale, instance.size.y * scale };
36 instance.realPos = { (instance.position.x - instance.size.x / 2.0f) * scale, (instance.position.y - instance.size.y / 2.0f) * scale };
37 }
38}

40void Sprite::draw() const
41{
42 auto &manager = GameManager::instance();
43 auto &resourceManager = manager.resourceManager();
44 const Texture2D &texture = resourceManager.getTexture(_textureTag);
45
46 Rectangle source = _textureRect.width > 0 && _textureRect.height > 0 ? _textureRect : Rectangle{ 0, 0, static_cast<float>(texture.width), static_cast<float>(texture.height) };
47
48 for (const auto &instance : _instances) {
49 Rectangle dest = { instance.realPos.x, instance.realPos.y, instance.realSize.x, instance.realSize.y };
50
51 #ifdef DEBUG
52 DrawRectangleLinesEx(dest, 2.0f, RED);
53 #endif
54
55 DrawTexturePro(texture, source, dest, Vector2{0, 0}, 0.0f, WHITE);
