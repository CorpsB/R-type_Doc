29void Sprite::update()
30{
31 auto &settings = GameSettings::getInstance();
32
33 const float scale = settings.getScale();
34
35 _realSize = {_size.x * scale, _size.y * scale};
36 _realPos = {(_position.x - _size.x / 2.0f) * scale, (_position.y - _size.y / 2.0f) * scale};
37}

39void Sprite::draw() const
40{
41 auto &resourceManager = ResourceManager::getInstance();
42 const Texture2D &texture = resourceManager.getTexture(_textureTag);
43 Rectangle source = { 0, 0, static_cast<float>(texture.width), static_cast<float>(texture.height) };
44 Rectangle dest = { _realPos.x, _realPos.y, _realSize.x, _realSize.y };
45
46 if (texture.id == 0) {
47 std::cerr << "Texture invalid: " << _textureTag << std::endl;
48 return;
49 }
50
51 #ifdef DEBUG
52 DrawRectangleLinesEx(
53 Rectangle{_realPos.x, _realPos.y, _realSize.x, _realSize.y},
54 2.0f,
55 RED
56 );
57 #endif
58
59 DrawTexturePro(texture, source, dest, Vector2{0, 0}, 0.0f, WHITE);
60}
