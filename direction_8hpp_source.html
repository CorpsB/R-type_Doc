39/**
40 * @brief Normalizes a direction vector in place.
41 * @param dx Horizontal component to normalize.
42 * @param dy Vertical component to normalize.
43 *
44 * @details
45 * The function guards against zero-length vectors by defaulting to `(1, 0)`.
46 * Keeping the helper inline allows usage from multiple translation units
47 * without requiring a separate implementation file.
48 */
49inline void normalize_direction(float &dx, float &dy)
50{
51 const float len = std::sqrt(dx * dx + dy * dy);
52 if (len < 1e-6f) {
53 dx = 1.0f;
54 dy = 0.0f;
55 } else {
56 dx /= len;
57 dy /= len;
58 }
59}
