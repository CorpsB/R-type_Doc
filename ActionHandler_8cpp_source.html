27 * @brief Extracts the first whitespace-separated token from a string.
28 */
29 std::string extractFirstToken(const std::string &text)
30 {
31 std::istringstream stream(text);
32 std::string token;
33 if (!(stream >> token)) {
34 return {};
35 }
36 return token;

39 /**
40 * @brief Parses key:value pairs from a whitespace-delimited string.
41 */
42 std::unordered_map<std::string, std::string> parseKeyValueArguments(const std::string &text)
43 {
44 std::unordered_map<std::string, std::string> arguments;
45 std::istringstream stream(text);
46 std::string token;
47 while (stream >> token) {
48 auto delimiterPosition = token.find(':');
49 if (delimiterPosition == std::string::npos) {
50 continue;
51 }
52 std::string key = token.substr(0, delimiterPosition);
53 std::string value = token.substr(delimiterPosition + 1);
54 boost::algorithm::trim(key);
55 boost::algorithm::trim(value);
56 arguments[key] = value;
57 }
58 return arguments;
59 }

61 /**
62 * @brief Resolves an argument value while supporting special `self` references.
63 */
64 uint32_t resolveArgument(
65 const std::unordered_map<std::string, std::string> &arguments,
66 const std::string &key,
67 uint32_t fallback,
68 bool allowSelf,
69 const std::string &rawInput)
70 {
71 auto iterator = arguments.find(key);
72 if (iterator == arguments.end()) {
73 return asge::world::INVALID_ID;
74 }
75
76 if (allowSelf && iterator->second == "self") {
77 return fallback;
78 }
79
80 try {
81 return static_cast<uint32_t>(std::stoul(iterator->second));
82 }
83 catch (...) {
84 std::cerr << "ActionHandler::resolveArgument failed to parse '" << rawInput << "'" << std::endl;
85 return asge::world::INVALID_ID;
86 }
87 }

89 /**
90 * @brief Parses coordinate arguments, optionally honouring the current entity id.
91 */
92 asge::script::ActionHandler::ParsedXY parseCoordinateArguments(
93 const std::string &text, bool allowSelf, uint32_t selfIdentifier)
94 {
95 auto arguments = parseKeyValueArguments(text);
96 const uint32_t xValue = resolveArgument(arguments, "x", selfIdentifier, allowSelf, text);
97 const uint32_t yValue = resolveArgument(arguments, "y", selfIdentifier, allowSelf, text);
98
99 asge::script::ActionHandler::ParsedXY coordinates {};
100 coordinates.x = xValue;
101 coordinates.y = yValue;
102 return coordinates;
103 }

109void asge::script::ActionHandler::processAudioAction(
110 const std::string &command, Services &services, const std::string &remainder)
111{
112 if (!services._audio) {
113 if (services._debug) {
114 services.debug().log(
115 asge::debug::Level::Warning,
116 "ActionHandler::audio command: no audio service available.");
117 }
118 return;
119 }
120
121 if (command == "play") {
122 std::string audioIdentifier = extractFirstToken(remainder);
123 if (audioIdentifier.empty()) {
124 if (services._debug) {
125 services.debug().log(
126 asge::debug::Level::Warning,
127 "ActionHandler::audio play: missing audio id.");
128 }
129 return;
130 }
131
132 try {
133 std::optional<std::string> songPath;
134 std::optional<audio::PlayOptions> songOptions;
135
136 auto *audioTemplate = services.scripts().getAudio(audioIdentifier);
137 if (audioTemplate != nullptr) {
138 songPath.emplace(
139 services.scripts().getAssetsRoot() + "/" + audioTemplate->link);
140 songOptions.emplace(audio::PlayOptions {
141 static_cast<float>(audioTemplate->volume), audioTemplate->loop, 1.0F });
142 }
143
144 services.audio().playSong(
145 audioIdentifier, services.storage(), songOptions, songPath);
146 }
147 catch (...) {
148 if (services._debug) {
149 services.debug().log(
150 asge::debug::Level::Warning,
151 "ActionHandler::audio play: exception when playing '" + audioIdentifier + "'.");
152 }
153 }
154 return;
155 }
156
157 if (command == "stop") {
158 std::string audioIdentifier = extractFirstToken(remainder);
159 if (audioIdentifier.empty()) {
160 if (services._debug) {
161 services.debug().log(
162 asge::debug::Level::Warning,
163 "ActionHandler::audio stop: missing audio id.");
164 }
165 return;
166 }
167
168 try {
169 services.audio().stopSong(audioIdentifier);
170 }
171 catch (...) {
172 if (services._debug) {
173 services.debug().log(
174 asge::debug::Level::Warning,
175 "ActionHandler::audio stop: exception when stopping '" + audioIdentifier + "'.");
176 }
177 }
178 return;
179 }
180
181 if (command == "register") {
182 std::istringstream stream(remainder);
183 std::string audioIdentifier;
184 if (!(stream >> audioIdentifier)) {
185 if (services._debug) {
186 services.debug().log(
187 asge::debug::Level::Warning,
188 "ActionHandler::audio register: missing audio id.");
189 }
190 return;
191 }
192
193 std::string path;
194 std::getline(stream, path);
195 boost::algorithm::trim(path);
196 if (path.empty()) {
197 if (services._debug) {
198 services.debug().log(
199 asge::debug::Level::Warning,
200 "ActionHandler::audio register: missing path for audio id '" + audioIdentifier + "'.");
201 }
202 return;
203 }
204
205 try {
206 services.audio().createEntry(audioIdentifier, path);
207 if (services._debug) {
208 services.debug().log(
209 asge::debug::Level::Info,
210 "ActionHandler::audio register: registered '" + audioIdentifier + "' -> " + path);
211 }
212 }
213 catch (...) {
214 if (services._debug) {
215 services.debug().log(
216 asge::debug::Level::Warning,
217 "ActionHandler::audio register: exception while registering '" + audioIdentifier + "'.");
218 }
219 }
220 return;
221 }
222
223 if (services._debug) {
224 services.debug().log(
225 asge::debug::Level::Warning,
226 "ActionHandler::processAudioAction unknown audio command '" + command + "'.");
227 }
228}

233void asge::script::ActionHandler::processWorldAction(
234 const std::string &command,
235 Services &services,
236 const std::string &remainder,
237 const asge::script::QueuedAction &action)
238{
239 if (command == "create") {
240 if (services.config().runMode == asge::EngineRunMode::Client)
241 return;
242
243 std::istringstream stream(remainder);
244 std::string templateIdentifier;
245 if (!(stream >> templateIdentifier)) {
246 if (services._debug) {
247 services.debug().log(
248 asge::debug::Level::Warning,
249 "ActionHandler::world create: missing template id in action '" + action.action + "'.");
250 }
251 return;
252 }
253
254 const auto *templateHandle = services.scripts().getTemplate(templateIdentifier);
255 if (templateHandle == nullptr) {
256 if (services._debug) {
257 services.debug().log(
258 asge::debug::Level::Warning,
259 "ActionHandler::world create: unknown template '" + templateIdentifier + "'.");
260 }
261 return;
262 }
263
264 std::string rawArguments;
265 std::getline(stream, rawArguments);
266 boost::algorithm::trim(rawArguments);
267 auto coordinates = parseCoordinateArguments(rawArguments, false, 0);
268 if (coordinates.x == asge::world::INVALID_ID || coordinates.y == asge::world::INVALID_ID) {
269 if (services._debug) {
270 services.debug().log(
271 asge::debug::Level::Warning,
272 "ActionHandler::world create: failed parsing coordinates for '" + action.action + "'. Continuing without coordinates.");
273 }
274 }
275
276 std::optional<double> overrideX = coordinates.x;
277 std::optional<double> overrideY = coordinates.y;
278
279 try {
280 uint32_t entityId = asge::script::EntityFactory::createFromTemplate(
281 services, *templateHandle, overrideX, overrideY);
282 asge::network::EngineProtocol::sendEntityCreate(services, entityId);
283
284 if (entityId == 0) {
285 if (services._debug) {
286 services.debug().log(
287 asge::debug::Level::Warning,
288 "ActionHandler::world create: EntityFactory returned 0 for template '" + templateIdentifier + "'.");
289 }
290 return;
291 }
292
293 if (services._debug) {
294 services.debug().log(
295 asge::debug::Level::Info,
296 "ActionHandler::world create: spawned '" + templateIdentifier + "' eid=" + std::to_string(entityId));
297 }
298 }
299 catch (...) {
300 if (services._debug) {
301 services.debug().log(
302 asge::debug::Level::Warning,
303 "ActionHandler::world create: exception while creating entity from template '" + templateIdentifier + "'.");
304 }
305 }
306 return;
307 }
308
309 if (command == "destroy") {
310 if (services.config().runMode == asge::EngineRunMode::Client)
311 return;
312
313 auto arguments = parseKeyValueArguments(remainder);
314 uint32_t entityId = resolveArgument(arguments, "id", action.entity_id, true, remainder);
315 if (entityId == asge::world::INVALID_ID) {
316 if (services._debug) {
317 services.debug().log(
318 asge::debug::Level::Warning,
319 "ActionHandler::world destroy: invalid entity id '" + remainder + "'.");
320 }
321 return;
322 }
323
324 try {
325 if (services._scripts && services._world) {
326 auto *templateComponent = services.world().getComponent<components::template_attr_t>(entityId);
327
328 if (templateComponent != nullptr) {
329 services.scripts().notifyEntityDestroyed(entityId, templateComponent->template_id);
330 } else if (services._debug) {
331 services.debug().log(
332 asge::debug::Level::Warning,
333 "ActionHandler::world destroy: fail to notify eid=" + std::to_string(entityId));
334 }
335 }
336
337 services.world().destroyEntity(entityId);
338 network::EngineProtocol::sendEntityDestroy(services, entityId);
339 if (services._debug) {
340 services.debug().log(
341 asge::debug::Level::Info,
342 "ActionHandler::world destroy: destroyed eid=" + std::to_string(entityId));
343 }
344 }
345 catch (...) {
346 if (services._debug) {
347 services.debug().log(
348 asge::debug::Level::Warning,
349 "ActionHandler::world destroy: failed to destroy eid=" + std::to_string(entityId));
350 }
351 }
352 return;
353 }
354
355 if (command == "teleport") {
356 auto arguments = parseKeyValueArguments(remainder);
357 uint32_t entityId = resolveArgument(arguments, "id", action.entity_id, true, remainder);
358 if (entityId == asge::world::INVALID_ID) {
359 if (services._debug) {
360 services.debug().log(
361 asge::debug::Level::Warning,
362 "ActionHandler::world teleport: invalid entity id '" + remainder + "'.");
363 }
364 return;
365 }
366
367 auto *positionComponent = services.world().getComponent<asge::components::position_attr_t>(entityId);
368 if (positionComponent == nullptr) {
369 if (services._debug) {
370 services.debug().log(
371 asge::debug::Level::Warning,
372 "ActionHandler::world teleport: cannot read current position for eid=" + std::to_string(entityId));
373 }
374 return;
375 }
376
377 const uint32_t xValue = resolveArgument(arguments, "x",
378 static_cast<uint32_t>(positionComponent->x), true, remainder);
379 const uint32_t yValue = resolveArgument(arguments, "y",
380 static_cast<uint32_t>(positionComponent->y), true, remainder);
381 if (xValue == asge::world::INVALID_ID || yValue == asge::world::INVALID_ID) {
382 if (services._debug) {
383 services.debug().log(
384 asge::debug::Level::Warning,
385 "ActionHandler::world teleport: failed parsing coordinates in '" + action.action + "'.");
386 }
387 return;
388 }
389
390 try {
391 positionComponent->x = static_cast<float>(xValue);
392 positionComponent->y = static_cast<float>(yValue);
393 std::vector<uint16_t> comp_ids { net::GameProtocol::AvailableComponents::POSITION };
394 network::EngineProtocol::sendEntityComponents(services, entityId, comp_ids);
395
396 if (services._debug) {
397 services.debug().log(
398 asge::debug::Level::Info,
399 "ActionHandler::world teleport: eid=" + std::to_string(entityId) +
400 " -> x:" + std::to_string(xValue) + " y:" + std::to_string(yValue));
401 }
402 }
403 catch (...) {
404 if (services._debug) {
405 services.debug().log(
406 asge::debug::Level::Warning,
407 "ActionHandler::world teleport: failed to teleport eid=" + std::to_string(entityId));
408 }
409 }
410 return;
411 }
412
413 if (services._debug) {
414 services.debug().log(
415 asge::debug::Level::Warning,
416 "ActionHandler::processWorldAction unknown world command '" + command + "'.");
417 }
418}

420/**
421 * @copydoc asge::script::ActionHandler::processScriptActions(Services &, const std::vector<asge::script::QueuedAction> &)
422 */
423void asge::script::ActionHandler::processScriptActions(
424 Services &services, const std::vector<QueuedAction> &actions)
425{
426 for (const auto &queuedAction : actions) {
427 std::istringstream stream(queuedAction.action);
428 std::string service;
429 std::string command;
430 if (!(stream >> service >> command)) {
431 if (services._debug) {
432 services.debug().log(
433 asge::debug::Level::Warning,
434 "ActionHandler::processScriptActions Malformed action: " + queuedAction.action);
435 }
436 continue;
437 }
438
439 std::string remainder;
440 std::getline(stream, remainder);
441 boost::algorithm::trim(remainder);
442
443 if (service == "audio" && services.config().runMode == asge::EngineRunMode::Client) {
444 processAudioAction(command, services, remainder);
445 } else if (service == "world") {
446 processWorldAction(command, services, remainder, queuedAction);
447 } else if (services._debug) {
448 services.debug().log(
449 asge::debug::Level::Warning,
450 "ActionHandler::processScriptActions unknown service '" + service + "'.");
451 }
452 }
