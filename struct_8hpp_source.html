257 /**
258 * @struct Entity
259 * @brief Structure used to store and describe an entity.
260 * @details This structure contains all the necessary data to represent and display any type
261 * and nature of entity across all graphical libraries. It allows rendering functions to receive
262 * a single unified object as input for displaying entities.
263 */
264 struct Entity {
265 EntityType type; ///< Nature/type of the entity (player, wall, etc.).
266 Position pos; ///< Position of the entity on the grid.
267 char display_char; ///< ASCII character used for display (e.g., for terminal-based rendering).
268 uint32_t color; ///< Color used for basic rendering (rectangles, circles).
269 std::string str; ///< Optional text content for text-based entities.
270 Sprite sprite; ///< Sprite used for graphical rendering.
271
272 /**
273 * @brief Equality operator.
274 * @details Compares two entities based on all their meaningful attributes (excluding the sprite).
275 * @param other The entity to compare with.
276 * @return true if all compared attributes are equal; false otherwise.
277 */
278 bool operator==(const Entity &other) const
279 {
280 return type == other.type &&
281 pos == other.pos &&
282 display_char == other.display_char &&
283 color == other.color &&
284 str == other.str;
285 }

300 /**
301 * @union color_u
302 * @brief Represents a color as either a 32-bit integer or an array of 4 components (RGBA).
303 *
304 * @details This union allows accessing a color in two different ways:
305 * - As a single 32-bit value (`color`)
306 * - As four 8-bit components (`components`) representing Red, Green, Blue, and Alpha channels.
307 *
308 * Useful for low-level pixel manipulation or rendering.
309 */
310 typedef union color_u {
311 uint32_t color; ///< The color represented as a 32-bit integer (e.g., 0xRRGGBBAA).
312 uint8_t components[4];///< The color represented as an array: [Red, Green, Blue, Alpha].
313 } color_t;

316 /**
317 * @enum components_e
318 * @brief Enum to index color components (Red, Green, Blue, Alpha) within color_t.components.
319 * @details This enum adapts the component order based on system endianness.
320 * It is intended to be used with the `components` array of the `color_t` union
321 * to access individual color channels in a portable way.
322 * @see color_t
323 *
324 * @note The component order differs between big-endian and little-endian systems:
325 * - On big-endian: R = 0, G = 1, B = 2, A = 3
326 * - On little-endian: A = 0, B = 1, G = 2, R = 3
327 */
328#if __BYTE_ORDER__ == __ORDER_BIG_ENDIAN__
329 enum components_e { R, G, B, A };
330#else
331 enum components_e { A, B, G, R };

336/**
337 * @brief Custom hash function for arcade::types::Position.
338 *
339 * @details This specialization of `std::hash` allows using `arcade::types::Position`
340 * as a key in `std::unordered_map` or other hash-based containers.
341 * It combines the hash of the `x` and `y` components using a variant
342 * of the [Boost hash combine](https://www.boost.org/doc/libs/1_55_0/doc/html/hash/reference.html#boost.hash_combine).
343 *
344 * @see arcade::types::Position
345 */
346namespace std {
347 template<>
348/**
349 * @brief Hash specialization for arcade::types::Position.
350 * @details Enables the use of Position objects as keys in hash-based containers
351 * like std::unordered_map and std::unordered_set.
352 *
353 * The hash function combines the hashes of the x and y coordinates using the
354 * Boost-style hash combining method to reduce collisions.
355 *
356 * @see arcade::types::Position
357 */
358
359 struct hash<arcade::types::Position> {
360 size_t operator()(const arcade::types::Position &pos) const
361 {
362 std::size_t h1 = std::hash<uint32_t>{}(pos.x);
363 std::size_t h2 = std::hash<uint32_t>{}(pos.y);
364
365 return h1 ^ (h2 + 0x9e3779b9 + (h1 << 6) + (h1 >> 2));
366 }
367 };
