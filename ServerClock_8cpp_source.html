68void ServerClock::run()
69{
70 using clock = std::chrono::steady_clock;
71
72 const auto period = std::chrono::duration<double>(1.0 / _hz);
73 auto next = clock::now(); + std::chrono::duration_cast<clock::duration>(period);
74
75 while (_running) {
76 std::this_thread::sleep_until(next);
77 next += std::chrono::duration_cast<clock::duration>(period);
78
79 {
80 std::lock_guard<std::mutex> lk(_mtx);
81 ++_tick_count;
82 }
83 _cv.notify_all();
84 }
85 _cv.notify_all();
86}
