23void rtype::ecs::DrawSystem::update(Coordinator &coord)
24{
25 auto &manager = rtype::client::GameManager::instance();
26 auto &res = manager.resourceManager();
27
28 for (auto entity : _filtered_entities) {
29 auto *sprite = coord.getComponent<sprite_attr_t>(entity);
30 auto *pos = coord.getComponent<position_attr_t>(entity);
31
32 if (!sprite || !pos)
33 continue;
34
35 try {
36 const Texture2D &tex = res.getTexture(sprite->spriteTag);
37 Rectangle src = sprite->rect.width > 0 && sprite->rect.height > 0 ? sprite->rect : Rectangle{0.0f, 0.0f, static_cast<float>(tex.width), static_cast<float>(tex.height)};
38 Vector2 size = sprite->size.x > 0 && sprite->size.y > 0 ? sprite->size : Vector2{static_cast<float>(tex.width), static_cast<float>(tex.height)};
39 Rectangle dst = { static_cast<float>(pos->x), static_cast<float>(pos->y), size.x, size.y };
40
41 DrawTexturePro(tex, src, dst, Vector2{0, 0}, 0.0f, WHITE);
42
43 #ifdef DEBUG
44 DrawRectangleLinesEx(dst, 1.5f, RED);
45 #endif
46 } catch (const std::exception &) {
47 #ifdef DEBUG
48 DrawText(("Missing texture: " + sprite->spriteTag).c_str(), pos->x, pos->y, 10, RED);
49 #endif
50 }
51 }
52}
