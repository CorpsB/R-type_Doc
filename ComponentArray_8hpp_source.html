10/**
11 * @file ComponentArray.hpp
12 * @brief Component storage implementation for the ECS.
13 * @since 0.1
14 * @ingroup ecs
15 * @see IComponentArray.hpp, ComponentManager.hpp
16 *
17 * @details
18 * ComponentArray is a contiguous storage for components of type T mapped to entity IDs.
19 * It provides insertion, removal, lookup and lifecycle integration for component instances.
20 * The implementation uses dense storage with index indirection tables for O(1) insert/remove.
21 */
22

24#include <vector>
25#include <limits>

29/**
30 * @brief Sentinel value used to denote an invalid or missing mapping.
31 */
32static constexpr uint32_t INVALID_ID = std::numeric_limits<uint32_t>::max(); ///< invalid id sentinel

34/**
35 * @class ComponentArray
36 * @brief Dense component container mapping entity IDs to components.
37 *
38 * @tparam T Component type stored in this container.
39 *
40 * @details
41 * ComponentArray stores components in a compact vector and maintains two indirection
42 * vectors to map between component slot indices and their owning entity IDs:
43 * - _component_to_owner: index -> entity_id
44 * - _owner_to_component: entity_id -> index
45 *
46 * This design enables O(1) insertion, removal (by swap-with-last) and lookup.
47 *
48 * @see IComponentArray
49 */
50template<typename T>
51class ComponentArray : public IComponentArray {
52 public:
53 ComponentArray() = default;
54
55 /**
56 * @brief Insert a component instance for a given entity id.
57 *
58 * @param id Entity identifier the component will belong to.
59 * @param component Component instance to insert (copied).
60 * @return true if insertion succeeded; false if the entity already has a component of this type.
61 *
62 * @details
63 * If the internal owner->component mapping is too small to contain \p id, the mapping
64 * vector is resized and initialized with INVALID_ID. The new component is appended to the
65 * dense component vector and mapping tables are updated accordingly.
66 */
67 bool insertComponent(uint32_t id, const T &component);
68
69 /**
70 * @brief Remove the component associated with a given entity id.
71 *
72 * @param id Entity identifier whose component should be removed.
73 * @return true if removal succeeded; false if the entity did not own a component of this type.
74 *
75 * @details
76 * Removal is implemented by swapping the component at \p id's slot with the last component,
77 * updating indirection tables, and popping the vectors to keep storage dense.
78 */
79 bool removeComponent(uint32_t id);
80
81 /**
82 * @brief Return a pointer to the component instance for an entity, or nullptr.
83 *
84 * @param id Entity identifier to query.
85 * @return Pointer to component instance if present; nullptr otherwise.
86 */
87 T* getComponentPtr(uint32_t id);
88
89 /**
90 * @brief Check whether an entity has a component in this container.
91 *
92 * @param id Entity identifier to check.
93 * @return true if entity owns a component of type T; false otherwise.
94 */
95 bool has(uint32_t id) const;
96
97 /**
98 * @brief Handle entity destruction (ECS lifecycle hook).
99 *
100 * @param id Entity identifier that was destroyed.
101 * @details
102 * This override removes the component associated with the destroyed entity, if any.
103 */
104 void entityDestroyed(uint32_t id) override;
105
106 private:
107 std::vector<T> _components; ///< dense array of components
108 std::vector<uint32_t> _component_to_owner; ///< index -> entity id
109 std::vector<uint32_t> _owner_to_component; ///< entity id -> index (INVALID_ID if absent)
110};

112template<typename T>
113inline bool ComponentArray<T>::insertComponent(uint32_t id, const T &component)
114{
115 if (id >= _owner_to_component.size())
116 _owner_to_component.resize(id + 1, INVALID_ID);
117 if (_owner_to_component[id] != INVALID_ID)
118 return false;
119
120 uint32_t idx = _components.size();
121
122 _components.push_back(component);
123 _component_to_owner.push_back(id);
124 _owner_to_component[id] = idx;
125 return true;
126}
127
128template<typename T>
129inline bool ComponentArray<T>::removeComponent(uint32_t id)
130{
131 if (id >= _owner_to_component.size())
132 return false;
133
134 uint32_t idx = _owner_to_component[id];
135 if (idx == INVALID_ID)
136 return false;
137
138 uint32_t last_idx = _components.size() - 1;
139
140 if (idx != last_idx) {
141 _components[idx] = std::move(_components[last_idx]);
142
143 uint32_t mv_entity_id = _component_to_owner[last_idx];
144 _component_to_owner[idx] = mv_entity_id;
145 _owner_to_component[mv_entity_id] = idx;
146 }
147
148 _components.pop_back();
149 _component_to_owner.pop_back();
150 _owner_to_component[id] = INVALID_ID;
151 return true;
152}
153
154template<typename T>
155inline T *ComponentArray<T>::getComponentPtr(uint32_t id)
156{
157 if (id >= _owner_to_component.size())
158 return nullptr;
159
160 uint32_t idx = _owner_to_component[id];
161
162 if (idx == INVALID_ID)
163 return nullptr;
164
165 return &_components[idx];
166}
167
168template<typename T>
169inline bool ComponentArray<T>::has(uint32_t id) const
170{
171 if (id >= _owner_to_component.size())
172 return false;
173
174 return _owner_to_component[id] != INVALID_ID;
175}
176
177template<typename T>
178void ComponentArray<T>::entityDestroyed(uint32_t id)
179{
180 removeComponent(id);
181}
