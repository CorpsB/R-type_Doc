12 * @brief Component storage implementation for the ECS.
13 * @since 0.1

17 * @details

35/**

39 * @tparam T Component type stored in this array.

98template<typename T>
99inline bool ComponentArray<T>::insertComponent(uint32_t id, const T &component)
100{
101 try {
102 if (id >= _owner_to_component.size())
103 _owner_to_component.resize(id + 1, INVALID_ID);
104 if (_owner_to_component[id] != INVALID_ID)
105 throw asge::error::Error(
106 asge::error::ErrorType::World,
107 "ComponentArray::insertComponent",
108 "Entity already has this component type.",
109 asge::error::ErrorSeverity::Warning
110 );
111
112 uint32_t idx = _components.size();
113
114 _components.push_back(component);
115 _component_to_owner.push_back(id);
116 _owner_to_component[id] = idx;
117 return true;
118 }
119 catch (const std::bad_alloc &e) {
120 throw asge::error::Error(
121 asge::error::ErrorType::World,
122 "ComponentArray::insertComponent",
123 std::string("Memory allocation failed: ") + e.what(),
124 asge::error::ErrorSeverity::Fatal
125 );
126 }
127 catch (const asge::error::Error &) {
128 throw;
129 }
130 catch (const std::exception &e) {
131 throw asge::error::Error(
132 asge::error::ErrorType::World,
133 "ComponentArray::insertComponent",
134 std::string("Unexpected exception during insertion: ") + e.what(),
135 asge::error::ErrorSeverity::Recoverable
136 );
137 }
138}

140template<typename T>
141inline bool ComponentArray<T>::removeComponent(uint32_t id)
142{
143 try {
144 if (id >= _owner_to_component.size() || _owner_to_component[id] == INVALID_ID)
145 return false;
146
147 uint32_t idx = _owner_to_component[id];
148 uint32_t last_idx = _components.size() - 1;
149
150 if (idx != last_idx) {
151 _components[idx] = std::move(_components[last_idx]);
152
153 uint32_t mv_entity_id = _component_to_owner[last_idx];
154 _component_to_owner[idx] = mv_entity_id;
155 _owner_to_component[mv_entity_id] = idx;
156 }
157
158 _components.pop_back();
159 _component_to_owner.pop_back();
160 _owner_to_component[id] = INVALID_ID;
161 return true;
162 }
163 catch (const std::out_of_range &e) {
164 throw asge::error::Error(
165 asge::error::ErrorType::World,
166 "ComponentArray::removeComponent",
167 std::string("Invalid index access: ") + e.what(),
168 asge::error::ErrorSeverity::Recoverable
169 );
170 }
171 catch (const asge::error::Error &) {
172 throw;
173 }
174 catch (const std::exception &e) {
175 throw asge::error::Error(
176 asge::error::ErrorType::World,
177 "ComponentArray::removeComponent",
178 std::string("Unexpected exception during removal: ") + e.what(),
179 asge::error::ErrorSeverity::Recoverable
180 );
181 }
182}

184template<typename T>
185inline T *ComponentArray<T>::getComponentPtr(uint32_t id)
186{
187 try {
188 if (id >= _owner_to_component.size())
189 return nullptr;
190
191 uint32_t idx = _owner_to_component[id];
192 if (idx == INVALID_ID)
193 return nullptr;
194
195 return &_components.at(idx);
196 }
197 catch (const std::out_of_range &e) {
198 throw asge::error::Error(
199 asge::error::ErrorType::World,
200 "ComponentArray::getComponentPtr",
201 std::string("Component index out of range: ") + e.what(),
202 asge::error::ErrorSeverity::Recoverable
203 );
204 }
205 catch (const asge::error::Error &) {
206 throw;
207 }
208 catch (const std::exception &e) {
209 throw asge::error::Error(
210 asge::error::ErrorType::World,
211 "ComponentArray::getComponentPtr",
212 std::string("Unexpected exception while accessing component: ") + e.what(),
213 asge::error::ErrorSeverity::Recoverable
214 );
215 }
