303 bool compareDoubles(const std::string &operation, double leftValue, double rightValue)
304 {
305 if (operation == "<") {
306 return leftValue < rightValue;
307 }
308 if (operation == "<=") {
309 return leftValue <= rightValue;
310 }
311 if (operation == "==") {
312 return leftValue == rightValue;
313 }
314 if (operation == ">=") {
315 return leftValue >= rightValue;
316 }
317 if (operation == ">") {
318 return leftValue > rightValue;
319 }
320 return false;
321 }

329std::optional<asge::script::ScriptHandle> asge::script::JSONRuntime::parseString(
330 const std::string &path, const std::string &content)
331{
332 nlohmann::json document;
333 try {
334 document = nlohmann::json::parse(content);
335 }
336 catch (const std::exception &exception) {
337 if (_debugger) {
338 _debugger->log(debug::Level::Error,
339 std::string("[JSONRuntime] template parse error ") + path + " : " + exception.what());
340 }
341 return std::nullopt;
342 }
343
344 if (!document.contains("id") || !document["id"].is_string()) {
345 if (_debugger) {
346 _debugger->log(
347 debug::Level::Warning, "[JSONRuntime] missing/invalid id in " + path);
348 }
349 return std::nullopt;
350 }
351
352 ScriptHandle handle;
353 handle.id = document["id"].get<std::string>();
354 handle.src_path = path;
355 handle.language = "json";
356 handle.version = 1;
357
358 if (document.contains("extends") && document["extends"].is_string()) {
359 handle.extends = document["extends"].get<std::string>();
360 } else {
361 handle.extends = std::nullopt;
362 }
363
364 if (document.contains("components") && document["components"].is_object()) {
365 handle.components = document["components"];
366 } else {
367 handle.components = nlohmann::json::object();
368 }
369
370 if (document.contains("behavior") && document["behavior"].is_object()) {
371 auto &behaviorJson = document["behavior"];
372 if (behaviorJson.contains("on_create") && behaviorJson["on_create"].is_array()) {
373 for (auto &entry : behaviorJson["on_create"]) {
374 if (entry.is_string()) {
375 handle.behavior.on_create.push_back(entry.get<std::string>());
376 }
377 }
378 }
379 if (behaviorJson.contains("on_update") && behaviorJson["on_update"].is_array()) {
380 for (auto &entry : behaviorJson["on_update"]) {
381 if (entry.is_string()) {
382 handle.behavior.on_update.push_back(entry.get<std::string>());
383 }
384 }
385 }
386 if (behaviorJson.contains("on_position") && behaviorJson["on_position"].is_array()) {
387 for (auto &triggerJson : behaviorJson["on_position"]) {
388 PositionTrigger trigger;
389 if (triggerJson.contains("condition")) {
390 trigger.condition = triggerJson["condition"];
391 }
392 if (triggerJson.contains("actions") && triggerJson["actions"].is_array()) {
393 for (auto &actionJson : triggerJson["actions"]) {
394 if (actionJson.is_string()) {
395 trigger.actions.push_back(actionJson.get<std::string>());
396 }
397 }
398 }
399 if (triggerJson.contains("once") && triggerJson["once"].is_boolean()) {
400 trigger.once = triggerJson["once"].get<bool>();
401 }
402 if (triggerJson.contains("throttle_ms") && triggerJson["throttle_ms"].is_number_unsigned()) {
403 trigger.throttle_ms = triggerJson["throttle_ms"].get<std::uint32_t>();
404 }
405 handle.behavior.on_position.push_back(std::move(trigger));
406 }
407 }
408 if (behaviorJson.contains("on_collision") && behaviorJson["on_collision"].is_array()) {
409 for (auto &entry : behaviorJson["on_collision"]) {
410 if (entry.is_string()) {
411 handle.behavior.on_collision.push_back(entry.get<std::string>());
412 }
413 }
414 }
415 if (behaviorJson.contains("on_death") && behaviorJson["on_death"].is_array()) {
416 for (auto &entry : behaviorJson["on_death"]) {
417 if (entry.is_string()) {
418 handle.behavior.on_death.push_back(entry.get<std::string>());
419 }
420 }
421 }
422 }
423
424 return handle;
425}

443asge::script::ScriptHandle asge::script::JSONRuntime::mergeHandles(
444 const ScriptHandle &parent, const ScriptHandle &child)
445{
446 ScriptHandle mergedHandle;
447 mergedHandle.id = child.id;
448 mergedHandle.src_path = child.src_path;
449 mergedHandle.language = child.language;
450 mergedHandle.version = std::max(parent.version, child.version) + 1;
451 mergedHandle.extends = std::nullopt;
452
453 try {
454 nlohmann::json mergedComponents = parent.components;
455 for (auto &componentEntry : child.components.items()) {
456 mergedComponents[componentEntry.key()] = componentEntry.value();
457 }
458 mergedHandle.components = std::move(mergedComponents);
459 }
460 catch (const std::exception &exception) {
461 throw asge::error::Error(
462 asge::error::ErrorType::Script,
463 "asge::script::JSONRuntime::mergeHandles",
464 exception.what(),
465 asge::error::ErrorSeverity::Recoverable);
466 }
467
468 BehaviorBlock behavior;
469 behavior.on_create = parent.behavior.on_create;
470 behavior.on_create.insert(
471 behavior.on_create.end(), child.behavior.on_create.begin(), child.behavior.on_create.end());
472 behavior.on_update = parent.behavior.on_update;
473 behavior.on_update.insert(
474 behavior.on_update.end(), child.behavior.on_update.begin(), child.behavior.on_update.end());
475 behavior.on_position = parent.behavior.on_position;
476 behavior.on_position.insert(
477 behavior.on_position.end(), child.behavior.on_position.begin(), child.behavior.on_position.end());
478 behavior.on_collision = parent.behavior.on_collision;
479 behavior.on_collision.insert(
480 behavior.on_collision.end(), child.behavior.on_collision.begin(), child.behavior.on_collision.end());
481 behavior.on_death = parent.behavior.on_death;
482 behavior.on_death.insert(
483 behavior.on_death.end(), child.behavior.on_death.begin(), child.behavior.on_death.end());
484 mergedHandle.behavior = std::move(behavior);
485
486 return mergedHandle;
487}

623bool asge::script::JSONRuntime::evaluateCondition(
624 const ScriptValue &condition, double x, double y)
625{
626 if (!condition.is_object()) {
627 return false;
628 }
629
630 auto getOperand = [&](const std::string &key) -> std::optional<double> {
631 if (key == "x") {
632 return x;
633 }
634 if (key == "y") {
635 return y;
636 }
637 return std::nullopt;
638 };
639
640 for (auto &conditionEntry : condition.items()) {
641 const std::string operandKey = conditionEntry.key();
642 const ScriptValue &expression = conditionEntry.value();
643
644 auto leftOperand = getOperand(operandKey);
645 if (!leftOperand) {
646 return false;
647 }
648 double leftValue = *leftOperand;
649
650 if (!expression.is_object()) {
651 return false;
652 }
653
654 for (auto &comparison : expression.items()) {
655 const std::string operation = comparison.key();
656 const ScriptValue rightOperandSpecification = comparison.value();
657
658 auto rightOperand = toOptionalDouble(rightOperandSpecification);
659 if (!rightOperand) {
660 return false;
661 }
662
663 if (!compareDoubles(operation, leftValue, *rightOperand)) {
664 return false;
665 }
666 }
667 }
668 return true;
669}
