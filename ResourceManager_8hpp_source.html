8/**
9 * @file ResourceManager.hpp
10 * @brief Declaration of the ResourceManager singleton handling game asset caching.
11 * @since 1.0
12 * @ingroup client-core
13 * @see Texture2D, Font, Sound, GameSettings
14 *
15 * @details
16 * The ResourceManager provides a global caching and lifecycle management system
17 * for Raylib resources such as textures, fonts, and sounds. It ensures assets
18 * are loaded only once and shared efficiently across the client.
19 *
20 * ### Responsibilities:
21 * - Load and store resources using string-based keys.
22 * - Provide fast retrieval through hash map lookup.
23 * - Release all loaded assets during shutdown.
24 *
25 * The singleton pattern guarantees safe, centralized access to all resources.
26 */

36/**
37 * @class ResourceManager
38 * @brief Singleton responsible for loading, caching, and managing Raylib assets.
39 *
40 * @details
41 * This class handles all texture, font, and sound resource operations in the client.
42 * Assets are cached in internal hash maps to prevent duplicate loading and improve
43 * performance. When shutting down, all resources are safely unloaded to free memory.
44 *
45 * ### Example:
46 * @code{.cpp}
47 * auto &res = ResourceManager::getInstance();
48 * res.loadTexture("player", "assets/player.png");
49 * Texture2D &player = res.getTexture("player");
50 * DrawTexture(player, 100, 100, WHITE);
51 * res.unloadAll();
52 * @endcode
53 *
54 * @note Follows the Meyers Singleton pattern for thread-safe initialization.
55 */
56class ResourceManager {
57 public:
58 /**
59 * @brief Retrieves the global ResourceManager instance.
60 * @return Reference to the unique ResourceManager.
61 *
62 * @details
63 * Implements the Meyers Singleton pattern, ensuring thread-safe
64 * initialization and unique instance persistence.
65 */
66 static ResourceManager &getInstance();
67
68 /**
69 * @brief Loads a texture into memory if not already cached.
70 * @param key Unique identifier for the texture.
71 * @param path Path to the texture file (e.g., "assets/sprites/player.png").
72 *
73 * @throws rtype::exception::Error If loading fails.
74 * @note If the key already exists, this function does nothing.
75 */
76 void loadTexture(const std::string &key, const std::string &path);
77
78 /**
79 * @brief Retrieves a previously loaded texture.
80 * @param key Lookup key for the texture.
81 * @return Reference to the corresponding Texture2D object.
82 * @throws rtype::exception::Error If the texture key is not found.
83 */
84 Texture2D &getTexture(const std::string &key);
85
86 /**
87 * @brief Loads a font from disk if not already cached.
88 * @param key Unique identifier for the font.
89 * @param path Path to the font file (e.g., "assets/fonts/Roboto.ttf").
90 * @param size Font size in pixels (default: 32).
91 *
92 * @throws rtype::exception::Error If loading fails.
93 * @note Unicode support is enabled via extended codepoint ranges.
94 */
95 void loadFont(const std::string &key, const std::string &path, int size = 32);
96
97 /**
98 * @brief Retrieves a cached font.
99 * @param key Lookup key for the font.
100 * @return Reference to the associated Font object.
101 * @throws rtype::exception::Error If the font key is not found.
102 */
103 Font &getFont(const std::string &key);
104
105 /**
106 * @brief Loads a sound into memory if not already cached.
107 * @param key Unique identifier for the sound.
108 * @param path Path to the sound file (e.g., "assets/sounds/explosion.wav").
109 *
110 * @throws rtype::exception::Error If loading fails.
111 * @note Managed via Raylib’s LoadSound()/UnloadSound() API.
112 */
113 void loadSound(const std::string &key, const std::string &path);
114
115 /**
116 * @brief Retrieves a previously loaded sound.
117 * @param key Lookup key for the sound.
118 * @return Reference to the associated Sound object.
119 * @throws rtype::exception::Error If the sound key is not found.
120 */
121 Sound &getSound(const std::string &key);
122
123 /**
124 * @brief Unloads all cached textures, fonts, and sounds.
125 *
126 * @details
127 * Iterates through every cache and releases all Raylib resources.
128 * Clears all internal maps afterward.
129 *
130 * @post All cached resources are fully unloaded from memory.
131 */
132 void unloadAll();
133 private:
134 ResourceManager() {}; ///< Private constructor enforcing singleton usage.
135
136 ResourceManager(const ResourceManager &) = delete; ///< Copy constructor disabled.
137 ResourceManager &operator=(const ResourceManager &) = delete; ///< Assignment disabled.
138
139 std::unordered_map<std::string, Texture2D> _textures; ///< Cached textures.
140 std::unordered_map<std::string, Font> _fonts; ///< Cached fonts.
141 std::unordered_map<std::string, Sound> _sounds; ///< Cached sounds.
142};
