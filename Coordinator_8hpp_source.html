10/**
11 * @file Coordinator.hpp
12 * @brief High-level facade that connects entity/component/system managers.
13 * @since 0.1
14 * @ingroup ecs
15 * @see EntityManager.hpp, ComponentManager.hpp, SystemManager.hpp
16 *
17 * @details
18 * Coordinator provides a simple façade over the three central ECS managers:
19 * - EntityManager: allocation and lifecycle of entity ids,
20 * - ComponentManager: registration and storage of component arrays,
21 * - SystemManager: registration of systems and maintenance of execution signatures.
22 *
23 * The Coordinator exposes convenience methods to create/destroy entities, to
24 * register/add/remove/get components and to register/deregister/configure systems.
25 * Template operations are forwarded to the appropriate manager. This header
26 * contains inline template definitions that must be visible to translation units
27 * using the Coordinator API.
28 */

38/**
39 * @class Coordinator
40 * @brief Central coordinator that orchestrates entities, components and systems.
41 *
42 * @details
43 * The Coordinator owns the three manager instances and exposes a compact API
44 * that game code and higher-level subsystems can use to interact with the ECS.
45 * It is intended to reduce boilerplate by forwarding template operations to the
46 * correct manager and triggering necessary bookkeeping (for instance: updating
47 * an entity signature and notifying systems when a component is added/removed).
48 *
49 * @see EntityManager, ComponentManager, SystemManager
50 */

190template<typename T>
191inline bool Coordinator::addComponent(uint32_t id, const T &component)
192{
193 if (!_component_manager->addComponent<T>(id, component))
194 return false;
195
196 auto &signature = _entity_manager->getSignature(id);
197 signature.set(_component_manager->getComponentTypeID<T>(), true);
198 _entity_manager->setSignature(id, signature);
199
200 _system_manager->entitySignatureChanged(id, signature);
201 return true;
202}
203
204template<typename T>
205inline bool Coordinator::removeComponent(uint32_t id)
206{
207 if (!_component_manager->removeComponent<T>(id))
208 return false;
209
210 auto &signature = _entity_manager->getSignature(id);
211 signature.set(_component_manager->getComponentTypeID<T>(), false);
212 _entity_manager->setSignature(id, signature);
213
214 _system_manager->entitySignatureChanged(id, signature);
215 return true;
216}
