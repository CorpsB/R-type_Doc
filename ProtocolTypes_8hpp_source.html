34/**
35 * @namespace protocol
36 * @brief Core transport primitives and low-level protocol types.
37 *
38 * @details
39 * The `protocol` namespace contains the foundational types and constants used
40 * by the UDP transport core: wire packet header layout (`packet_header_t`),
41 * runtime packet wrapper (`packet_t`), reliability flags (`flag_t`), event
42 * payloads and diagnostic helpers. It also declares small utility containers
43 * used by the core (pending send state, per-peer receive state) and constants
44 * that control retransmit behaviour and maximum packet size.
45 *
46 * All multi-byte fields on the wire are encoded in network byte order (big
47 * endian). Implementations that construct or parse packet headers MUST convert
48 * between host and network byte order using the provided helpers (e.g.
49 * `protocol::utils::to_network_u32` / `protocol::utils::from_network_u32`).
50 *
51 * @since 0.1
52 * @ingroup protocol
53 */

71/**
72 * @enum flag_t
73 * @brief Reliability semantics encoded in packet header flags.
74 *
75 * @details
76 * - UNRELIABLE: do not retransmit (used for time-sensitive state).
77 * - RELIABLE_UNORDERED: retransmit until ack; order not enforced.
78 * - RELIABLE_ORDERED: retransmit and deliver in sequence (buffer until missing).
79 */
80enum class flag_t : uint8_t {
81 UNRELIABLE = 0, ///< no retransmit
82 RELIABLE_UNORDERED = 1, ///< retransmit; unordered
83 RELIABLE_ORDERED = 2 ///< retransmit; ordered delivery
84};

86#pragma pack(push, 1)
87/**
88 * @struct packet_header_t
89 * @brief Fixed-size packet header at the beginning of each UDP datagram.
90 *
91 * @details
92 * Wire layout (packed, network byte order):
93 * - uint16_t message_type
94 * - uint8_t flags
95 * - uint8_t reserved
96 * - uint32_t sequence
97 * - uint32_t ack
98 * - uint32_t ack_bits
99 * - uint16_t payload_length
100 *
101 * @warning Implementations MUST serialize fields to big-endian before sending.
102 */
103struct packet_header_t {
104 uint16_t message_type; ///< application message type (numeric)
105 flag_t flags; ///< bitfield (low bits = reliability)
106 uint32_t sequence; ///< sender sequence number
107 uint32_t ack; ///< last sequence number seen from peer
108 uint32_t ack_bits; ///< bitmask for ack-1 .. ack-32 (lsb = ack-1)
109 uint16_t payload_length; ///< length in bytes of the payload that follows
110};
111#pragma pack(pop)

113/**
114 * @enum event_type_t
115 * @brief Event categories produced by the Protocol core.

117enum class event_type_t : uint8_t {
118 NONE = 0, ///< no event
119 PACKET_RECEIVED, ///< a packet was received
120 CONNECTION_LOST, ///< remote timed out / unreachable
121 CONNECTION_ESTABLISHED, ///< bind/connect succeeded
122 TRANSMIT_FAILED ///< reliable send failed after retries
123};

125/**
126 * @struct transmit_failed_t
127 * @brief Payload delivered when a reliable transmission exhausts retries.
128 */
129struct transmit_failed_t {
130 uint32_t sequence; ///< failed packet sequence

132 byte_buffer_t payload; ///< payload that could not be delivered
133 udp::endpoint remote; ///< target endpoint

174/**
175 * @struct core_event_t
176 * @brief Core protocol event delivered by the low-level UDP transport.
177 *
178 * @details
179 * The Protocol core produces discrete events that the upper layers consume.
180 * Each core_event_t contains:
181 * - type: an event_type_t enumerator describing the kind of event
182 * (PACKET_RECEIVED, CONNECTION_ESTABLISHED, CONNECTION_LOST, TRANSMIT_FAILED).
183 * - payload: a variant carrying the event data:
184 * - std::monostate for NONE,
185 * - packet_t for PACKET_RECEIVED (parsed header + payload + remote endpoint),
186 * - transmit_failed_t for TRANSMIT_FAILED (reliable send that exhausted retries).
187 *
188 * Usage:
189 * - Consumers call Protocol::get_event() which fills a core_event_t.
190 * - For PACKET_RECEIVED events the payload holds the parsed packet that can be
191 * passed to higher-level protocol adapters (e.g. game facade).
192 */
193struct core_event_t {
194 event_type_t type;
195 core_event_payload_t payload;
196};
