38/**
39 * @brief A single position-trigger entry (as represented in JSON).
40 *
41 * @details
42 * PositionTrigger stores the raw condition (kept as JSON for later evaluation),
43 * the list of action strings to execute when the condition matches, a flag to run
44 * the trigger only once, and an optional throttle in milliseconds to limit re-firing.
45 */
46struct PositionTrigger {
47 nlohmann::json condition; ///< Raw condition object (runtime evaluates this).
48 std::vector<std::string> actions; ///< Actions represented as raw strings (parsed/executed later).
49 bool once = false; ///< True if the trigger should run only a single time.
50 std::uint32_t throttle_ms = 0; ///< Minimum milliseconds between consecutive activations.
51};
52
53/**
54 * @brief Behavior collection for a template/prefab.
55 *
56 * @details
57 * Contains lists of action strings for common lifecycle hooks and position-based
58 * triggers that are evaluated by the engine at runtime.
59 */
60struct BehaviorBlock {
61 std::vector<std::string> on_create; ///< Actions executed when an entity/template is created.
62 std::vector<std::string> on_update; ///< Actions executed each update tick.
63 std::vector<PositionTrigger> on_position; ///< Position-based triggers evaluated against an entity position.
64 std::vector<std::string> on_collision; ///< Actions executed upon collision events.
65 std::vector<std::string> on_death; ///< Actions executed when the entity dies.
66};
67
68/**
69 * @brief ScriptHandle: describes a loaded script/template.
70 *
71 * @details
72 * Holds metadata, raw JSON components and behavior hooks for a parsed template.
73 * The handle can optionally reference a parent template via the `extends` field.
74 */
75struct ScriptHandle {
76 std::string id; ///< Template identifier, e.g. "EnemyBase".
77 std::string src_path; ///< Logical source path the template was loaded from.
78 std::string language = "json"; ///< Runtime language identifier ("json", etc.).
79 std::uint64_t version = 0; ///< Version number incremented on reload/merge.
80 nlohmann::json components; ///< Raw components object used to spawn ECS components.
81 BehaviorBlock behavior; ///< Behavior hooks attached to the template.
82 std::optional<std::string> extends; ///< Optional parent template id (value of top-level "extends").
83};
84
85/**
86 * @brief Queued action produced by ScriptManager when hooks run or entity created.
87 *
88 * @details
89 * ScriptManager enqueues these actions for later processing by the engine's
90 * script execution pipeline. An entity_id of 0 indicates the action is not
91 * bound to a concrete entity (template-level or spawn request).
92 */
93struct QueuedAction {
94 std::uint32_t entity_id = 0; ///< Entity identifier (0 means unbound/template-level).
95 std::string template_id; ///< Originating template id (if any).
96 std::string action; ///< Raw action string (not parsed).
97};

99/**
100 * @brief Sprite asset description (one JSON file defines a single sprite).
101 */
102struct SpriteAsset {
103 std::string id; ///< Unique sprite identifier.
104 std::string link; ///< Target resource reference (texture path or atlas key).
105 int width = 0; ///< Sprite width in pixels.
106 int height = 0; ///< Sprite height in pixels.
107 double scale = 1.0; ///< Default render scale.
108 int top = 0; ///< Top offset within the atlas/texture sheet.
109 int left = 0; ///< Left offset within the atlas/texture sheet.
110 std::string src_path; ///< Logical path used to load the asset.
111};

113/**
114 * @brief Audio asset description (one JSON file defines a single audio resource).
115 */
116struct AudioAsset {
117 std::string id; ///< Unique audio identifier.
118 std::string link; ///< Resource reference (file path or alias).
119 double volume = 1.0; ///< Default playback volume.
120 bool loop = false; ///< Whether the audio should loop by default.
121 std::string src_path; ///< Logical path recorded for debugging or hot-reload.
122};
